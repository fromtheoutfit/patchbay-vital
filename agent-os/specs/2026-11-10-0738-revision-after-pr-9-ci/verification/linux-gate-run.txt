#!/usr/bin/env bash
# The gate as CI runs it, in a throwaway ubuntu:24.04 container, on a copy of the tree.
# Usage (inside the container): linux-gate.sh fixed|old-host
# /src is this repository and /standards ../patchbay-standards, both mounted read-only; nothing is
# written to either. old-host puts host.py back at HEAD in the copy, so the new tests run against
# the old lookup.
set -uo pipefail
mode=${1:?fixed or old-host}

echo "== setup (apt: git ca-certificates curl, and libatomic1, which pedalboard's native module links"
echo "   and CI's runner image has; uv from astral.sh at the version pyproject.toml pins)"
export DEBIAN_FRONTEND=noninteractive
apt-get update -qq >/dev/null && apt-get install -y -qq git ca-certificates curl libatomic1 >/dev/null 2>&1 || exit 3
curl -LsSf https://astral.sh/uv/0.13.7/install.sh | env UV_INSTALL_DIR=/usr/local/bin UV_NO_MODIFY_PATH=1 sh >/dev/null 2>&1 || exit 3

mkdir -p /work/patchbay-vital /work/patchbay-standards
tar -C /src --exclude=./.venv --exclude=./.mypy_cache --exclude=./.ruff_cache \
    --exclude=./.pytest_cache --exclude=./.hypothesis -cf - . \
  | tar -C /work/patchbay-vital --no-same-owner -xf - || exit 3
tar -C /standards -cf - . | tar -C /work/patchbay-standards --no-same-owner -xf - || exit 3
git config --global --add safe.directory '*'
cd /work/patchbay-vital || exit 3
if [ "$mode" = old-host ]; then
  git checkout HEAD -- src/patchbay_vital/host.py || exit 3
  echo "== old-host: src/patchbay_vital/host.py put back at $(git rev-parse --short HEAD) in the copy"
fi

echo "== machine: $(uname -sm); $(. /etc/os-release && echo "$PRETTY_NAME"); $(uv --version)"
echo "== branch: $(git rev-parse --abbrev-ref HEAD) at $(git rev-parse --short HEAD), working tree as mounted"
echo "== git status --short (the copy)"
git status --short

echo
echo '$ uv sync --locked'
uv sync --locked 2>&1 | tail -3
echo
echo '$ scripts/gate.sh'
scripts/gate.sh
gate_exit=$?
echo
echo "exit=$gate_exit"
