"""Issue #16's spec: `main`'s shipped methods against this branch's, directly, on every value.

The fifth review's direct count, for this change. Nothing is reproduced. Two pools of processes
each import `patchbay_vital` from one tree (`pools.py`), `main`'s (an extract of its commit) or
this branch's, and each calls its own tree's scalar methods on every float32 normalized value r
of a record, as a caller would:

  P0 = scale.to_preset(r)            the preset the parameter holds
  D0 = record.from_normalized(r)     its display
  TP = record.to_preset(D0)          the display's preset
  TN = record.to_normalized(D0)      the normalized value a set of the display sends
  HN = scale.to_normalized(P0)       the held preset's normalized value

With `--columns five` all five are called in both trees and compared. With `--columns two` only
the two methods issue #16's change can reach are: D0 is still made by each tree's own
`from_normalized`, as the input, and compared; P0 and HN are not called in `main`'s tree
(`envelope_group.py` shows their statements are the same in both trees).

This process compares the two trees on every r and counts, by binade of r:

  S!=, D!=   P0 or D0 differs
  PRESET     TP differs from `main`'s
  NORM       TN differs from `main`'s
  HELDNORM   HN differs from `main`'s
  ANY        any compared value differs

and, from this branch's own result TP on a linear scale, what the rule did. P1 is the curve's
float32 for D0, kept within the scale; `landed` is what a set of P1 holds,
scale.to_preset(scale.to_normalized(P1)); `other` is the float32 next to P1 away from `landed`:

  KEPT       TP == P1 and a set of it holds it (0 <= TN <= 1 and scale.to_preset(TN) == TP)
  PLUGIN     TP == landed != P1 and landed shows D0: what a set holds, Vital's own rounding
  OTHER      TP == other, landed != P1: the rule's one exception
  FELL       TP == landed != P1 and landed does not show D0: what a set holds where neither
             candidate shows the display. D0 is a display a held preset shows, so this is expected
             on no value: it is where `main` kept the curve's float32, its third branch.
  UNKNOWN    none of these

On `volume` every count is also split at its floor exactly, by this branch's P0 <= 2 ** -43 and
P0 > 2 ** -43, value by value, never by binade. On another record the split's two rows are not
printed: with `--columns two` its P0 is not called, and every value counts as above the floor.

OUT counts what the run stops on, at once: ANY difference from `main`, any FELL, any UNKNOWN.
Every display counted is one a held preset shows, where issue #16 changes nothing; a difference
is outside what the rule is meant to change. `--control` counts the same and does not stop.

Run from the repository root:
  uv run --locked python direct_count.py --main <main's src> --branch <this branch's src>
      --record <id> --columns five|two [--from <binade> --to <binade>] [--time [--budget <hours>]]
      [--control] [--label <text>] [--main-processes <n>] [--branch-processes <n>]
"""

import argparse
import math
import sys
import time

import numpy as np

import pools

FLOOR = 2.0**-43
CHUNK = 1 << 16
SPANS = [("zero", 0, 1), ("sub", 1, 1 << 23)]
SPANS += [(str(e - 127), e << 23, (e + 1) << 23) for e in range(1, 127)]
SPANS += [("one", 0x3F800000, 0x3F800001)]
COMPARED = {"five": ("S!=", "D!=", "PRESET", "NORM", "HELDNORM"), "two": ("D!=", "PRESET", "NORM")}
ROWS = {"five": (0, 1, 2, 3, 4), "two": (1, 2, 3)}
CLASSES = ("KEPT", "PLUGIN", "OTHER", "FELL", "UNKNOWN")


def work(job):
    """Call one tree's methods on every float32 normalized value of a bit-pattern range."""
    name, lo, hi, five, classify = job
    record = pools.record(name)
    scale, curve = record.scale, record.curve
    linear = scale.kind == "linear"
    split = classify and name == "volume"  # this branch's P0, for the split at volume's floor
    values = np.arange(lo, hi, dtype=np.uint32).view(np.float32).astype(np.float64).tolist()
    out = np.full((5, len(values)), np.nan)
    rule = np.full(len(values), -1, np.int8)
    step = pools.TREE["float32"].next_float32
    for i, r in enumerate(values):
        d0 = record.from_normalized(r)
        tp = record.to_preset(d0)
        tn = record.to_normalized(d0)
        if five or split:
            p0 = scale.to_preset(r)
            out[0, i] = p0
        if five:
            out[4, i] = scale.to_normalized(p0)
        out[1:4, i] = (d0, tp, tn)
        if not (classify and linear):
            continue
        p1 = min(max(curve.to_preset(d0), scale.minimum), scale.maximum)
        if tp == p1:
            rule[i] = 0 if 0.0 <= tn <= 1.0 and scale.to_preset(tn) == tp else 4
            continue
        landed = scale.to_preset(scale.to_normalized(p1))
        if landed == p1:
            rule[i] = 4
        elif tp == landed:
            rule[i] = 1 if record.to_display(tp) == d0 else 3
        elif tp == step(p1, math.copysign(math.inf, p1 - landed)):
            rule[i] = 2
        else:
            rule[i] = 4
    return lo, out, rule


def tally(columns, ours, theirs, rule):
    """Return the counts at or below the floor and above it, and the values the run stops on."""
    differs = [ours[k] != theirs[k] for k in ROWS[columns]]
    anything = np.logical_or.reduce(differs)
    classes = [rule == c for c in range(len(CLASSES))]
    out = anything | classes[3] | classes[4]
    flags = (*differs, anything, *classes, out)
    below = ours[0] <= FLOOR  # a NaN, where P0 was not called, is above
    rows = [[int(np.count_nonzero(side)), *(int(np.count_nonzero(f & side)) for f in flags)]
            for side in (below, ~below)]
    return np.array(rows, dtype=np.int64), np.flatnonzero(out)


def spans(first, last):
    labels = [label for label, _, _ in SPANS]
    lo = 0 if first is None else labels.index(first)
    hi = len(SPANS) - 1 if last is None else labels.index(last)
    return SPANS[lo : hi + 1]


def line(label, row):
    return f"{label:>12} " + " ".join(f"{int(v):>11}" for v in row)


def timing(open_pools, sizes, name, five):
    """Time each tree on 200,000 values: 4 ranges of 50,000, at once, in 4 of its processes."""
    rng = np.random.default_rng(21)
    report = {}
    starts = rng.integers(0x10000000, 0x3F000000, 4, dtype=np.uint32).tolist()
    for side, pool in open_pools.items():
        started = time.time()
        jobs = [pool.apply_async(work, ((name, start, start + 50_000, five, side == "branch"),))
                for start in starts]
        for job in jobs:
            job.get()
        report[side] = (time.time() - started) / 50_000 * 1e6
        print(f"  {side}: {report[side]:.1f} us a value, 4 ranges of 50,000 values in 4 processes"
              " at once")
    total = 1_065_353_217
    wall = max(report[s] / sizes[s] for s in sizes) * total / 3.6e9
    cpu = sum(report.values()) * total / 3.6e9
    print(f"   every float32 normalized value of {name}, {total:,}: about {cpu:.1f} CPU-hours,"
          f" {wall:.2f} hours wall on these pools")
    return wall


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--main", required=True)
    parser.add_argument("--branch", required=True)
    parser.add_argument("--record", required=True)
    parser.add_argument("--columns", choices=("five", "two"), required=True)
    parser.add_argument("--from", dest="first")
    parser.add_argument("--to", dest="last")
    parser.add_argument("--main-processes", type=int, default=6)
    parser.add_argument("--branch-processes", type=int, default=7)
    parser.add_argument("--time", action="store_true")
    parser.add_argument("--control", action="store_true")
    parser.add_argument("--label", default="")
    parser.add_argument("--budget", type=float, default=5.0)
    args = parser.parse_args()
    five = args.columns == "five"
    sizes = {"main": args.main_processes, "branch": args.branch_processes}
    open_pools = pools.open_pools({"main": args.main, "branch": args.branch}, sizes)
    if args.time:
        print(f"== timing on {args.record}, --columns {args.columns}")
        wall = timing(open_pools, sizes, args.record, five)
        for pool in open_pools.values():
            pool.close()
        if wall > args.budget:
            print(f"   more than the {args.budget} hours a record the owner allowed: stopping")
            return 3
        return 0
    columns = ("n", *COMPARED[args.columns], "ANY", *CLASSES, "OUT")
    chosen = spans(args.first, args.last)
    print(f"== {args.record}{args.label}, --columns {args.columns}: binades {chosen[0][0]} to"
          f" {chosen[-1][0]}, {sum(hi - lo for _, lo, hi in chosen):,} float32 normalized values,"
          " each through both trees")
    print(line("binade", ()) + " ".join(f"{c:>11}" for c in columns))
    started = time.time()
    total = np.zeros((2, len(columns)), dtype=np.int64)
    quiet = []
    kept = columns.index("KEPT")
    for label, lo, hi in chosen:
        row = np.zeros((2, len(columns)), dtype=np.int64)
        jobs = []
        for start in range(lo, hi, CHUNK):
            end = min(hi, start + CHUNK)
            jobs.append((
                open_pools["main"].apply_async(work, ((args.record, start, end, five, False),)),
                open_pools["branch"].apply_async(work, ((args.record, start, end, five, True),)),

            _, ours, rule = ours_job.get()

        total += row

"""Issue #16's spec: the envelope records are one group, and which of the count's calls can differ.

Two things the direct count of `env_1_delay` rests on, each read from the shipped source of both
trees, `main`'s and this branch's, imported by one pool a tree (`pools.py`):

1. The 12 `env_N_delay` and `env_N_hold` records have one group key, every field the map's
   methods read of a record: the parameter's kind and bounds, the scale's kind and bounds, the
   curve's kind, multiply and offset. So `env_1_delay`'s count is the group's.

2. Which methods can give another value than `main`'s. `parameters.py` and `float32.py` are
   parsed in each tree, every docstring dropped, and each function's statements compared between
   the trees. Then, for each of the five calls the count makes on a value, the functions it can
   reach are followed through the source: a call of a name, a method or property read off `self`,
   off a `scale` (NormalizedScale), a `curve` (DisplayCurve) or a `record` (MappedParameter), and
   a module-level table a function reads. A call whose every reachable function has the same
   statements in both trees cannot differ from `main`'s; the others are the ones to count. A
   method-named attribute whose receiver this cannot place is printed, so nothing is passed over;
   one read off a `parameter` is core's `Parameter`'s own field, outside these two files.

Run from the repository root:
  uv run --locked python envelope_group.py --main <main's src> --branch <this branch's src>
"""

import argparse
import ast
import sys

import pools

ENVELOPE = [f"env_{n}_{part}" for n in range(1, 7) for part in ("delay", "hold")]
FIELDS = ("parameter.kind", "parameter.minimum", "parameter.maximum", "scale.kind",
          "scale.minimum", "scale.maximum", "curve.kind", "curve.multiply", "curve.offset")
CALLS = {
    "P0 = scale.to_preset(r)": "NormalizedScale.to_preset",
    "D0 = record.from_normalized(r)": "MappedParameter.from_normalized",
    "TP = record.to_preset(D0)": "MappedParameter.to_preset",
    "TN = record.to_normalized(D0)": "MappedParameter.to_normalized",
    "HN = scale.to_normalized(P0)": "NormalizedScale.to_normalized",
}
RECEIVERS = {"scale": "NormalizedScale", "curve": "DisplayCurve", "record": "MappedParameter"}
OUTSIDE = {"parameter"}  # core's `Parameter`, a model of patchbay-core: its fields are not these files'


def keys(_):
    """Return each envelope record's group key in this worker's tree."""
    out = []
    for name in ENVELOPE:
        r = pools.record(name)
        p, s, c = r.parameter, r.scale, r.curve
        out.append((name, p.kind, p.minimum, p.maximum, s.kind, s.minimum, s.maximum, c.kind,
                    c.multiply, c.offset))
    return out


def sources(_):
    """Return this worker's tree's parameters.py and float32.py as text."""
    return {name: open(pools.TREE[name].__file__, encoding="utf-8").read()
            for name in ("parameters", "float32")}


def without_docstring(node):
    """Drop a module's, class's or function's docstring, in place."""
    body = node.body
    if body and isinstance(body[0], ast.Expr) and isinstance(body[0].value, ast.Constant) \
            and isinstance(body[0].value.value, str):
        del body[0]
    # a string that documents the assignment before it is a docstring too
    node.body = [s for s in body if not (isinstance(s, ast.Expr) and isinstance(s.value, ast.Constant)
                                         and isinstance(s.value.value, str))]


def parse(text, module):
    """Return {qualified name: (node, class name)} of a module's functions and its tables."""
    tree = ast.parse(text)
    without_docstring(tree)
    found = {}
    for node in tree.body:
        if isinstance(node, ast.ClassDef):
            without_docstring(node)
            rest = []
            for item in node.body:
                if isinstance(item, ast.FunctionDef):
                    without_docstring(item)
                    found[f"{node.name}.{item.name}"] = (item, node.name)
                else:
                    rest.append(item)
            found[f"{node.name}, its fields"] = (ast.Module(body=rest, type_ignores=[]), node.name)
        elif isinstance(node, ast.FunctionDef):
            without_docstring(node)
            found[node.name] = (node, None)
        elif isinstance(node, (ast.Assign, ast.AnnAssign)):
            targets = node.targets if isinstance(node, ast.Assign) else [node.target]
            for target in targets:
                if isinstance(target, ast.Name):
                    found[f"{target.id}, a module-level value"] = (node, None)
    return {(f"float32.{name}" if module == "float32" else name): value
            for name, value in found.items()}


def reaches(name, functions, unplaced):
    """Return the names a function's statements can reach, one step."""
    node, owner = functions[name]
    out = set()
    for sub in ast.walk(node):
        if isinstance(sub, ast.Name):
            for candidate in (sub.id, f"float32.{sub.id}", f"{sub.id}, a module-level value",
                              f"float32.{sub.id}, a module-level value"):
                if candidate in functions and candidate != name:
                    out.add(candidate)
        elif isinstance(sub, ast.Attribute):
            methods = [q for q in functions if q.endswith(f".{sub.attr}") and not q.startswith("float32.")]
            if not methods:
                continue
            receiver = ast.unparse(sub.value)
            last = receiver.rsplit(".", 1)[-1]
            cls = owner if receiver == "self" else RECEIVERS.get(last)
            if cls and f"{cls}.{sub.attr}" in functions:
                out.add(f"{cls}.{sub.attr}")
            elif cls is None and last not in OUTSIDE:
                unplaced.add(f"{name}: {receiver}.{sub.attr}")
    return out


def closure(start, functions, unplaced):
    seen, todo = set(), [start]
    while todo:
        name = todo.pop()
        if name in seen:
            continue
        seen.add(name)
        todo.extend(reaches(name, functions, unplaced))
    return seen


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--main", required=True)
    parser.add_argument("--branch", required=True)
    args = parser.parse_args()
    open_pools = pools.open_pools({"main": args.main, "branch": args.branch}, {"main": 1, "branch": 1})

    print("== 1. the 12 envelope delay and hold records, every field the map's methods read:")
    listed = {side: pool.apply(keys, (None,)) for side, pool in open_pools.items()}
    print("   " + ", ".join(FIELDS))
    for row in listed["branch"]:
        print(f"   {row[0]:>12}: {row[1:]!r}")
    distinct = {row[1:] for row in listed["branch"]}
    print(f"== distinct keys among them: {len(distinct)}; main's tree lists the same 12:"
          f" {listed['main'] == listed['branch']}")

    print("== 2. each function's statements, docstrings dropped, main's tree against this branch's")
    texts = {side: pool.apply(sources, (None,)) for side, pool in open_pools.items()}
    parsed = {side: {**parse(texts[side]["parameters"], "parameters"),
                     **parse(texts[side]["float32"], "float32")} for side in texts}
    ours, theirs = parsed["branch"], parsed["main"]
    names = sorted(set(ours) | set(theirs))
    same = {n: n in ours and n in theirs and ast.dump(ours[n][0]) == ast.dump(theirs[n][0])
            for n in names}
    print(f"   {len(names)} functions, class bodies and module-level values in parameters.py and"
          f" float32.py; the same statements in both trees: {sum(same.values())}; different:"
          f" {[n for n in names if not same[n]]}")
    unplaced = set()
    print("== what each of the count's calls can reach, in this branch's tree, and whether any of it"
          " differs from main's:")
    verdict = {}
    for call, start in CALLS.items():
        reached = sorted(closure(start, ours, unplaced))
        changed = [n for n in reached if not same[n]]
        verdict[call] = changed
        print(f"   {call}")
        print(f"      reaches {len(reached)}: {', '.join(reached)}")
        print(f"      of them, with other statements than main's: {changed if changed else 'none'}")
    print(f"== method-named attributes whose receiver was not placed: {sorted(unplaced) or 'none'}")
    can = [call for call, changed in verdict.items() if changed]
    cannot = [call for call, changed in verdict.items() if not changed]
    print("== so, on any record: the calls that can differ from main's:")
    for call in can:
        print(f"      {call}, through {verdict[call]}")
    print("   and the calls that cannot, every function they reach having main's statements:")
    for call in cannot:
        print(f"      {call}")
    for pool in open_pools.values():
        pool.close()
    return 0 if len(distinct) == 1 and not unplaced else 1


if __name__ == "__main__":
    sys.exit(main())

"""Issue #16's spec: two pools of processes, each importing `patchbay_vital` from one tree.

Nothing is reproduced. A worker puts one tree's `src` first on `sys.path` before its first import
of the package, so every method it calls is that tree's shipped one: `main`'s, an extract of its
commit made with `git archive`, or this branch's, the working tree. `open_pools` prints the files
each pool imported and stops unless they are under the tree asked for.
"""

import multiprocessing
import sys

TREE = {}


def use(src):
    """A worker's initializer: import `patchbay_vital` from one tree, before anything else does."""
    sys.path.insert(0, src)
    import patchbay  # noqa: PLC0415
    import patchbay_vital.errors  # noqa: PLC0415
    import patchbay_vital.float32  # noqa: PLC0415
    import patchbay_vital.parameters  # noqa: PLC0415

    TREE["parameters"] = patchbay_vital.parameters
    TREE["float32"] = patchbay_vital.float32
    TREE["errors"] = patchbay_vital.errors
    TREE["id"] = patchbay.ParameterId


def where():
    """Return the files this worker's tree was imported from."""
    return TREE["parameters"].__file__, TREE["float32"].__file__


def record(name):
    """Return this worker's tree's record for an id."""
    return TREE["parameters"].parameter_map().find(TREE["id"](name))


def open_pools(sources, sizes):
    """Open one pool a tree, `sources` and `sizes` keyed by the tree's name; print what each imports."""
    context = multiprocessing.get_context("spawn")
    pools = {
        side: context.Pool(sizes[side], initializer=use, initargs=(sources[side],)) for side in sources
    }
    for side, pool in pools.items():
        files = pool.apply(where)
        print(f"== the {side} pool, {sizes[side]} processes, imports {files[0]}")
        print(f"   and {files[1]}")
        if not all(f.startswith(sources[side]) for f in files):
            print("   NOT the tree asked for: stopping")
            for other in pools.values():
                other.terminate()
            raise SystemExit(2)
    return pools
