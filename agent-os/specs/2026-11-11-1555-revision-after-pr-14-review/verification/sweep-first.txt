"""The revision after PR 14's review: every float32 normalized value through the property's paths.

Counted for every scale/curve group of the committed map whose curve this arithmetic reproduces
bit for bit (linear, quadratic, quartic, square_root), and for each of the six records with a
curve offset directly. The exponential groups are listed and not counted: their skew is the
platform's pow or log2, which numpy's own functions do not reproduce.

P0 = scale.to_preset(r), the preset the parameter holds; D0 = to_display(P0); P1 = to_preset(D0);
N = scale.to_normalized; P2 = scale.to_preset(N(P1)), what a set of D0 holds.
  A1   to_display(P1) != D0      _assert_comes_back: the display back through the preset
  HELD S(N(P0)) != P0            the scale's inverse on the preset the parameter holds
  A2   S(N(P1)) != P1            the display's preset back through the normalized path
  A3   P1 != P0                  the preset back; the property asserts it above the floor only
  SET  to_display(P2) - D0       in double steps, value order: 0, +1, -1 or other
The floor is `_ABSORBED`'s, preset 2 ** -43, for volume only; a record without one asserts A3
everywhere. `bound!` counts what breaks the property as this revision states it: A1 or HELD
anywhere, A3 or a nonzero SET above the floor, a SET of more than one step at or below it.

The arithmetic is the map's, exactly: r * span is exact in a double (24 x 24 bits); a sum is
rounded once to float32 with TwoSum's remainder deciding a double that lands on a float32
midpoint; a quotient's nearest float32 is chosen from three candidates by exact distance, ties to
the even significand. A value whose subtraction this cannot certify exact is given to the shipped
scalar `NormalizedScale.to_normalized` instead, and counted (`scalar`). Before counting, each
group is checked against the shipped scalar methods on 200,007 values, every one of P0, D0, P1,
P2 and S(N(P0)); a group with a mismatch is not counted.
"""

import math
import multiprocessing
import time

import numpy as np
from patchbay import ParameterId

from patchbay_vital.parameters import parameter_map

FLOOR = {"volume": 2.0**-43}
REPRODUCED = ("linear", "quadratic", "quartic", "square_root")
ROWS = (5.857149595214349e-20, 3.4694490199051456e-18, 1.439742266862027e-17, 1.53642479121845e-17)
CHUNK = 1 << 22
SIGN = np.int64(0x7FFFFFFFFFFFFFFF)


def two_sum(a, b):
    s = a + b
    bp = s - a
    ap = s - bp
    return s, (a - ap) + (b - bp)


def round32(s, err):
    """Return the float32 nearest s + err exactly, s a double and err its exact remainder."""
    c32 = s.astype(np.float32)
    c = c32.astype(np.float64)
    diff = s - c
    toward = np.where(diff > 0, np.float32(np.inf), np.float32(-np.inf))
    nb = np.nextafter(c32, toward).astype(np.float64)
    away = (diff != 0) & (2 * diff == nb - c) & (err != 0) & (np.sign(err) == np.sign(diff))
    return np.where(away, nb, c)


def half_away(v):
    return np.copysign(np.floor(np.abs(v) + 0.5), v)


def ordered(x):
    """Return each double's place in value order as an int64, both zeros at 0."""
    bits = x.view(np.int64)
    return np.where(bits < 0, -(bits & SIGN), bits)


class Vec:
    def __init__(self, rec):
        s, c = rec.scale, rec.curve
        self.rec = rec
        self.mn, self.mx, self.span, self.ix = s.minimum, s.maximum, s.span, s.kind == "indexed"
        self.kind, self.mul, self.off = c.kind, c.multiply, c.offset
        assert self.kind in REPRODUCED
        self.scalar = 0

    def S(self, r):
        s, err = two_sum(r * self.span, np.float64(self.mn))
        v = round32(s, err)
        return half_away(v) if self.ix else v

    def N(self, p):
        a, ea = two_sum(p, np.float64(-self.mn))
        q = (a / self.span).astype(np.float32)
        best = None
        unsure = ea != 0
        for c32 in (np.nextafter(q, np.float32(-np.inf)), q, np.nextafter(q, np.float32(np.inf))):
            c = c32.astype(np.float64)
            d, ed = two_sum(c * self.span, -a)
            unsure |= ed != 0
            d = np.abs(d)
            even = (c32.view(np.uint32) & 1) == 0
            if best is None:
                best, bd, be = c, d, even
            else:
                take = (d < bd) | ((d == bd) & even & ~be)
                best, bd, be = np.where(take, c, best), np.where(take, d, bd), np.where(take, even, be)
        if unsure.any():
            index = np.flatnonzero(unsure)
            self.scalar += index.size
            best = best.copy()
            best[index] = [self.rec.scale.to_normalized(float(x)) for x in p[index]]
        return best

    def skew(self, p):
        if self.kind == "linear":
            return p
        if self.kind == "quadratic":
            return p * p
        if self.kind == "quartic":
            s = p * p
            return s * s
        return np.sqrt(np.maximum(p, 0.0))

    def unskew(self, y):
        if self.kind == "linear":
            return y
        if self.kind == "quadratic":
            return np.sqrt(np.maximum(y, 0.0))
        if self.kind == "quartic":
            return np.sqrt(np.maximum(np.sqrt(np.maximum(y, 0.0)), 0.0))
        return y * y

    def disp(self, p):
        return self.mul * self.skew(p) + self.off

    def inv(self, d):
        v = self.unskew((d - self.off) / self.mul)
        p = np.minimum(np.maximum(v.astype(np.float32).astype(np.float64), self.mn), self.mx)
        return half_away(p) if self.ix else p

    def paths(self, r):
        p0 = self.S(r)
        d0 = self.disp(p0)
        p1 = self.inv(d0)
        p2 = self.S(self.N(p1))
        held = self.S(self.N(p0))
        return p0, d0, p1, p2, held


def scalar_paths(rec, x):
    d = rec.from_normalized(x)
    return (
        rec.scale.to_preset(x),
        d,
        rec.to_preset(d),
        rec.scale.to_preset(rec.to_normalized(d)),
        rec.scale.to_preset(rec.scale.to_normalized(rec.scale.to_preset(x))),
    )


def check(name):
    """Return mismatches and scalar fallbacks of the vectorised paths against the scalar methods."""
    rec = parameter_map().find(ParameterId(name))
    v = Vec(rec)
    rng = np.random.default_rng(14)
    bits = rng.integers(0, 0x3F800001, 200_000, dtype=np.uint32)
    named = np.array([*ROWS, 0.0, 1.0, 2.0**-149], dtype=np.float32)
    sample = np.concatenate([bits.view(np.float32), named]).astype(np.float64)
    vector = v.paths(sample)
    bad = 0
    for i in range(sample.size):
        try:
            scalar = scalar_paths(rec, float(sample[i]))
        except Exception:  # noqa: BLE001 - a raise is a mismatch, counted
            bad += 1
            continue
        bad += scalar != tuple(float(column[i]) for column in vector)
    return sample.size, bad, v.scalar


SPANS = [("zero", 0, 1), ("sub", 1, 1 << 23)]
SPANS += [(str(e - 127), e << 23, (e + 1) << 23) for e in range(1, 127)]
SPANS += [("one", 0x3F800000, 0x3F800001)]
COLUMNS = ("below", "A1", "HELD", "A2", "A3", "SET0", "SET+1", "SET-1", "SET?", "above!", "bound!")


def count(name):
    """Count every float32 normalized value through one record's paths, by binade."""
    started = time.time()
    checked, mismatches, check_scalar = check(name)
    result = {"name": name, "checked": checked, "mismatches": mismatches,
              "check_scalar": check_scalar, "rows": [], "quiet": []}
    if mismatches:
        return result
    rec = parameter_map().find(ParameterId(name))
    v = Vec(rec)
    floor = FLOOR.get(name, -math.inf)
    total = np.zeros(len(COLUMNS) + 1, dtype=np.int64)
    for label, lo, hi in SPANS:
        row = np.zeros(len(COLUMNS) + 1, dtype=np.int64)
        for start in range(lo, hi, CHUNK):
            r = np.arange(start, min(hi, start + CHUNK), dtype=np.uint32)
            r = r.view(np.float32).astype(np.float64)
            p0, d0, p1, p2, held = v.paths(r)
            below = p0 <= floor
            a1, hd, a2, a3 = v.disp(p1) != d0, held != p0, p2 != p1, p1 != p0
            steps = ordered(v.disp(p2)) - ordered(d0)
            s0, sp, sm = steps == 0, steps == 1, steps == -1
            far = ~(s0 | sp | sm)
            above = ~below & (a1 | hd | a2 | a3 | ~s0)
            bound = a1 | hd | (~below & (a3 | ~s0)) | (below & far)
            flags = (below, a1, hd, a2, a3, s0, sp, sm, far, above, bound)
            row += [r.size, *(int(np.count_nonzero(f)) for f in flags)]
        total += row
        # a binade is printed where anything but SET0 is counted
        if row[2:].any() and (row[1:6].any() or row[7:].any()):
            result["rows"].append((label, row.tolist()))
        else:
            result["quiet"].append(label)
    result["total"] = total.tolist()
    result["scalar"] = v.scalar
    result["seconds"] = time.time() - started
    return result


def group_key(rec):
    p, s, c = rec.parameter, rec.scale, rec.curve
    return (p.kind, s.kind, s.minimum, s.maximum, c.kind, c.multiply, c.offset, p.minimum, p.maximum)


def main():
    m = parameter_map()
    groups = {}
    for rec in m.parameters:
        groups.setdefault(group_key(rec), []).append(rec.id)
    counted = [(key, ids) for key, ids in groups.items() if key[4] in REPRODUCED]
    skipped = [(key, ids) for key, ids in groups.items() if key[4] not in REPRODUCED]
    offset = [rec.id for rec in m.parameters if rec.curve.offset != 0]
    print(f"== the committed map: {len(m.parameters)} records in {len(groups)} groups, keyed by")
    print("   (parameter kind, scale kind, scale min, scale max, curve kind, multiply, offset,")
    print("    parameter min, parameter max): every method the paths call reads these alone")
    print(f"-- counted, {len(counted)} groups ({', '.join(REPRODUCED)}), each by its first id:")
    for key, ids in counted:
        more = f" and {len(ids) - 1} more" if len(ids) > 1 else ""
        print(f"  {ids[0]}{more}: {key}")
    print(f"-- not counted, {len(skipped)} groups: the skew is the platform's pow or log2, which")
    print("   numpy's exp2 and log2 do not reproduce bit for bit; the property's exponent-spread")
    print("   draws and pinned examples hold them")
    for key, ids in skipped:
        more = f" and {len(ids) - 1} more" if len(ids) > 1 else ""
        print(f"  {ids[0]}{more}: {key}")
    print(f"-- the six records with a curve offset, each counted directly: {offset}")
    names = [ids[0] for _, ids in counted]
    names += [name for name in offset if name not in names]

    started = time.time()
    with multiprocessing.Pool(10) as pool:
        results = {r["name"]: r for r in pool.imap_unordered(count, names)}
    print(f"\n== {len(names)} counts in {time.time() - started:.0f} s on 10 processes")

    print("\n== the check against the shipped scalar methods, per count, before counting")
    print("   (200,000 bit patterns uniform over [0, 1], the four rows of tests/data/volume-floor.txt,")
    print("    0, 1 and 2 ** -149; P0, D0, P1, P2 and S(N(P0)) each compared)")
    for name in names:
        r = results[name]
        print(f"  {name}: {r['checked']} values, mismatches {r['mismatches']},"
              f" scalar fallbacks {r['check_scalar']}")

    print("\n== every float32 normalized value, 1,065,353,217 a count, by its exponent")
    print("  binade: the float32 exponent of r ('sub' subnormal); below: P0 at or below the floor;")
    print("  then each count; above!: any failure above the floor; bound!: the property broken")
    head = f"{'binade':>6} {'n':>10} " + " ".join(f"{c:>9}" for c in COLUMNS)
    grand = np.zeros(len(COLUMNS) + 1, dtype=np.int64)
    for name in names:
        r = results[name]
        rec = m.find(ParameterId(name))
        print(f"\n-- {name}: scale {rec.scale.kind} [{rec.scale.minimum!r}, {rec.scale.maximum!r}],"
              f" curve {rec.curve.kind} x{rec.curve.multiply!r} {rec.curve.offset:+},"
              f" parameter [{rec.parameter.minimum!r}, {rec.parameter.maximum!r}]")
        if r["mismatches"]:
            print("  NOT COUNTED: the check found a mismatch")
            continue
        print(head)
        for label, row in r["rows"]:
            print(f"{label:>6} {row[0]:>10} " + " ".join(f"{x:>9}" for x in row[1:]))
        t = r["total"]
        print(f"{'total':>6} {t[0]:>10} " + " ".join(f"{x:>9}" for x in t[1:]))
        grand += t
        quiet = r["quiet"]
        print(f"  only SET0 in the {len(quiet)} binades not printed; scalar fallbacks {r['scalar']};"
              f" {r['seconds']:.0f} s")

    print("\n== summary over every count")
    for column, value in zip(("n", *COLUMNS), grand, strict=True):
        print(f"  {column}: {value}")

    volume = m.find(ParameterId("volume"))
    print("\n== volume's floor and the rows of tests/data/volume-floor.txt")
    print(f"  N(2 ** -43) = {volume.scale.to_normalized(2.0**-43)!r}")
    for r in ROWS:
        d0 = volume.from_normalized(r)
        p1 = volume.to_preset(d0)
        p2 = volume.scale.to_preset(volume.to_normalized(d0))
        shown = volume.to_display(p2)
        step = 0 if shown == d0 else (1 if shown == math.nextafter(d0, math.inf) else
                                      (-1 if shown == math.nextafter(d0, -math.inf) else None))
        held = volume.scale.to_preset(volume.scale.to_normalized(p1)) == p1
        print(f"  r = {r!r} ({np.float32(r).view(np.uint32):08x}): P0 {volume.scale.to_preset(r)!r},"
              f" D0 {d0!r}, P1 {p1!r} (held {held}), P2 {p2!r}, set's display step {step:+d}")


if __name__ == "__main__":
    main()
