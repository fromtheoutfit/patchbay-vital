```python
"""The revision after the review of patchbay-vital's PR 19: arbitrary displays, `main`'s methods against this branch's, called directly.

An arbitrary display is a double. The doubles within a parameter's bounds cannot be enumerated,
so most of what follows is a SAMPLE, each part labelled with its size, generator and seed. One
part is a COUNT: every double from -80 to `volume`'s display at its floor's preset, 2 ** -43.

Nothing is reproduced. Two pools each import `patchbay_vital` from one tree (`pools.py`) and call
that tree's shipped scalar methods on each display D of a group's first record, as a caller would:

  TP = record.to_preset(D)        the display's preset
  TN = record.to_normalized(D)    the normalized value a set of the display sends

and each says of its own TP:

  holds   its own tree's scale.holds(TP)
  set     0 <= TN <= 1 and scale.to_preset(TN) == TP: a normalized value the host can set lands
          on TP, through the forward scale, so the scale holds TP and a set of D holds it
  shows   record.to_display(TP) == D

A display past the parameter's bounds is refused (ParameterRangeError); a refusal is compared too.
This process compares the two trees on every D and counts, by scale and curve group:

  displays   those called and not refused
  refused    those both trees refuse
  changed    TP differs from `main`'s
  was unheld of the changed, `main`'s TP is one `main`'s scale does not hold
  now held   of the changed, this branch's TP is `set`
  norm same  of the changed, TN is `main`'s
  next       of the changed, this branch's TP is the float32 next to `main`'s
  OUT        what stops the run, at once: a changed display whose `main` TP was held; a TP of
             this branch that is not `set`, changed or not; a TN that differs from `main`'s; a
             display one tree refuses and the other does not

The parts:

  A  SAMPLE  200,000 displays a continuous group, uniform over the parameter's bounds, drawn by
             numpy.random.default_rng(16).uniform, one generator across the groups in map order
  B  SAMPLE  `volume`, 20,000 displays uniform over its bounds, twice: random.Random(8).uniform
             (the sample issue #16 cites, 1,085 changed) and numpy.random.default_rng(8).uniform
             (the fifth review's, 1,108 on the third branch)
  C  SAMPLE  the double just below and just above a held display, on every continuous group. The
             held displays are those of float32 normalized values taken by bit pattern so every
             exponent is reached: 16 patterns among the subnormals and 16 in each binade from -126
             to -1, drawn by numpy.random.default_rng(14).integers, with 0.0 and 1.0: 2,034 values
             a group, 4,068 displays. Each tree makes the held display with its own
             from_normalized, compared as D0!=. `held` is how many distinct held displays the

COLUMNS = ("displays", "refused", "changed", "was unheld", "now held", "norm same", "next", "OUT")
_DOUBLE = struct.Struct("<d")
_BITS = struct.Struct("<q")


def listing(_):
    """Return each continuous group's first record and size, and `volume`'s two floor displays."""
    groups = {}
    for r in pools.TREE["parameters"].parameter_map().parameters:
        p, s, c = r.parameter, r.scale, r.curve
        key = (p.kind, p.minimum, p.maximum, s.kind, s.minimum, s.maximum, c.kind, c.multiply,
               c.offset)
        groups.setdefault(key, []).append(r.id)
    volume = pools.record("volume")
    ends = (volume.to_display(0.0), volume.to_display(2.0**-43))
    return [(ids[0], key, len(ids)) for key, ids in groups.items()], ends


def call(rec, refused, display):
    """Return one tree's status, TP, TN, holds, set and shows for a display."""
    try:
        tp = rec.to_preset(display)
        tn = rec.to_normalized(display)
    except refused:
        return 1, math.nan, math.nan, False, False, False
    scale = rec.scale
    landed = 0.0 <= tn <= 1.0 and scale.to_preset(tn) == tp
    return 0, tp, tn, scale.holds(tp), landed, rec.to_display(tp) == display


def pack(rows):
    """Return a list of `call`'s results as arrays."""
    status, tp, tn, holds, landed, shows = zip(*rows, strict=True)
    return (np.array(status, np.int8), np.array(tp), np.array(tn), np.array(holds, bool),
            np.array(landed, bool), np.array(shows, bool))


def work_displays(job):
    """Call one tree's methods on a list of displays of a record."""
    name, displays = job
    rec = pools.record(name)
    refused = pools.TREE["errors"].ParameterRangeError
    return pack([call(rec, refused, d) for d in displays])


def work_beside(job):
    """Call one tree's methods on the double below and above each normalized value's display."""
    name, values = job
    rec = pools.record(name)
    refused = pools.TREE["errors"].ParameterRangeError
    held = [rec.from_normalized(r) for r in values]
    below = pack([call(rec, refused, math.nextafter(d, -math.inf)) for d in held])
    above = pack([call(rec, refused, math.nextafter(d, math.inf)) for d in held])
    return np.array(held), below, above


def double_at(first_bits, offset):
    """Return the double `offset` doubles above a negative double given by its bits."""
    return _DOUBLE.unpack(_BITS.pack(first_bits - offset))[0]


def work_floor(job):
    """Call one tree's methods on a run of consecutive doubles from `volume`'s lowest display up."""
    first_bits, start, count = job
    rec = pools.record("volume")
    refused = pools.TREE["errors"].ParameterRangeError
    displays = [double_at(first_bits, start + i) for i in range(count)]
    packed = pack([call(rec, refused, d) for d in displays])
    # how many doubles this tree's preset's display is from the display asked
    off = np.array([
        abs(_BITS.unpack(_DOUBLE.pack(rec.to_display(tp)))[0] - (first_bits - start - i))
        if status == 0 else -1
        for i, (status, tp) in enumerate(zip(packed[0].tolist(), packed[1].tolist(), strict=True))
    ])
    return (*packed, off)


def steps(a, b):
    """Return how many float32s apart two arrays of float32 values are, -1 across a zero."""
    x = a.astype(np.float32).view(np.int32).astype(np.int64)
    y = b.astype(np.float32).view(np.int32).astype(np.int64)
    apart = np.abs(x - y)
    apart[(x < 0) != (y < 0)] = -1
    return apart


def tally(ours, theirs):
    """Return a row of COLUMNS for one batch, and where it stops the run."""
    o_status, o_tp, o_tn, _, o_set, _ = ours[:6]
    t_status, t_tp, t_tn, t_holds, _, _ = theirs[:6]
    called = (o_status == 0) & (t_status == 0)
    refused = (o_status == 1) & (t_status == 1)
    changed = called & (o_tp != t_tp)
    out = (o_status != t_status) | (changed & t_holds) | (called & ~o_set) | (called & (o_tn != t_tn))
    row = [
        np.count_nonzero(called), np.count_nonzero(refused), np.count_nonzero(changed),
        np.count_nonzero(changed & ~t_holds), np.count_nonzero(changed & o_set),
        np.count_nonzero(changed & (o_tn == t_tn)),
        np.count_nonzero(changed & (steps(np.nan_to_num(o_tp), np.nan_to_num(t_tp)) == 1)),
        np.count_nonzero(out),
    ]
    return np.array(row, dtype=np.int64), np.flatnonzero(out)


def line(label, key, row):
    """Return one printed row: a group, or a total."""
    head = f"{label:>32} {key[3]:>8} {key[6]:>20}" if key else f"{label:>32} {'':>8} {'':>20}"

def stop(pools_, where, ours, theirs, indices, displays):

        for start in range(0, len(displays), CHUNK):
            part = displays[start : start + CHUNK]
            jobs.append((name, key, part, pools_["main"].apply_async(work_displays, ((name, part),)),
                         pools_["branch"].apply_async(work_displays, ((name, part),))))
    rows, order = {}, []
    for name, key, part, theirs_job, ours_job in jobs:
        theirs, ours = theirs_job.get(), ours_job.get()
        row, outs = tally(ours, theirs)
        if (name, key) not in rows:
            rows[(name, key)] = np.zeros(len(COLUMNS), dtype=np.int64)
            order.append((name, key))
        rows[(name, key)] += row
        if outs.size:
            return stop(pools_, name, ours, theirs, outs, part)
    total = np.zeros(len(COLUMNS), dtype=np.int64)
    for name, key in order:
        total += rows[(name, key)]
        print(line(name, key, rows[(name, key)]))
    if len(order) > 1:
        print(line("total", None, total))
        some = sum(1 for item in order if rows[item][2])
        print(f"   groups with a changed display: {some} of {len(order)}; without: {len(order) - some}")
    sys.stdout.flush()
    return 0


def run_beside(pools_, groups):
    """Part C: the double below and above held displays, on every continuous group."""
    rng = np.random.default_rng(14)
    print("== C. SAMPLE: the double below and above a held display, every continuous group;"
          " 16 float32 bit patterns among the subnormals and in each binade from -126 to -1,"
          " numpy.random.default_rng(14).integers, with 0.0 and 1.0: 2,034 values a group")
    print(header() + f" {'held':>10} {'D0!=':>10}")
    jobs = []
    for name, key, _ in groups:
        bits = [rng.integers(1, 1 << 23, 16, dtype=np.uint32)]
        bits += [rng.integers(e << 23, (e + 1) << 23, 16, dtype=np.uint32) for e in range(1, 127)]
        bits.append(np.array([0, 0x3F800000], dtype=np.uint32))
        values = np.concatenate(bits).view(np.float32).astype(np.float64).tolist()
        jobs.append((name, key, values, pools_["main"].apply_async(work_beside, ((name, values),)),
                     pools_["branch"].apply_async(work_beside, ((name, values),))))
    total = np.zeros(len(COLUMNS) + 2, dtype=np.int64)
    changed_groups = 0
    for name, key, values, theirs_job, ours_job in jobs:
        their_held, their_below, their_above = theirs_job.get()
        our_held, our_below, our_above = ours_job.get()
        row = np.zeros(len(COLUMNS) + 2, dtype=np.int64)
        row[-2] = np.unique(our_held).size
        row[-1] = np.count_nonzero(our_held != their_held)
        for ours, theirs, side in ((our_below, their_below, -math.inf), (our_above, their_above, math.inf)):
            counted, outs = tally(ours, theirs)
            row[:-2] += counted
            if outs.size:
                displays = [math.nextafter(d, side) for d in our_held.tolist()]
                return stop(pools_, name, ours, theirs, outs, displays)
        if row[-1]:
            print(line(name, key, row))
            print("== stopped at once: the two trees' held displays differ")
            return 1
        total += row
        changed_groups += bool(row[2])
        print(line(name, key, row))
    print(line("total", None, total))
    print(f"   groups with a changed display: {changed_groups} of {len(jobs)}")
    print("   A display past a bound is refused by both trees: the double below the lowest display"
          " a parameter shows, or above the highest. No group is left out, curved or indexed.")
    sys.stdout.flush()
    return 0


def run_floor(pools_, ends):
    """Part D: every double of `volume` from its display at preset 0 to its display at 2 ** -43."""
    low, high = ends
    first_bits = _BITS.unpack(_DOUBLE.pack(low))[0]
    count = first_bits - _BITS.unpack(_DOUBLE.pack(high))[0] + 1
    print(f"== D. COUNT: volume, every double from {low!r}, its display at preset 0, to {high!r},"
          f" its display at preset 2 ** -43: {count:,} doubles, each through both trees")
    print(header())
    jobs = []
    for start in range(0, count, FLOOR_CHUNK):
        job = (first_bits, start, min(FLOOR_CHUNK, count - start))
        jobs.append((job, pools_["main"].apply_async(work_floor, (job,)),
                     pools_["branch"].apply_async(work_floor, (job,))))
    total = np.zeros(len(COLUMNS), dtype=np.int64)
    shows = np.zeros(2, dtype=np.int64)
    away = {}
    for job, theirs_job, ours_job in jobs:
        theirs, ours = theirs_job.get(), ours_job.get()
        row, outs = tally(ours, theirs)
        total += row
        if outs.size:
            displays = [double_at(first_bits, job[1] + i) for i in range(job[2])]
            return stop(pools_, "volume's floor", ours, theirs, outs, displays)
        shows += (np.count_nonzero(ours[5]), np.count_nonzero(theirs[5]))
        changed = ours[1] != theirs[1]
        for distance, number in zip(*np.unique(ours[6][changed], return_counts=True), strict=True):
            away[int(distance)] = away.get(int(distance), 0) + int(number)
    key = ("continuous", 0, 0, "linear", 0, 0, "square_root")
    print(line("volume", key, total))
    print(f"   the preset shows the display asked, exactly: this branch's on {int(shows[0]):,},"
          f" main's on {int(shows[1]):,}")
    print("   of the changed, how many doubles this branch's preset's display is from the display"
          f" asked: {dict(sorted(away.items()))}")
    sys.stdout.flush()
    return 0
```

```python
"""The revision after the review of patchbay-vital's PR 19: two pools of processes, each importing `patchbay_vital` from one tree.

Nothing is reproduced. A worker puts one tree's `src` first on `sys.path` before its first import
of the package, so every method it calls is that tree's shipped one: `main`'s, an extract of its
commit made with `git archive`, or this branch's, the working tree. `open_pools` prints the files
each pool imported and stops unless they are under the tree asked for.
"""

import multiprocessing
import sys

TREE = {}


def use(src):
    """A worker's initializer: import `patchbay_vital` from one tree, before anything else does."""
    sys.path.insert(0, src)
    import patchbay  # noqa: PLC0415
    import patchbay_vital.errors  # noqa: PLC0415
    import patchbay_vital.float32  # noqa: PLC0415
    import patchbay_vital.parameters  # noqa: PLC0415

    TREE["parameters"] = patchbay_vital.parameters
    TREE["float32"] = patchbay_vital.float32
    TREE["errors"] = patchbay_vital.errors
    TREE["id"] = patchbay.ParameterId


def where():
    """Return the files this worker's tree was imported from."""
    return TREE["parameters"].__file__, TREE["float32"].__file__


def record(name):
    """Return this worker's tree's record for an id."""
    return TREE["parameters"].parameter_map().find(TREE["id"](name))


def open_pools(sources, sizes):
    """Open one pool a tree, `sources` and `sizes` keyed by the tree's name; print what each imports."""
    context = multiprocessing.get_context("spawn")
    pools = {
        side: context.Pool(sizes[side], initializer=use, initargs=(sources[side],)) for side in sources
    }
    for side, pool in pools.items():
        files = pool.apply(where)
        print(f"== the {side} pool, {sizes[side]} processes, imports {files[0]}")
        print(f"   and {files[1]}")
        if not all(f.startswith(sources[side]) for f in files):
            print("   NOT the tree asked for: stopping")
            for other in pools.values():
                other.terminate()
            raise SystemExit(2)
    return pools
```
