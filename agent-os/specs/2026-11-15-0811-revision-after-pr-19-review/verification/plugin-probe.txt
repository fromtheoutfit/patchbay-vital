```python
"""The revision after the review of patchbay-vital's PR 19: what Vital holds after a set of an arbitrary display through the map.

A SAMPLE, on the installed Vital: 200 displays a group for every scale and curve group of the map
on a linear scale, uniform over the parameter's bounds, numpy.random.default_rng(16).uniform, one
generator across the groups in map order; and, beside it, the pinned `tempo_displays` of
`tests/data/pinned-values.txt` on `beats_per_minute`.

Each display D is set as the integration module sets one, with its own `_held_after_a_set`,
imported from it, after a load of the plugin's initial state, the sequence its tests run:

  V  = the state's float after host.set(id, record.to_normalized(D))     what Vital holds
  TP = record.to_preset(D)                                              this branch's map
  G  = record.from_normalized(host.get(id))                             what `get` shows

`main`'s `to_preset(D)` and whether `main`'s scale holds it come from `main`'s own tree, imported
by one pool of processes (`pools.py`) and called directly. Counted, by group:

  sets        the displays set
  V == TP     Vital holds this branch's preset for the display, exactly
  changed     this branch's TP differs from `main`'s
  was unheld  of those, `main`'s TP is one `main`'s scale does not hold
  V == main   of the changed, Vital holds `main`'s value (expected on none)
  G == shown  `get` shows the display of the preset held, record.to_display(V)

The run stops at the first display where V is not TP.

Run from the repository root:
  uv run --locked python plugin_probe.py --main <main's src>
"""

import argparse
import sys
import time

import numpy as np

import pools

SETS = 200


def main_values(job):
    """Return `main`'s to_preset for each display of a record, and whether its scale holds it."""
    name, displays = job
    rec = pools.record(name)
    presets = [rec.to_preset(d) for d in displays]
    return presets, [rec.scale.holds(p) for p in presets]


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--main", required=True)
    args = parser.parse_args()
    started = time.time()
    open_pools = pools.open_pools({"main": args.main}, {"main": 2})

    sys.path.insert(0, "tests/integration")
    import types  # noqa: PLC0415
    from pathlib import Path  # noqa: PLC0415

    import test_vital_plugin as module  # noqa: PLC0415
    from patchbay import ParameterId  # noqa: PLC0415
    from patchbay_vital import parameters  # noqa: PLC0415
    from patchbay_vital.host import DEFAULT_PATH, VitalHost  # noqa: PLC0415

    print(f"== this process imports {parameters.__file__} and {module.__file__}")
    host = VitalHost()
    initial = host.state()
    the_map = parameters.parameter_map()
    print(f"== Vital at {DEFAULT_PATH}: {host.name} {host.version}; the map is of"
          f" {the_map.instrument} {the_map.vital_version}, {the_map.host}, {the_map.machine}")

    groups = {}
    for r in the_map.parameters:
        p, s, c = r.parameter, r.scale, r.curve
        key = (p.kind, p.minimum, p.maximum, s.kind, s.minimum, s.maximum, c.kind, c.multiply,
               c.offset)
        groups.setdefault(key, r)
    linear = [r for key, r in groups.items() if key[0] == "continuous" and key[3] == "linear"]
    rng = np.random.default_rng(16)
    batches = [(r, rng.uniform(r.parameter.minimum, r.parameter.maximum, SETS).tolist())
               for r in linear]
    pinned = module._pinned(types.SimpleNamespace(rootpath=Path.cwd()))["tempo_displays"]
    tempo = the_map.find(ParameterId("beats_per_minute"))
    batches.append((tempo, list(pinned)))

    print(f"== SAMPLE: {SETS} displays a group on the {len(linear)} continuous groups of a linear"
          f" scale, numpy.random.default_rng(16).uniform: {SETS * len(linear):,} sets, each after a"
          " load of the initial state; then the pinned tempo_displays")
    columns = ("sets", "V == TP", "changed", "was unheld", "V == main", "G == shown")
    print(f"{'group, by its first record':>32} {'curve':>20} " + " ".join(f"{c:>10}" for c in columns))
    total = np.zeros(len(columns), dtype=np.int64)
    for index, (record, displays) in enumerate(batches):
        theirs, their_holds = open_pools["main"].apply(main_values, ((record.id, displays),))
        row = np.zeros(len(columns), dtype=np.int64)
        lines = []
        for display, main_preset, main_held in zip(displays, theirs, their_holds, strict=True):
            host.load_state(initial)
            held = module._held_after_a_set(host, record, display)
            preset = record.to_preset(display)
            shown = record.from_normalized(host.get(record.id))
            changed = preset != main_preset
            row += (1, held == preset, changed, changed and not main_held,
                    changed and held == main_preset, shown == record.to_display(held))
            lines.append(f"   display {display!r}: main's to_preset {main_preset!r}"
                         f" ({'held' if main_held else 'unheld'}), this branch's {preset!r},"
                         f" Vital holds {held!r}, shown as {shown!r}")
            if held != preset:
                print(lines[-1])
                print("== stopped: Vital holds another value than the map's")
                return 1
        label = record.id if index < len(linear) else f"{record.id}, the pinned"
        print(f"{label:>32} {record.curve.kind:>20} " + " ".join(f"{int(v):>10}" for v in row))
        if index >= len(linear):
            print("\n".join(lines))
        else:
            total += row
        sys.stdout.flush()
    print(f"{'total of the sample':>32} {'':>20} " + " ".join(f"{int(v):>10}" for v in total))
    print(f"  {time.time() - started:.0f} s")
    for pool in open_pools.values():
        pool.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""The revision after the review of patchbay-vital's PR 19: two pools of processes, each importing `patchbay_vital` from one tree.

Nothing is reproduced. A worker puts one tree's `src` first on `sys.path` before its first import
of the package, so every method it calls is that tree's shipped one: `main`'s, an extract of its
commit made with `git archive`, or this branch's, the working tree. `open_pools` prints the files
each pool imported and stops unless they are under the tree asked for.
"""

import multiprocessing
import sys

TREE = {}


def use(src):
    """A worker's initializer: import `patchbay_vital` from one tree, before anything else does."""
    sys.path.insert(0, src)
    import patchbay  # noqa: PLC0415
    import patchbay_vital.errors  # noqa: PLC0415
    import patchbay_vital.float32  # noqa: PLC0415
    import patchbay_vital.parameters  # noqa: PLC0415

    TREE["parameters"] = patchbay_vital.parameters
    TREE["float32"] = patchbay_vital.float32
    TREE["errors"] = patchbay_vital.errors
    TREE["id"] = patchbay.ParameterId


def where():
    """Return the files this worker's tree was imported from."""
    return TREE["parameters"].__file__, TREE["float32"].__file__


def record(name):
    """Return this worker's tree's record for an id."""
    return TREE["parameters"].parameter_map().find(TREE["id"](name))


def open_pools(sources, sizes):
    """Open one pool a tree, `sources` and `sizes` keyed by the tree's name; print what each imports."""
    context = multiprocessing.get_context("spawn")
    pools = {
        side: context.Pool(sizes[side], initializer=use, initargs=(sources[side],)) for side in sources
    }
```
