    """Return one record's scale: kind, bounds, span, and what 0 and 1 land on."""

    return (scale.kind, scale.minimum, scale.maximum, scale.span, scale.to_preset(0.0),

def rises(job):
    """Say whether a scale's forward value never falls over a sorted sample of normalized values."""
    name, values = job
    scale = pools.record(name).scale
    landed = [scale.to_preset(value) for value in values]
    return all(a <= b for a, b in zip(landed, landed[1:], strict=False))


def nearest_quotient(scale, preset):
    """Return the float32 nearest (preset - minimum) / span, from the exact quotient."""
    step = pools.TREE["float32"].round_to_float32
    return step((Fraction(preset) - Fraction(scale.minimum)) / Fraction(scale.span))


def what_a_set_holds(scale, preset):
    """Return what a set of a preset holds, from the forward scale alone."""
    step = pools.TREE["float32"].next_float32
    nearest = nearest_quotient(scale, preset)
    beside = (nearest, step(nearest, -math.inf), step(nearest, math.inf))
    for value in beside:
        if 0.0 <= value <= 1.0 and scale.to_preset(value) == preset:
            return preset
    return scale.to_preset(nearest)


def around(scale, presets):
    """Return each preset with the float32 below and above it, and the scale's ends with theirs."""
    step = pools.TREE["float32"].next_float32
    out = set()
    ends = (scale.minimum, scale.maximum, scale.to_preset(0.0), scale.to_preset(1.0))
    for preset in (*presets, *ends):
        out.update((preset, step(preset, -math.inf), step(preset, math.inf)))
    return out


def pinned_presets(job):
    """Return the presets section 2 asks of one record, worked out in this worker's tree."""
    name, normalized, displays = job
    record = pools.record(name)
    scale, curve = record.scale, record.curve
    low, high = record.parameter.minimum, record.parameter.maximum
    found = set()

    asked += [(None, d) for d in displays if low <= d <= high]
    for r, display in asked:
        own = min(max(curve.to_preset(display), scale.minimum), scale.maximum)
        if scale.kind == "indexed":

        presets = [own, what_a_set_holds(scale, own)]
        if r is not None:

        found |= around(scale, presets)
    return sorted(found)

    found = {float(k) for k in range(low - 1, high + 2)}
    found |= {k + 0.5 for k in range(low, high)}
    found |= {step(scale.minimum, -math.inf), step(scale.maximum, math.inf)}
    return sorted(found)


def ask(job):
    """Ask this worker's tree's `holds` of each preset: True, False, or what it raised."""
    name, presets = job
    scale = pools.record(name).scale
    out = []
    for preset in presets:
        try:
            out.append(scale.holds(preset))
        except Exception as error:  # noqa: BLE001 - the name of whatever a tree raises is the answer
            out.append(type(error).__name__)
    return out


def read_pinned():
    """Return the file's normalized values and its displays."""
    normalized, displays = [], []
    for line in PINNED.read_text(encoding="utf-8").splitlines():
        if line.startswith("#"):
            continue
        name, text = line.split(" ")[:2]
        (displays if name in DISPLAY_SETS else normalized).append(float(text))
    return normalized, displays


def compare(open_pools, jobs, label):
    """Ask both trees and print the count; return how many answers differ or raise."""
    before = open_pools["before"].map(ask, jobs, chunksize=8)
    after = open_pools["this tree"].map(ask, jobs, chunksize=8)
    asked = held = refused = differ = raised = 0
    first = []
    for (name, presets), old, new in zip(jobs, before, after, strict=True):
        for preset, a, b in zip(presets, old, new, strict=True):
            asked += 1
            held += a is True and b is True
            refused += a is False and b is False
            raised += isinstance(a, str) or isinstance(b, str)
            if a != b:
                differ += 1
                if len(first) < 5:
                    first.append((name, preset, a, b))
    print(f"   {label}: {len(jobs):,} records, {asked:,} presets asked of holds in both trees;"
          f" True in both {held:,}, False in both {refused:,}; either tree raises {raised:,};"
          f" answers that differ {differ:,}")
    for name, preset, a, b in first:
        print(f"      differs: {name} holds({preset!r}): before {a!r}, this tree {b!r}")
    return differ + raised

"""The revision after the review of patchbay-vital's PR 19: two pools of processes, each importing `patchbay_vital` from one tree.

Nothing is reproduced. A worker puts one tree's `src` first on `sys.path` before its first import
of the package, so every method it calls is that tree's shipped one: `main`'s, an extract of its
commit made with `git archive`, or this branch's, the working tree. `open_pools` prints the files
each pool imported and stops unless they are under the tree asked for.
"""

import multiprocessing
import sys

TREE = {}


def use(src):
    """A worker's initializer: import `patchbay_vital` from one tree, before anything else does."""
    sys.path.insert(0, src)
    import patchbay  # noqa: PLC0415
    import patchbay_vital.errors  # noqa: PLC0415
    import patchbay_vital.float32  # noqa: PLC0415
    import patchbay_vital.parameters  # noqa: PLC0415

    TREE["parameters"] = patchbay_vital.parameters
    TREE["float32"] = patchbay_vital.float32
    TREE["errors"] = patchbay_vital.errors
    TREE["id"] = patchbay.ParameterId


def where():
    """Return the files this worker's tree was imported from."""
    return TREE["parameters"].__file__, TREE["float32"].__file__


def record(name):
    """Return this worker's tree's record for an id."""
    return TREE["parameters"].parameter_map().find(TREE["id"](name))


def open_pools(sources, sizes):
    """Open one pool a tree, `sources` and `sizes` keyed by the tree's name; print what each imports."""
    context = multiprocessing.get_context("spawn")
    pools = {
        side: context.Pool(sizes[side], initializer=use, initargs=(sources[side],)) for side in sources
    }
    for side, pool in pools.items():
        files = pool.apply(where)
        print(f"== the {side} pool, {sizes[side]} processes, imports {files[0]}")
        print(f"   and {files[1]}")
        if not all(f.startswith(sources[side]) for f in files):
            print("   NOT the tree asked for: stopping")
            for other in pools.values():
                other.terminate()
            raise SystemExit(2)
    return pools
