"""The revision after the review of patchbay-vital's PR 19: every whole number of every indexed record, `main`'s methods against this branch's.

A count, not a sample: an indexed scale holds whole numbers, few enough to call every one. Issue
#16 sends `MappedParameter.to_preset`'s whole number through the set path, as a linear preset
goes, where `main` returned it without asking the scale. This shows the change moves no value of
the committed map. Two pools each import `patchbay_vital` from one tree (`pools.py`) and call
that tree's shipped scalar methods, as a caller would, on every indexed record of the map:

  for every whole number k from the scale's minimum to its maximum
    D  = record.to_display(k)          the display of k
    TP = record.to_preset(D)           the display's preset
    TN = record.to_normalized(D)       the normalized value a set of the display sends
    FN = record.from_normalized(TN)    the display `get` shows after that set
  for every half k + 0.5 below the maximum
    H  = record.to_display(k + 0.5)    a display between two whole numbers
    TP, TN of H, or `refused` where the record is a choice or a boolean, which takes a whole
    index and nothing else (ParameterRangeError)

This process compares the two trees' values on every k and counts, by scale and curve group:

  D!=, TP!=, TN!=, FN!=   a value differs from `main`'s
  not k                    this branch's TP is not k itself
  not back                 this branch's FN is not D: the display does not come back
  halves, refused, H!=     the halves called, those both trees refuse, and those where a value or
                           a refusal differs from `main`'s

It also compares, record by record, what the two trees' maps say of each indexed record: its
kind, its group key, its choices and its choice labels.

Run from the repository root:
  uv run --locked python indexed_count.py --main <main's src> --branch <this branch's src>
"""

import argparse
import sys
import time
from collections import defaultdict

import numpy as np

import pools

CHUNK = 1 << 14
COLUMNS = ("records", "calls", "D!=", "TP!=", "TN!=", "FN!=", "not k", "not back", "halves",
           "refused", "H!=")


def listing(_):
    """Return what this worker's map says of each indexed record."""
    out = []
    for r in pools.TREE["parameters"].parameter_map().parameters:
        if r.scale.kind != "indexed":
            continue
        p, s, c = r.parameter, r.scale, r.curve
        key = (p.kind, p.minimum, p.maximum, s.kind, s.minimum, s.maximum, c.kind, c.multiply,
               c.offset)
        out.append((r.id, key, p.choices, r.choice_labels))
    return out


def work(job):
    """Call one tree's methods on the whole numbers lo to hi - 1 of a record, and the halves."""
    name, lo, hi = job
    rec = pools.record(name)
    refused = pools.TREE["errors"].ParameterRangeError
    top = rec.scale.maximum
    whole = np.empty((5, hi - lo))  # k, D, TP, TN, FN
    half = np.full((3, hi - lo), np.nan)
    status = np.zeros(hi - lo, np.int8)  # 0 no half (the maximum), 1 values, 2 refused
    for i, k in enumerate(range(lo, hi)):
        d = rec.to_display(float(k))
        tn = rec.to_normalized(d)
        whole[:, i] = (k, d, rec.to_preset(d), tn, rec.from_normalized(tn))
        if k < top:
            h = rec.to_display(k + 0.5)
            half[0, i] = h
            try:
                half[1:, i] = (rec.to_preset(h), rec.to_normalized(h))
                status[i] = 1
            except refused:
                status[i] = 2
    return whole, half, status


def differs(a, b):
    """Say where two arrays differ, a NaN in both being no difference."""
    return (a != b) & ~(np.isnan(a) & np.isnan(b))


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--main", required=True)
    parser.add_argument("--branch", required=True)
    parser.add_argument("--main-processes", type=int, default=6)
    parser.add_argument("--branch-processes", type=int, default=6)
    args = parser.parse_args()
    started = time.time()
    open_pools = pools.open_pools(
        {"main": args.main, "branch": args.branch},
        {"main": args.main_processes, "branch": args.branch_processes},
    )
    lists = {side: pool.apply(listing, (None,)) for side, pool in open_pools.items()}
    same = lists["main"] == lists["branch"]
    records = lists["branch"]
    print(f"== the two trees' maps list {len(lists['main'])} and {len(records)} indexed records;"
          f" every one has the same id, kind, group key, choices and choice labels in both: {same}")
    if not same:
        print("== stopped: the maps differ")
        return 1
    groups = defaultdict(list)
    for name, key, _, _ in records:
        groups[key].append(name)
    scales = {(key[4], key[5]) for key in groups}
    print(f"== {len(records)} indexed records in {len(groups)} scale and curve groups, on"
          f" {len(scales)} distinct scales")
    jobs = []
    for name, key, _, _ in records:
        lo, hi = int(key[4]), int(key[5]) + 1
        for start in range(lo, hi, CHUNK):
            job = (name, start, min(hi, start + CHUNK))
            jobs.append((key, open_pools["main"].apply_async(work, (job,)),
                         open_pools["branch"].apply_async(work, (job,))))
    rows = {key: np.zeros(len(COLUMNS), dtype=np.int64) for key in groups}
    for key, names in groups.items():
        rows[key][0] = len(names)
    first = None
    for key, theirs_job, ours_job in jobs:
        theirs, their_half, their_status = theirs_job.get()
        ours, our_half, our_status = ours_job.get()
        row = rows[key]
        row[1] += ours.shape[1]
        for column in range(4):
            row[2 + column] += int(np.count_nonzero(differs(ours[1 + column], theirs[1 + column])))
        row[6] += int(np.count_nonzero(ours[2] != ours[0]))  # TP is not k
        row[7] += int(np.count_nonzero(ours[4] != ours[1]))  # FN is not D
        row[8] += int(np.count_nonzero(our_status))
        row[9] += int(np.count_nonzero((our_status == 2) & (their_status == 2)))  # noqa: PLR2004
        half_differs = (our_status != their_status) | differs(our_half, their_half).any(axis=0)
        row[10] += int(np.count_nonzero(half_differs))
        if first is None and (differs(ours, theirs).any() or half_differs.any()):
            first = (key, ours, theirs)
    print(f"{'group, by its first record':>34} {'kind':>10} {'curve':>12} {'scale':>16} "
          + " ".join(f"{c:>8}" for c in COLUMNS))
    total = np.zeros(len(COLUMNS), dtype=np.int64)
    for key, names in groups.items():
        total += rows[key]
        scale = f"{key[4]:g}..{key[5]:g}"
        print(f"{names[0]:>34} {key[0]:>10} {key[6]:>12} {scale:>16} "
              + " ".join(f"{int(v):>8}" for v in rows[key]))
    print(f"{'total':>34} {'':>10} {'':>12} {'':>16} " + " ".join(f"{int(v):>8}" for v in total))
    on_scales = sum(int(hi - lo) + 1 for lo, hi in scales)
    print(f"== {int(total[1]):,} whole numbers called, one call for each of a record's, over"
          f" {int(total[0])} records.")
    print(f"   The {len(scales)} distinct scales hold {on_scales:,} whole numbers between their ends.")
    print("   The two totals differ because a scale that several records share is called once for"
          " each of them:")
    by_scale = defaultdict(int)
    for key, names in groups.items():
        by_scale[(key[4], key[5])] += len(names)
    for (lo, hi), count in sorted(by_scale.items(), key=lambda item: -(item[0][1] - item[0][0])):
        size = int(hi - lo) + 1
        print(f"     scale {lo:g}..{hi:g}: {size:,} whole numbers, {count} records,"
              f" {size * count:,} calls")
    moved = int(total[2:6].sum() + total[10])
    print(f"== values that differ from main's, of the whole numbers and the halves: {moved}")
    print(f"   this branch's to_preset is not the whole number itself on {int(total[6])}; its"
          f" display does not come back through the normalized path on {int(total[7])}")
    print(f"  {time.time() - started:.0f} s")
    for pool in open_pools.values():
        pool.close()
    if moved or total[6] or total[7]:
        print("== stopped: a value differs; the first group:", first[0] if first else None)
        return 1
    return 0


if __name__ == "__main__":
    sys.exit(main())

TREE = {}


def use(src):
    """A worker's initializer: import `patchbay_vital` from one tree, before anything else does."""
    sys.path.insert(0, src)
    import patchbay  # noqa: PLC0415
    import patchbay_vital.errors  # noqa: PLC0415
    import patchbay_vital.float32  # noqa: PLC0415
    import patchbay_vital.parameters  # noqa: PLC0415

    TREE["parameters"] = patchbay_vital.parameters
    TREE["float32"] = patchbay_vital.float32
    TREE["errors"] = patchbay_vital.errors
    TREE["id"] = patchbay.ParameterId


def where():
    """Return the files this worker's tree was imported from."""
    return TREE["parameters"].__file__, TREE["float32"].__file__


def record(name):
    """Return this worker's tree's record for an id."""
    return TREE["parameters"].parameter_map().find(TREE["id"](name))


def open_pools(sources, sizes):
    """Open one pool a tree, `sources` and `sizes` keyed by the tree's name; print what each imports."""
    context = multiprocessing.get_context("spawn")
    pools = {
        side: context.Pool(sizes[side], initializer=use, initargs=(sources[side],)) for side in sources
    }
    for side, pool in pools.items():
        files = pool.apply(where)
        print(f"== the {side} pool, {sizes[side]} processes, imports {files[0]}")
        print(f"   and {files[1]}")
        if not all(f.startswith(sources[side]) for f in files):
            print("   NOT the tree asked for: stopping")
            for other in pools.values():
                other.terminate()
            raise SystemExit(2)
    return pools
