        indexed scale the nearest whole number, as the plugin holds it. Where a linear scale does
        not hold that float32, in this order:

def _landing(scale: NormalizedScale, preset: float) -> tuple[float, float]:
    """Return `to_normalized`'s value for a preset and the preset value it lands on, each once.
    The step toward the preset is kept only inside `[0, 1]`, where the host can set it: on
    `NormalizedScale(kind="linear", minimum=-1.0, maximum=16777216.0)` the float32 above 1 lands on
    the maximum, which the nearest quotient, 1, misses.
    """
    nearest = round_to_float32((Fraction(preset) - Fraction(scale.minimum)) / Fraction(scale.span))
    landed = scale.to_preset(nearest)
    if landed == preset:
        return nearest, landed
    toward = next_float32(nearest, math.copysign(math.inf, preset - landed))
    if 0.0 <= toward <= 1.0:
        stepped = scale.to_preset(toward)
        if stepped == preset:
            return toward, stepped
    return nearest, landed
def next_float32(value: float, toward: float) -> float:
    """Return the float32 next to `value` in the direction of `toward`, as C's `nextafterf` does.
    Both are float32s to `nextafterf`, so a double that is not one is first rounded to the nearest,
    `toward` as well as `value`: `toward` 1.0 + 1e-12 is 1.0, and from 1.0 there is no step. A
    `toward` equal to the value is returned; from a zero the step is the smallest subnormal of
    `toward`'s sign, and from an infinity the largest float32; a NaN gives a NaN. The step is made
    on the float32's bits, so it is exact.
    """
    start, toward = to_float32(value), to_float32(toward)
    if math.isnan(start) or math.isnan(toward):
        return math.nan
    if start == toward:
        return toward
    if start == 0:
        return math.copysign(math.ldexp(1.0, _SUBNORMAL_QUANTUM), toward)
    bits: int = _BITS.unpack(_FLOAT32.pack(start))[0]
    bits += 1 if (start < toward) == (start > 0) else -1
    stepped: float = _FLOAT32.unpack(_BITS.pack(bits))[0]
    return stepped
