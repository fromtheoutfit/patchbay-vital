"""The revision after PR 14's sixth review: the floor pass, `e852aac`'s shipped methods against the new tree's.

Nothing is reproduced. Pools of processes each import `patchbay_vital` from one tree, `e852aac`'s
(an extract of that commit's `src`) or the new one (the working tree), by putting that tree's `src`
first on `sys.path` before the first import, and call that tree's own scalar methods on every
float32 normalized value r of a record, as a caller would:

  P0 = scale.to_preset(r)            the preset the parameter holds
  D0 = record.from_normalized(r)     its display
  TP = record.to_preset(D0)          the display's preset
  TN = record.to_normalized(D0)      the normalized value a set of the display sends

The new pool also imports `tests/unit/test_parameters.py` from the working tree and asks its own
excuse, `_unnamed_by_its_display(record, D0, P0, TP)`, wherever TP differs from P0, with the two
candidates from its own `_what_a_set_holds(record, D0)`: what a set of the curve's float32 holds
(HELD) and the float32 on that float32's other side (OTHER). The test's helper is called once a
value: a one-entry cache in front of it hands the predicate the triple already worked out.

Counted by binade of r, each count also split at `volume`'s floor exactly, by P0 <= 2 ** -43 and
P0 > 2 ** -43, value by value:

  DIFF       TP != P0, the values `_assert_comes_back` asks the excuse of
  BY_HELD    of those, excused with TP the HELD candidate
  BY_OTHER   of those, excused with TP the OTHER candidate
  NEITHER    of those, not excused: a stop
and, where both trees run (section A and the envelope powers):
  P0!=, D0!= P0 or D0 differs between the trees (the scale and curve forward, which this revision
             leaves alone)
  TP!=       the new `to_preset` differs from `e852aac`'s: a stop
  TN!=       the new, derived `to_normalized` differs from `e852aac`'s: a stop

Run from the repository root:
  uv run --locked python floor_pass.py --old <e852aac's src> --new <the working tree's src>
      --tests <the working tree's tests/unit> --record <id> [--from <binade> --to <binade>]
      [--new-only] [--time] [--powers] [--old-processes <n>] [--new-processes <n>]
"""

import argparse
import math
import multiprocessing
import sys
import time

import numpy as np

FLOOR = 2.0**-43
CHUNK = 1 << 16
SPANS = [("zero", 0, 1), ("sub", 1, 1 << 23)]
SPANS += [(str(e - 127), e << 23, (e + 1) << 23) for e in range(1, 127)]
SPANS += [("one", 0x3F800000, 0x3F800001)]
COMPARED = ("P0!=", "D0!=", "TP!=", "TN!=")
CLASSES = ("DIFF", "BY_HELD", "BY_OTHER", "NEITHER")
STOPS = ("TP!=", "TN!=", "NEITHER")

_TREE = {}


def use(src, tests):
    """A worker's initializer: import `patchbay_vital` from one tree, before anything else does."""
    sys.path.insert(0, src)
    import patchbay  # noqa: PLC0415
    import patchbay_vital.parameters  # noqa: PLC0415

    _TREE["parameters"] = patchbay_vital.parameters
    _TREE["id"] = patchbay.ParameterId
    if tests:
        sys.path.insert(1, tests)
        import test_parameters  # noqa: PLC0415

        real = test_parameters._what_a_set_holds
        last = {}

        def once(record, display):
            if last.get("key") != (record.id, display):
                last["key"], last["value"] = (record.id, display), real(record, display)
            return last["value"]

        test_parameters._what_a_set_holds = once
        _TREE["tests"] = test_parameters


def where():
    """Return the files this worker's tree and tests were imported from."""
    files = [_TREE["parameters"].__file__]
    if "tests" in _TREE:
        files.append(_TREE["tests"].__file__)
    return files


def work(job):
    """Call one tree's methods on every float32 normalized value of a bit-pattern range."""
    name, lo, hi, normalized_too = job
    record = _TREE["parameters"].parameter_map().find(_TREE["id"](name))
    scale = record.scale
    tests = _TREE.get("tests")
    values = np.arange(lo, hi, dtype=np.uint32).view(np.float32).astype(np.float64).tolist()
    out = np.full((4, len(values)), np.nan)
    kind = np.zeros(len(values), np.int8)  # 0 same, 1 held, 2 other, 3 neither
    for i, r in enumerate(values):
        p0 = scale.to_preset(r)
        d0 = record.from_normalized(r)
        tp = record.to_preset(d0)
        out[0, i], out[1, i], out[2, i] = p0, d0, tp
        if normalized_too:
            out[3, i] = record.to_normalized(d0)
        if tests is None or tp == p0:
            continue
        _, held, _ = tests._what_a_set_holds(record, d0)
        if not tests._unnamed_by_its_display(record, d0, p0, tp):
            kind[i] = 3
        else:
            kind[i] = 1 if tp == held else 2
    return lo, out, kind


def tally(new, old, kind):
    """Return the counts at or below the floor and above it, and where the run stops."""
    p0 = new[0]
    if old is None:
        differs = [np.zeros(p0.shape, bool)] * 4
    else:
        differs = [new[k] != old[k] for k in range(4)]
    classes = [kind != 0, kind == 1, kind == 2, kind == 3]  # noqa: PLR2004
    flags = (*differs, *classes)
    below = p0 <= FLOOR
    rows = []
    for side in (below, ~below):
        rows.append([int(np.count_nonzero(side)), *(int(np.count_nonzero(f & side)) for f in flags)])
    stop = differs[2] | differs[3] | classes[3]
    return np.array(rows, dtype=np.int64), np.flatnonzero(stop)


def spans(first, last):
    labels = [label for label, _, _ in SPANS]
    lo = 0 if first is None else labels.index(first)
    hi = len(SPANS) - 1 if last is None else labels.index(last)
    return SPANS[lo : hi + 1]


def line(label, row):
    return f"{label:>12} " + " ".join(f"{int(v):>11}" for v in row)


def pools_for(args, new_only):
    context = multiprocessing.get_context("spawn")
    if new_only:
        sizes = {"new": args.old_processes + args.new_processes}
    else:
        sizes = {"old": args.old_processes, "new": args.new_processes}
    sources = {"old": (args.old, None), "new": (args.new, args.tests)}
    pools = {s: context.Pool(n, initializer=use, initargs=sources[s]) for s, n in sizes.items()}
    for side, pool in pools.items():
        files = pool.apply(where)
        print(f"== the {side} pool, {sizes[side]} processes, imports {files[0]}")
        for f in files[1:]:
            print(f"   and {f}")
        if not files[0].startswith(sources[side][0]):
            print("   NOT the tree asked for: stopping")
            sys.exit(2)
    return pools, sizes


def timing(args):
    """Time both sections on 200,000 values, and estimate the whole pass."""
    rng = np.random.default_rng(13)
    below = (127 - 54) << 23
    estimate = 0.0
    for section, new_only, lo, hi, count in (
        ("A", False, 0, below, below),
        ("B", True, below, 0x3F800001, 0x3F800001 - below),
    ):
        pools, sizes = pools_for(args, new_only)
        took = {}
        for side, pool in pools.items():
            started = time.time()
            jobs = []
            for start in rng.integers(lo, hi - 50_000, 4, dtype=np.uint32).tolist():
                job = (args.record, start, start + 50_000, not new_only)
                jobs.append(pool.apply_async(work, (job,)))
            for job in jobs:
                job.get()
            took[side] = (time.time() - started) / 50_000 * 1e6
            print(f"  section {section}, {side}: {took[side]:.1f} us a value, 4 ranges of 50,000"
                  " values in 4 processes at once")
        wall = max(took[s] / sizes[s] for s in took) * count / 3.6e9
        cpu = sum(took.values()) * count / 3.6e9
        print(f"  section {section}, {count:,} values: about {cpu:.1f} CPU-hours, {wall:.2f} hours"
              " wall on these pools")
        estimate += wall
        for pool in pools.values():
            pool.terminate()
    print(f"== the whole pass, sections A and B: about {estimate:.2f} hours wall")
    if estimate > args.budget:
        print(f"   more than the {args.budget} hours the owner allowed: stopping")
        return 3
    return 0


def powers(name):
    """Return, from one tree, each held power of two of a record's scale and its display's methods."""
    parameters = _TREE["parameters"]
    record = parameters.parameter_map().find(_TREE["id"](name))
    scale = record.scale
    found = []
    for exponent in range(-149, 2):
        preset = math.ldexp(1.0, exponent)
        if scale.minimum <= preset <= scale.maximum and scale.holds(preset):
            display = record.to_display(preset)
            found.append((exponent, preset, scale.to_normalized(preset), display,
                          record.to_preset(display), record.to_normalized(display)))
    return found


def held_powers(args):
    pools, _ = pools_for(args, False)
    old = pools["old"].apply(powers, (args.record,))
    new = pools["new"].apply(powers, (args.record,))
    print(f"== {args.record}: every power of two from 2 ** -149 to 2 the scale holds, through both trees")
    print("   exponent  preset  normalized  display | to_preset(display), to_normalized(display):"
          " new, then e852aac's")
    if [row[:4] for row in old] != [row[:4] for row in new]:
        print("   the two trees hold different powers, or give them other values: stopping")
        return 1
    differ = 0
    for a, b in zip(new, old, strict=True):
        same = a[4:] == b[4:]
        differ += not same
        print(f"   {a[0]:>4} {a[1]!r} {a[2]!r} {a[3]!r} | {a[4]!r} {a[5]!r}; {b[4]!r} {b[5]!r}"
              f"{'' if same else '  DIFFERS'}")
    print(f"   {len(new)} held powers of two; to_preset or to_normalized differs on {differ}")
    for pool in pools.values():
        pool.terminate()
    return 1 if differ else 0


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--old", required=True)
    parser.add_argument("--new", required=True)
    parser.add_argument("--tests", required=True)
    parser.add_argument("--record", required=True)
    parser.add_argument("--from", dest="first")
    parser.add_argument("--to", dest="last")
    parser.add_argument("--new-only", action="store_true")
    parser.add_argument("--old-processes", type=int, default=4)
    parser.add_argument("--new-processes", type=int, default=8)
    parser.add_argument("--time", action="store_true")
    parser.add_argument("--powers", action="store_true")
    parser.add_argument("--budget", type=float, default=4.0)
    args = parser.parse_args()
    if args.time:
        return timing(args)
    if args.powers:
        return held_powers(args)
    pools, _ = pools_for(args, args.new_only)
    columns = ("n", *([] if args.new_only else COMPARED), *CLASSES)
    keep = slice(None) if not args.new_only else [0, 5, 6, 7, 8]
    chosen = spans(args.first, args.last)
    trees = "the new tree" if args.new_only else "both trees"
    print(f"== {args.record}: binades {chosen[0][0]} to {chosen[-1][0]},"
          f" {sum(hi - lo for _, lo, hi in chosen):,} float32 normalized values, each through"
          f" {trees}")
    print(line("binade", ()) + " ".join(f"{c:>11}" for c in columns))
    started = time.time()
    total = np.zeros((2, 9), dtype=np.int64)
    for label, lo, hi in chosen:
        row = np.zeros((2, 9), dtype=np.int64)
        jobs = []
        for start in range(lo, hi, CHUNK):
            end = min(hi, start + CHUNK)
            job = (args.record, start, end, not args.new_only)
            jobs.append((
                None if args.new_only else pools["old"].apply_async(work, (job,)),
                pools["new"].apply_async(work, (job,)),
            ))
        for old_job, new_job in jobs:
            start, new, kind = new_job.get()
            old = None if old_job is None else old_job.get()[1]
            counted, stops = tally(new, old, kind)
            row += counted
            if stops.size:
                print(line(label, row.sum(axis=0)[keep]))
                print(f"== STOP: {stops.size} values in the range from bit pattern {start}; the first:")
                for i in stops[:5].tolist():
                    bits = np.array([start + i], dtype=np.uint32).view(np.float32)[0]
                    print(f"   r {float(bits)!r}: the new tree's P0, D0, TP, TN"
                          f" {[float(v) for v in new[:, i]]!r}, e852aac's"
                          f" {None if old is None else [float(v) for v in old[:, i]]!r},"
                          f" class {kind[i]}")
                print("== stopped at once, as the owner asked")
                for pool in pools.values():
                    pool.terminate()
                return 1
        total += row
        whole = row.sum(axis=0)
        print(line(label, whole[keep]))
        if row[0, 0] and row[1, 0]:
            print(line("<= 2**-43", row[0][keep]))
            print(line("> 2**-43", row[1][keep]))
        sys.stdout.flush()
    print(line("total", total.sum(axis=0)[keep]))
    print(line("P0 <= 2**-43", total[0][keep]))
    print(line("P0 > 2**-43", total[1][keep]))
    print(f"  {time.time() - started:.0f} s")
    for pool in pools.values():
        pool.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())

def _what_a_set_holds(record: MappedParameter, display: float) -> tuple[float, float, float]:
    """Return the curve's float32 for a display, what a set of it holds, and its other side."""
    own = record.curve.to_preset(display)
    held = record.scale.to_preset(record.scale.to_normalized(own))
    return own, held, next_float32(own, math.copysign(math.inf, own - held))

def _unnamed_by_its_display(
    record: MappedParameter, display: float, preset: float, back: float
) -> bool:
    """Say whether a display does not name its held preset, and `back` is what a set of it holds.

    The display names a preset where the curve's own float32 for it, `curve.to_preset`, is that
    preset: then nothing else is excused. Where it is not, `back` must be one of two candidates
    worked out here from the scale's and the curve's primitives, never from `to_preset`, so a wrong
    rule in the map cannot agree with itself: what a set of the curve's float32 holds, the scale's
    value at its normalized value, or the float32 on that float32's other side
    (`_what_a_set_holds`). The display, shown by `back` as well, is checked beside it in

import textwrap

    (NormalizedScale, "to_preset"), (NormalizedScale, "to_normalized"), (NormalizedScale, "holds"),

def source(owner, name):
    member = inspect.getattr_static(owner, name)
    member = member.fget if isinstance(member, property) else member
    member = getattr(member, "__wrapped__", member)
    lines, first = inspect.getsourcelines(member)
    return ast.parse(textwrap.dedent("".join(lines))), first


def reads(tree):
    """Return every `<name>.<attribute>` read in a function, as written."""
    found = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Attribute) and isinstance(node.value, ast.Name):
            found.add(f"{node.value.id}.{node.attr}")
        elif isinstance(node, ast.Attribute) and isinstance(node.value, ast.Attribute):
            inner = node.value
            if isinstance(inner.value, ast.Name):
                found.add(f"{inner.value.id}.{inner.attr}.{node.attr}")
    return sorted(found)


def main():
    m = parameter_map()
    records = [r for r in m.parameters
               if r.id.startswith("env_") and r.id.rsplit("_", 1)[1] in ("delay", "hold")]
    print(f"== the {len(records)} envelope delay and hold records, every field the calls read:")
    print("   id: parameter.kind parameter.minimum parameter.maximum | scale.kind scale.minimum"
          " scale.maximum | curve.kind curve.multiply curve.offset")
    keys = set()
