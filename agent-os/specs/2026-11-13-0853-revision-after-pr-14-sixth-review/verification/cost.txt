"""The revision after PR 14's sixth review: what a call of the map costs, in one tree.

Imports `patchbay_vital` from the tree given, first on sys.path, and prints its file to show it.
Times, the minimum of 7 repeats, each a loop over a fixed set of values after one warm-up loop:
- `volume`'s `to_preset` and `to_normalized` on 1,000 displays, those of the normalized values
  i / 999 for i in 0..999, through `from_normalized`;
- the same two on 1,000 displays at or below the floor, those of the normalized values
  i / 999 * 1.53642479121845e-17 (the floor's own normalized value, preset 2 ** -43), the
  region where the rule's branches are asked;
- `to_normalized` of every record of the map at its own default display, the whole map once.
Run from the repository root: uv run --locked python <scratchpad>/r6/cost.py <src> <label>
"""

import sys
import timeit

src, label = sys.argv[1], sys.argv[2]
sys.path.insert(0, src)

from patchbay import ParameterId  # noqa: E402

import patchbay_vital.parameters as p  # noqa: E402

the_map = p.parameter_map()
volume = the_map.find(ParameterId("volume"))
floor = 1.53642479121845e-17
sets = {
    "uniform": [volume.from_normalized(i / 999) for i in range(1000)],
    "floor": [volume.from_normalized(i / 999 * floor) for i in range(1000)],
}
defaults = [(r, r.parameter.default.value) for r in the_map.parameters]


def per_call(function, values):
    def loop():
        for value in values:
            function(value)

    loop()
    return min(timeit.repeat(loop, number=1, repeat=7)) / len(values) * 1e6


def whole_map():
    for record, display in defaults:
        record.to_normalized(display)


whole_map()
whole = min(timeit.repeat(whole_map, number=1, repeat=7)) * 1e3
print(f"{label}: imports {p.__file__.split('/src/')[-1]} from {src.rsplit('/', 2)[-2]}/src")
for name, values in sets.items():
    print(
        f"  volume, {name:7} 1,000 displays: to_preset {per_call(volume.to_preset, values):6.1f} µs,"
        f" to_normalized {per_call(volume.to_normalized, values):6.1f} µs a call"
    )
print(f"  the whole map, {len(defaults)} records at their defaults: to_normalized {whole:5.1f} ms")
