"""Every test, function and class name the four standards cite, searched for in the tree.

usage: cited_names.py <the revision's mutation.txt>   (from the repository root)

The four are the standards the revision after the review of PR 26 edited: `vital/plugin-state.md`,
`vital/instrument.md`, `vital/preset-tests.md` and `vital/preset-file.md`. Every span in
backticks that is one identifier, or identifiers joined by dots, is a candidate:

- a test name, `test_...`, is found where git lists a `tests/` file holding `def <name>(`;
- a name of a function or a class, a span whose last part begins with an underscore or a capital
  or is followed by `(` in the standard, or a dotted span, is found where git lists a `src/` or
  `tests/` file holding `def <last part>(`, `class <last part>` or `<last part> =` at the start of
  a line, after its indent; a dotted span's first part, where it names a module of the package,
  must hold it;
- any other identifier, a preset key or a constant's value among them, is listed as not
  searched for, with the rule that set it aside.

Also every mutant name the four cite (`MW1`, `MR12` and the like), against the mutation
transcript given and the issue #20 spec's, and every transcript name they cite (`name.txt`),
against the `verification/` directories of the specs under `agent-os/specs/`.
"""

import re
import subprocess
import sys
from pathlib import Path

STANDARDS = [
    Path("agent-os/standards/vital") / f"{name}.md"
    for name in ("plugin-state", "instrument", "preset-tests", "preset-file")
]
SPAN = re.compile(r"`([A-Za-z_][A-Za-z0-9_]*(?:\.[A-Za-z_][A-Za-z0-9_]*)*)`(\()?")
MUTANT = re.compile(r"\b(M[A-Z]{1,2}\d+)\b")
RANGE = re.compile(r"\b(M[A-Z]{1,2})(\d+) to (?:M[A-Z]{1,2})?(\d+)\b")
TRANSCRIPT = re.compile(r"`([a-z0-9-]+\.txt)`")
MODULES = {path.stem for path in Path("src/patchbay_vital").glob("*.py")}
FILE_SUFFIXES = (".py", ".md", ".txt", ".json", ".yml", ".toml", ".vital", ".sh")


def listed(*prefixes: str) -> list[Path]:
    out = subprocess.run(["git", "ls-files", "-co", "--exclude-standard", *prefixes], capture_output=True, text=True, check=True).stdout  # noqa: S603, S607
    return [Path(line) for line in out.splitlines() if line.endswith(".py")]


SOURCE = {path: path.read_text(encoding="utf-8") for path in listed("src", "tests")}


def defined(name: str, where: list[Path] | None = None) -> list[str]:
    pattern = re.compile(rf"^\s*(?:def {re.escape(name)}\(|async def {re.escape(name)}\(|class {re.escape(name)}\b|{re.escape(name)}\s*(?::[^=]*)?=)", re.MULTILINE)
    paths = where if where is not None else list(SOURCE)
    return sorted(str(path) for path in paths if path in SOURCE and pattern.search(SOURCE[path]))


def main() -> int:
    mutation = Path(sys.argv[1]).read_text(encoding="utf-8")
    old_mutation = Path("agent-os/specs/2026-11-16-1641-a-load-that-cannot-block/verification/mutation.txt").read_text(encoding="utf-8")
    transcripts = {path.name: path for path in Path("agent-os/specs").glob("*/verification/*.txt")}
    missing_total = 0
    for standard in STANDARDS:
        text = standard.read_text(encoding="utf-8")
        flat = " ".join(text.split())
        tests, names, skipped = {}, {}, {}
        for match in SPAN.finditer(flat):
            span, called = match.group(1), match.group(2) is not None
            last = span.split(".")[-1]
            if span.endswith(FILE_SUFFIXES) or span in MODULES:
                continue
            if last.startswith("test_"):
                tests[span] = [str(p) for p in SOURCE if str(p).startswith("tests/") and f"def {last}(" in SOURCE[p]]
            elif "." in span or last.startswith("_") or last[:1].isupper() or called:
                first = span.split(".")[0]
                where = [Path(f"src/patchbay_vital/{first}.py")] if "." in span and first in MODULES else None
                names[span] = defined(last, where)
            else:
                skipped[span] = "one lower-case identifier, not called: a key, a value or a word, not searched for"
        print(f"== {standard}")
        print(f"test names cited: {len(tests)}; not found: {sum(not found for found in tests.values())}")
        for span, found in sorted(tests.items()):
            print(f"  {'found  ' if found else 'NOT FOUND'}  {span}  {', '.join(found)}")
        print(f"function and class names cited: {len(names)}; not found: {sum(not found for found in names.values())}")
        for span, found in sorted(names.items()):
            print(f"  {'found  ' if found else 'NOT FOUND'}  {span}  {', '.join(found)}")
        print(f"other identifiers, not searched for: {len(skipped)}: {', '.join(sorted(skipped))}")
        cited = set(MUTANT.findall(flat))
        for prefix, low, high in RANGE.findall(flat):
            cited |= {f"{prefix}{n}" for n in range(int(low), int(high) + 1)}
        here = {m for m in cited if re.search(rf"^{m}  ", mutation, re.MULTILINE)}
        there = {m for m in cited if re.search(rf"^{m}  ", old_mutation, re.MULTILINE)}
        nowhere = sorted(cited - here - there)
        print(f"mutants cited: {len(cited)}; in this revision's mutation.txt: {len(here)}; in the issue #20 spec's only: {len(there - here)}; in neither: {nowhere}")
        cited_transcripts = sorted(set(TRANSCRIPT.findall(flat)))
        absent = [name for name in cited_transcripts if name not in transcripts]
        print(f"transcripts cited: {len(cited_transcripts)}; in no spec's verification/: {absent}")
        missing_total += sum(not f for f in tests.values()) + sum(not f for f in names.values()) + len(nowhere) + len(absent)
        print()
    print(f"in all, cited and not found: {missing_total}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"

#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*" | sed "s|$R|<scratchpad>/p26|g"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/p26/$(basename "$s")"; done
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/p26/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
