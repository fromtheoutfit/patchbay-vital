"""The tree the single rerun read, against the tree now: every path that differs, code or not.

usage: tie.py <transcript>...   (from the repository root)

Each transcript prints the tree before and after its run, by `git status` and a blob a path. The
four of the single rerun of the revision after the review of PR 26 must print one tree, before
and after alike. That tree is set against the tree now, as `tree.sh` prints it: a path is listed
where its blob differs or it is in one alone. A path under `src/`, `tests/`, `scripts/` or
`.github/`, or `pyproject.toml` or `uv.lock`, is code; any other is not. A code path that differs
is a stop: the rerun would then not be the final tree's.
"""

import re
import subprocess
import sys
from pathlib import Path

BLOB = re.compile(r"^    ([0-9a-f]{40}|\(deleted\))  (.+)$")
CODE = ("src/", "tests/", "scripts/", ".github/")
CODE_FILES = ("pyproject.toml", "uv.lock")


def trees(text: str) -> list[dict[str, str]]:
    """Each tree a transcript prints, before and after: path to blob."""
    found, current = [], None
    for line in text.splitlines():
        if line.startswith("the tree before the run") or line.startswith("the tree after the run"):
            current = {}
            found.append(current)
        elif current is not None:
            match = BLOB.match(line)
            if match:
                current[match.group(2)] = match.group(1)
            elif line.startswith("  HEAD ") or line.startswith("  git ") or line.startswith("    ") and not match:
                continue
            elif not line.startswith("  "):
                current = None
    return found


def code(path: str) -> bool:
    return path.startswith(CODE) or path in CODE_FILES


def main() -> int:
    read = []
    for name in sys.argv[1:]:
        both = trees(Path(name).read_text(encoding="utf-8"))
        print(f"{Path(name).name}: {len(both)} trees printed, {', '.join(str(len(t)) + ' paths' for t in both)}")
        read.extend(both)
    first = read[0]
    same = all(tree == first for tree in read)
    print(f"every tree the {len(sys.argv) - 1} transcripts printed, before and after, is one tree: {same}")
    if not same:
        print("STOP: the rerun's transcripts printed more than one tree")
        return 1
    now = trees("the tree before the run\n" + subprocess.run(["bash", str(Path(__file__).parent / "tree.sh")], capture_output=True, text=True, check=True).stdout)[0]  # noqa: S603, S607
    head = subprocess.run(["git", "rev-parse", "HEAD"], capture_output=True, text=True, check=True).stdout.strip()  # noqa: S603, S607
    print(f"the tree now: HEAD {head}, {len(now)} paths that git status names")
    differ = sorted(path for path in set(first) | set(now) if first.get(path) != now.get(path))
    print(f"\npaths that differ between the rerun's tree and the tree now: {len(differ)}")
    for path in differ:
        kind = "CODE" if code(path) else "not code"
        print(f"  {kind:<8}  {path}  rerun {first.get(path, '(absent)')[:12]}  now {now.get(path, '(absent)')[:12]}")
    codes = [path for path in differ if code(path)]
    print(f"\ncode paths that differ: {len(codes)} {codes}")
    if codes:
        print("STOP: a code path differs, so the rerun is not the final tree's")
        return 1
    print("every path that differs is not code: the rerun ran on the final tree's code, tests and scripts")
    return 0


if __name__ == "__main__":
    sys.exit(main())

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"
