```
"""Every mutant a standard cites with no spec named for it: the four standards before the fix and now.

usage: bare_citations.py <label=path>...   (from the repository root)

A mutant is a name like `ML2` or a range like `MS1 to MS4`, counted once either way. It is cited
with its spec where the clause that holds it names one: after it, up to the next ";", a
`mutation.txt`, a spec's or a revision's name, or "the revision after"; or before it, in the same
sentence and after the last ";", a spec's or a revision's name, "the spec of issue #N's", "in its"
or "and its", or a `mutation.txt` followed by "where". Any other is bare. Each path is printed with
its blob, so the copy read is the one named.
"""

import re
import subprocess
import sys
from pathlib import Path

TOKEN = re.compile(r"\b(M[A-Z]{1,2}\d+)(?: to (?:M[A-Z]{1,2})?\d+)?\b")
AFTER = re.compile(r"mutation\.txt|spec's|revision's|spec of issue|revision after|first revision|second review")
BEFORE = re.compile(r"(?:spec's|revision's|spec of issue #\d+'s|issue #\d+'s|in its|and its|mutation\.txt`, where) [^.;]*$|(?:and its|in its|where) $")


def bare(text: str) -> list[tuple[str, str]]:
    flat = " ".join(text.split())
    found = []
    for match in TOKEN.finditer(flat):
        start = max(flat.rfind(". ", 0, match.start()), flat.rfind("- **", 0, match.start()))
        after = flat[match.end() : match.end() + 160].split(";")[0]
        before = flat[max(start, match.start() - 200) : match.start()].split(";")[-1]
        if not (AFTER.search(after) or BEFORE.search(before)):
            found.append((match.group(0), before[-70:]))
    return found


def main() -> int:
    for argument in sys.argv[1:]:
        label, path = argument.split("=", 1)
        blob = subprocess.run(["git", "hash-object", path], capture_output=True, text=True, check=True).stdout.strip()  # noqa: S603, S607
        found = bare(Path(path).read_text(encoding="utf-8"))
        print(f"\n== {label}: {Path(path).name}, blob {blob}: {len(found)} mutants or ranges cited bare")
        for token, before in found:
            print(f"  {token:<12} ...{before}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"
```

```
set -euo pipefail
echo "$(uname -m), $(sysctl -n machdep.cpu.brand_string), $(sysctl -n hw.ncpu) cores, macOS $(sw_vers -productVersion) ($(sw_vers -buildVersion)), Python $(uv run --locked python -c 'import platform; print(platform.python_version())'), load $(sysctl -n vm.loadavg | tr -d '{}' | xargs), at $(date '+%Y-%m-%d %H:%M %Z')"
```

```
#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*" | sed "s|$R|<scratchpad>/p26|g"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in ${scripts[@]+"${scripts[@]}"} "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/p26/$(basename "$s")"; done
  for s in ${scripts[@]+"${scripts[@]}"} "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/p26/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
```
