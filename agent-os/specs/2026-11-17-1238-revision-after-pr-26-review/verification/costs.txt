"""The cost transcript (PR 26 review, findings 12, 13 and 15). Loads no plugin.

usage: costs.py   (run from the repository root, under `uv run --locked python`)

12: what a decode of a preset's text costs, the decode `_planted` adds to every load: the median
    of seven, on the installed preset of median size and on the largest, of the 75 the manifest
    names.
15: what importing the host costs on macOS, and what building the alert watch costs after it;
    and the frameworks pedalboard's native module links, by `otool -L`.
13: the first nesting depth the package's decoder refuses, on one text shape, `{"a":` then n
    arrays: called directly, as `_planted` calls it; through `PluginState.with_text`, the
    validator `load_state` runs first; and directly again under 800 more Python frames.
"""

import platform
import statistics
import subprocess
import sys
import time
from pathlib import Path

ROOT = Path.cwd()
PRESETS = Path.home() / "Music" / "Vital"
MANIFEST = ROOT / "tests" / "data" / "factory-manifest.txt"
REPEATS = 7

print(f"Python {platform.python_version()} ({sys.implementation.name}), recursion limit {sys.getrecursionlimit()}")
print()

print("== 15: importing the host, and building the alert watch")
started = time.perf_counter()
import pedalboard  # noqa: E402

after_pedalboard = time.perf_counter()
from patchbay_vital import host  # noqa: E402

after_host = time.perf_counter()
print(f"import pedalboard: {1000 * (after_pedalboard - started):.1f} ms, first in this process")
print(f"import patchbay_vital.host after it: {1000 * (after_host - after_pedalboard):.1f} ms, which builds the module's watch once")
builds = []
for _ in range(REPEATS):
    began = time.perf_counter()
    host._alert_watch()  # noqa: SLF001  the build the module makes at import
    builds.append(1000 * (time.perf_counter() - began))
print(f"host._alert_watch() again, {REPEATS} times: median {statistics.median(builds):.3f} ms, max {max(builds):.3f} ms")
native = Path(pedalboard.__file__).parent.parent / "pedalboard_native.cpython-312-darwin.so"
if not native.exists():
    native = next(Path(pedalboard.__file__).parent.parent.glob("pedalboard_native*.so"))
listing = subprocess.run(["/usr/bin/otool", "-L", str(native)], capture_output=True, text=True, check=True).stdout  # noqa: S603
print(f"otool -L {native.name}, the lines naming AppKit, Cocoa, Foundation or libobjc:")
for line in listing.splitlines():
    if any(name in line for name in ("AppKit", "Cocoa", "Foundation", "libobjc")):
        print(f"  {line.strip()}")
print()

print("== 12: a decode of a preset's text, as _planted makes it")
from patchbay_vital.preset import DECODER  # noqa: E402

names = [
    line
    for line in MANIFEST.read_text(encoding="utf-8").splitlines()
    if line and not line.startswith("#")
]
files = sorted((PRESETS / name for name in names), key=lambda path: path.stat().st_size)
print(f"installed presets read: {len(files)}, sizes {files[0].stat().st_size:,} to {files[-1].stat().st_size:,} bytes")
for label, path in (("the median-sized", files[len(files) // 2]), ("the largest", files[-1])):
    raw = path.read_bytes()
    times = []
    for _ in range(REPEATS):
        began = time.perf_counter()
        DECODER.decode(raw.decode("utf-8"))
        times.append(1000 * (time.perf_counter() - began))
    print(f"{label}, {path.name}, {len(raw):,} bytes: median {statistics.median(times):.2f} ms, min {min(times):.2f}, max {max(times):.2f}")
print()

print("== 13: the first depth the decoder refuses, on the text {\"a\": then n arrays, each closed")
from patchbay_vital.errors import PluginStateError  # noqa: E402
from patchbay_vital.state import PluginState  # noqa: E402

STATE = PluginState.model_validate(
    {
        "class_id": b"x" * 32,
        "container_version": 1,
        "wrapper_version": 1,
        "bypass": 0,
        "bank_version": 1,
        "plugin_id": b"Vita",
        "plugin_version": 1,
        "programs": 1,
        "reserved": b"\0" * 128,
        "text": b"{}",
    }
)


def text(depth: int) -> str:
    return '{"a":' + "[" * depth + "]" * depth + "}"


def direct(depth: int) -> bool:
    try:
        DECODER.decode(text(depth))
    except RecursionError:
        return False
    return True


def validated(depth: int) -> bool:
    try:
        STATE.with_text(text(depth).encode("utf-8"))
    except PluginStateError:
        return False
    return True


def first_refused(reads) -> int:  # noqa: ANN001
    low, high = 1, 1 << 16
    assert reads(low) and not reads(high)
    while high - low > 1:
        middle = (low + high) // 2
        if reads(middle):
            low = middle
        else:
            high = middle
    return high


def under(frames: int, call):  # noqa: ANN001, ANN201
    return call() if frames == 0 else under(frames - 1, call)


print(f"called directly, as _planted calls it: {first_refused(direct)}")
print(f"through PluginState.with_text, the validator load_state runs first: {first_refused(validated)}")
print(f"called directly under 800 more Python frames: {under(800, lambda: first_refused(direct))}")

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"

#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*" | sed "s|$R|<scratchpad>/p26|g"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/p26/$(basename "$s")"; done
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/p26/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
