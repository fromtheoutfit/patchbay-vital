```
"""The midpoint probe (PR 26 review, finding 10): what Vital holds for its switch between the ends.

usage: midpoint_run.py   (run from the repository root, under `uv run --locked python`)

The marker, `filter_1_on`, is a switch on a scale from 0 to 1, and the shipped `_planted` plants
0.0 when the text's value is at the midpoint or above and 1.0 otherwise. Each value below is put
in the start preset and loaded, in a child of its own (midpoint_child.py), under each end the
marker can be planted at. A load that leaves the marker as planted is the shipped rule's verdict
"ignored"; every text here is a good one Vital takes, so the right verdict is always "taken".

Each load is given LIMIT seconds from the child saying it is about to make it (runner.watch).
Throughout, every window on screen is listed 20 times a second (poller.windows) and one owned by
a child is a stop: the child is killed and the run ends. No window is expected: the watch answers
any alert, and none of these texts should raise one.

Wherever the outcome of two neighbouring values differs, the float32s between them are bisected,
by their bits, down to two adjacent float32s, so the boundary is located and not bracketed.
"""

import json
import struct
import sys
import threading
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE))

import poller  # noqa: E402
import runner  # noqa: E402

CHILD = str(HERE / "midpoint_child.py")
LIMIT = 3.0


def f32(value: float) -> float:
    return struct.unpack("<f", struct.pack("<f", value))[0]


def bits(value: float) -> int:
    return struct.unpack("<I", struct.pack("<f", value))[0]


def of_bits(word: int) -> float:
    return struct.unpack("<f", struct.pack("<I", word))[0]


VALUES = [
    0.0,
    0.25,
    0.49,
    of_bits(bits(0.5) - 1),
    0.5,
    of_bits(bits(0.5) + 1),
    0.51,
    0.75,
    0.99,
    1.0,
]


class Stop(Exception):  # noqa: N818  the run's one way to end early
    pass


def run_child(mode: str, value: float) -> dict[str, object]:
    """Run one child to its end under the load limit, watching the screen; its RESULT line."""
    child = runner.Child(mode, value.hex(), script=CHILD)
    seen: list[list[int]] = []
    done = threading.Event()

    def look() -> None:
        while not done.is_set():
            mine = [w for w in poller.windows() if w[0] == child.process.pid]
            if mine:
                seen.extend(mine)
                child.kill()
                return
            time.sleep(poller.PERIOD)

    threading.Thread(target=look, daemon=True).start()
    returned, lines = runner.watch(child, LIMIT, indent="      | ")
    done.set()
    if seen:
        raise Stop(f"A WINDOW APPEARED for the child of {mode} {value!r}: {seen}; it was killed")
    if not returned:
        raise Stop(f"a load of {mode} {value!r} did not return within {LIMIT:g} s")
    if child.process.wait() != 0:
        raise Stop(f"the child of {mode} {value!r} exited {child.process.returncode}")
    results = [line for line in lines if line.startswith("RESULT ")]
    if len(results) != 1:
        raise Stop(f"the child of {mode} {value!r} printed {len(results)} RESULT lines")
    return json.loads(results[0].removeprefix("RESULT "))


def held(found: float | None, value: float) -> str:
    """What a read or the text holds, against the value given as Vital's float32 has it."""
    if found is None:
        return "absent"
    return "as given" if found == f32(value) else repr(found)


def verdicts(result: dict[str, object]) -> dict[float, list[str]]:
    """By plant, the marker's verdict on each load by hand: ignored where the read is the plant."""
    out = {}
    for plant in (0.0, 1.0):
        by_hand = result[f"plant {plant}"]["by_hand"]  # type: ignore[index]
        out[plant] = ["ignored" if load["read"] == plant else "taken" for load in by_hand]
    return out


def outcome(result: dict[str, object]) -> tuple[object, ...]:
    """What is compared between neighbours: the holds and verdicts under each plant."""
    value = float(result["value"])  # type: ignore[arg-type]
    key: list[object] = []
    for plant in (0.0, 1.0):
        part = result[f"plant {plant}"]  # type: ignore[index]
        for load in part["by_hand"]:
            key.append((held(load["read"], value), held(load["text"], value)))
        key.append(tuple(verdicts(result)[plant]))
        key.append(tuple(load["outcome"].split(":")[0] for load in part["shipped_load_state"]))
    return tuple(key)


def show(result: dict[str, object]) -> None:
    value = float(result["value"])  # type: ignore[arg-type]
    print(f"  value {value!r} ({result['value_hex']}), the text holds {result['text_holds']!r}; the shipped rule plants {result['shipped_plant']!r}")
    for plant in (0.0, 1.0):
        part = result[f"plant {plant}"]  # type: ignore[index]
        verdict = verdicts(result)[plant]
        for number, load in enumerate(part["by_hand"], 1):
            print(
                f"    planted {plant}, load {number} by hand: the parameter reads {load['read']!r} "
                f"({held(load['read'], value)}), the state's text holds {load['text']!r} "
                f"({held(load['text'], value)}); the write {load['outcome']}; the marker's verdict "
                f"{verdict[number - 1]}, the right verdict taken"
                + ("" if verdict[number - 1] == "taken" else "  <-- WRONG")
            )
        for number, load in enumerate(part["shipped_load_state"], 1):
            print(
                f"    planted {plant}, load {number} by the shipped load_state, the plant forced: "
                f"{load['outcome']}; reads {load['read']!r}, the text {load['text']!r}"
            )


def main() -> int:
    by_value: dict[float, dict[str, object]] = {}
    instrument: dict[float, dict[str, object]] = {}
    try:
        print("== each listed value, a child for the host and a child for the instrument")
        for value in VALUES:
            print(f"-- {value!r}")
            by_value[value] = run_child("host", value)
            instrument[value] = run_child("instrument", value)
        print()
        print("== the boundaries, bisected over float32 bits where neighbours' outcomes differ")
        listed = sorted(by_value)
        for low, high in zip(listed, listed[1:]):
            if outcome(by_value[low]) == outcome(by_value[high]):
                continue
            lo, hi = bits(f32(low)), bits(f32(high))
            lo_key = outcome(by_value[low])
            print(f"-- between {low!r} and {high!r}: {hi - lo - 1} float32s between")
            while hi - lo > 1:
                mid = (lo + hi) // 2
                value = of_bits(mid)
                print(f"   bisecting at {value!r}")
                found = run_child("host", value)
                by_value[value] = found
                if outcome(found) == lo_key:
                    lo = mid
                else:
                    hi = mid
            print(f"   located: {of_bits(lo)!r} and {of_bits(hi)!r} are adjacent float32s, and their outcomes differ")
    except Stop as stop:
        print(f"STOP: {stop}")
        return 1
    print()
    print("== every value, in order")
    for value in sorted(by_value):
```

```
"""A child of the midpoint probe: one value of the marker's switch, loaded into its own Vital.

usage: midpoint_child.py host <value, float.hex> <the time it was started>
       midpoint_child.py instrument <value, float.hex> <the time it was started>

The text is the start preset with `filter_1_on` set to the value, written by the package's own
writer, as the instrument sends a preset. Before each load it says ABOUT TO LOAD and after it
RETURNED, so its parent kills it when a load does not return in time; it prints one JSON line
of results, prefixed RESULT.

host: for each end the marker can be planted at, 0.0 then 1.0, from the start preset loaded twice:
  by hand, the marker set to that end and the text written, twice, each followed by the value the
  parameter reads and the value the state's text holds; then the shipped `load_state` twice with
  that end forced as the plant, each followed by whether it was taken or refused and the reads.
  Also what the shipped `_planted` would plant for the text.
instrument: a `VitalInstrument` on its own Vital given the text as a whole patch: whether it was
  refused, what the model holds, what `get` returns, and what the plugin reads and holds.
"""

import json
import sys

from patchbay import EventBus, ParameterId, PresetKey

from patchbay_vital import dumps, patch_of
from patchbay_vital import host as host_module
from patchbay_vital.errors import PatchRefusedError, PluginStateError
from patchbay_vital.host import MARKER, VitalHost, _floats
from patchbay_vital.instrument import VitalInstrument
from patchbay_vital.parameters import parameter_map
from patchbay_vital.preset import VitalSettings
from patchbay_vital.start import start_preset
from patchbay_vital.state import wrap

ABOUT_TO_LOAD = "ABOUT TO LOAD"
RETURNED = "RETURNED"

mode, value_hex, _started = sys.argv[1:]
value = float.fromhex(value_hex)
START = start_preset()
values = {**START.settings.values, PresetKey(str(MARKER)): value}
PRESET = START.model_copy(update={"settings": VitalSettings(values=values)})
TEXT = dumps(PRESET).encode("utf-8")
START_TEXT = dumps(START).encode("utf-8")
RECORD = parameter_map().find(ParameterId(str(MARKER)))


def say(line: str) -> None:
    print(line, flush=True)


def reads(host: VitalHost) -> dict[str, float | None]:
    """The marker as the parameter reads it and as the state's text holds it."""
    return {"read": host.get(MARKER), "text": _floats(host.state()).get(str(MARKER))}


def load(call):  # noqa: ANN001, ANN201  a probe's wrapper
    say(ABOUT_TO_LOAD)
    try:
        call()
        outcome = "taken"
    except PluginStateError as error:
        outcome = f"refused: {error.reason}"
    say(RETURNED)
    return outcome


result: dict[str, object] = {"mode": mode, "value": value, "value_hex": value_hex}
if mode == "host":
    host = VitalHost()
    result["text_holds"] = _floats(TEXT).get(str(MARKER))
    result["shipped_plant"] = host_module._planted(RECORD, TEXT)  # noqa: SLF001
    for plant in (0.0, 1.0):
        by_plant: dict[str, object] = {}
        for _ in range(2):
            load(lambda: host.load_state(START_TEXT))
        by_hand = []
        for _ in range(2):
            host.set(MARKER, plant)
            outcome = load(lambda: host._write(wrap(host._containers.with_text(TEXT))))  # noqa: SLF001
            by_hand.append({"outcome": outcome, **reads(host)})
        by_plant["by_hand"] = by_hand
        for _ in range(2):
            load(lambda: host.load_state(START_TEXT))
        original = host_module._planted  # noqa: SLF001
        host_module._planted = lambda _record, _text, end=plant: end  # noqa: SLF001
        shipped = []
        try:
            for _ in range(2):
                outcome = load(lambda: host.load_state(TEXT))
                shipped.append({"outcome": outcome, **reads(host)})
        finally:
            host_module._planted = original  # noqa: SLF001
        by_plant["shipped_load_state"] = shipped
        result[f"plant {plant}"] = by_plant
elif mode == "instrument":
    instrument = VitalInstrument(EventBus())
    patch = patch_of(PRESET)
    result["in_values"] = str(MARKER) in patch.values
    say(ABOUT_TO_LOAD)
    try:
        instrument.load_patch(patch)
        result["outcome"] = "loaded"
    except PatchRefusedError as error:
        result["outcome"] = f"refused: {error.reason}"
    say(RETURNED)
    result["model"] = instrument.preset.settings.values.get(PresetKey(str(MARKER)))
    got = instrument.get(MARKER)
    result["get"] = [got.value, str(got.unit)]
    result.update(reads(instrument.host))
say("RESULT " + json.dumps(result))
```

```
"""How a probe run starts a child, reads it under a time limit, and kills what does not return."""

import queue
import subprocess
import sys
import threading
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
CHILD = str(HERE / "child.py")
ABOUT_TO_LOAD = "ABOUT TO LOAD"
RETURNED = "RETURNED"
START_UP = 60.0
"""Seconds a child is given for everything that is not a load that may block."""


class Child:
    """A child process whose lines are read under a time limit."""

    every: list["Child"] = []
    """Each child a run started, so the run can show at its end that none is left."""

    def __init__(self, *arguments: str, stdin: bool = False, script: str = CHILD) -> None:
        Child.every.append(self)
        self.started = time.time()
        self.killed: float | None = None
        self.ended = False
        """Whether the child closed its output, which a line of None alone does not say."""
        self.process = subprocess.Popen(  # noqa: S603  the probe's own interpreter and script
            [sys.executable, "-u", "-B", script, *arguments, repr(self.started)],
            stdin=subprocess.PIPE if stdin else subprocess.DEVNULL,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
        )
        self._lines: queue.Queue[bytes | None] = queue.Queue()
        threading.Thread(target=self._read, daemon=True).start()

    def _read(self) -> None:
        assert self.process.stdout is not None
        for line in self.process.stdout:
            self._lines.put(line)
        self.ended = True
        self._lines.put(None)

    def line(self, limit: float) -> str | None:
        """The child's next line, or None when it ended or said nothing within the limit."""
        try:
            line = self._lines.get(timeout=limit)
        except queue.Empty:
            return None
        return None if line is None else line.decode("utf-8", "replace").rstrip("\n")

    def kill(self) -> float:
        """Kill the child; return the milliseconds until it was gone."""
        start = time.perf_counter()
        self.killed = time.time()
        self.process.kill()
        self.process.wait()
        return (time.perf_counter() - start) * 1000.0

    def send(self, text: bytes) -> None:
        """Send a kept child one text: an 8-byte length, then its bytes."""
        assert self.process.stdin is not None
        self.process.stdin.write(len(text).to_bytes(8, "big"))
        self.process.stdin.write(text)
        self.process.stdin.flush()

    def close(self) -> int:
        """Close the child's input and wait for it; kill it if it stays."""
        if self.process.stdin is not None:
            self.process.stdin.close()
        try:
            return self.process.wait(timeout=START_UP)
        except subprocess.TimeoutExpired:
            self.kill()
            return self.process.returncode


def watch(child: Child, limit: float, indent: str = "    ") -> tuple[bool, list[str]]:
    """Print a child's lines until it ends; kill it when a load does not return within the limit.

    The child says ABOUT TO LOAD before each load that may block and RETURNED after it. Returns
    whether every load returned, and the lines.
    """
    lines: list[str] = []
    waiting = False
    while True:
        line = child.line(limit if waiting else START_UP)
        if line is None:
            if child.process.poll() is not None and not waiting:
                return True, lines
            if child.process.poll() is not None:
                print(f"{indent}THE CHILD ENDED DURING A LOAD, exit {child.process.returncode}")
                return False, lines
            what = f"within {limit:g} s" if waiting else f"within {START_UP:g} s of starting"
            gone = child.kill()
            print(
                f"{indent}DID NOT RETURN {what}: the child is killed, by design "
                f"(gone {gone:.0f} ms after the kill)"
            )
            return False, lines
        if line.startswith(ABOUT_TO_LOAD):
            waiting = True
        elif line.startswith(RETURNED):
            waiting = False
        lines.append(line)
        print(f"{indent}{line}", flush=True)


def memory(pid: int) -> float:
    """A process's resident memory in MiB, as ps gives it."""
    done = subprocess.run(  # noqa: S603
        ["/bin/ps", "-o", "rss=", "-p", str(pid)], capture_output=True, text=True, check=False
    )
    return int(done.stdout.strip() or 0) / 1024.0
```

```
"""A second process that watches the screen while a probe run's children load.

usage: poller.py <out file> [new]

With `new`, only the windows of processes that owned none at the first poll are written: those
of a run's children, and not those of what was on screen before it.

20 times a second it writes one JSON line: the time, the front application's pid (lsappinfo),
and every window on screen as [owner pid, layer, width, height] (CGWindowListCopyWindowInfo,
which gives a window's owner and bounds without the Screen Recording permission). It ends when
its input closes. It loads no plugin and opens no window.
"""

import ctypes
import ctypes.util
import json
import subprocess
import sys
import threading
import time

_cg = ctypes.CDLL(ctypes.util.find_library("CoreGraphics"))
_cf = ctypes.CDLL(ctypes.util.find_library("CoreFoundation"))

_cg.CGWindowListCopyWindowInfo.restype = ctypes.c_void_p
_cg.CGWindowListCopyWindowInfo.argtypes = [ctypes.c_uint32, ctypes.c_uint32]
_cf.CFArrayGetCount.restype = ctypes.c_long
_cf.CFArrayGetCount.argtypes = [ctypes.c_void_p]
_cf.CFArrayGetValueAtIndex.restype = ctypes.c_void_p
_cf.CFArrayGetValueAtIndex.argtypes = [ctypes.c_void_p, ctypes.c_long]
_cf.CFDictionaryGetValue.restype = ctypes.c_void_p
_cf.CFDictionaryGetValue.argtypes = [ctypes.c_void_p, ctypes.c_void_p]
_cf.CFNumberGetValue.restype = ctypes.c_bool
_cf.CFNumberGetValue.argtypes = [ctypes.c_void_p, ctypes.c_int, ctypes.c_void_p]
_cf.CFRelease.argtypes = [ctypes.c_void_p]


class _Rect(ctypes.Structure):
    _fields_ = (("x", ctypes.c_double), ("y", ctypes.c_double), ("w", ctypes.c_double), ("h", ctypes.c_double))


_cg.CGRectMakeWithDictionaryRepresentation.restype = ctypes.c_bool
_cg.CGRectMakeWithDictionaryRepresentation.argtypes = [ctypes.c_void_p, ctypes.POINTER(_Rect)]

ON_SCREEN_ONLY = 1
LONG_LONG = 11
"""kCFNumberLongLongType."""
PERIOD = 0.05


def _key(name: str) -> int:
    return int(ctypes.c_void_p.in_dll(_cg, name).value or 0)


OWNER, LAYER, BOUNDS = _key("kCGWindowOwnerPID"), _key("kCGWindowLayer"), _key("kCGWindowBounds")


def _number(dictionary: int, key: int) -> int:
    value = ctypes.c_longlong(0)
    number = _cf.CFDictionaryGetValue(dictionary, key)
    if number:
        _cf.CFNumberGetValue(number, LONG_LONG, ctypes.byref(value))
    return int(value.value)


def windows(everything: bool = False) -> list[list[int]]:
    """Every window on screen, or every window there is: owner's pid, layer, width, height."""
    found = []
    array = _cg.CGWindowListCopyWindowInfo(0 if everything else ON_SCREEN_ONLY, 0)
    if not array:
        return found
    for index in range(_cf.CFArrayGetCount(array)):
        entry = _cf.CFArrayGetValueAtIndex(array, index)
        rect = _Rect()
        bounds = _cf.CFDictionaryGetValue(entry, BOUNDS)
        if bounds:
            _cg.CGRectMakeWithDictionaryRepresentation(bounds, ctypes.byref(rect))
        found.append([_number(entry, OWNER), _number(entry, LAYER), round(rect.w), round(rect.h)])
    _cf.CFRelease(array)
    return found


def front() -> int | None:
    """The front application's pid, as lsappinfo gives it."""
    asn = subprocess.run(["/usr/bin/lsappinfo", "front"], capture_output=True, text=True, check=False).stdout.strip()  # noqa: S603
    if not asn:
        return None
    line = subprocess.run(["/usr/bin/lsappinfo", "info", "-only", "pid", asn], capture_output=True, text=True, check=False).stdout  # noqa: S603
    digits = line.strip().rpartition("=")[2]
    return int(digits) if digits.isdigit() else None


def main(out: str, new_only: bool = False) -> None:
    known: set[int] | None = None
    done = threading.Event()
    threading.Thread(target=lambda: (sys.stdin.read(), done.set()), daemon=True).start()
    with open(out, "w", encoding="utf-8") as log:  # noqa: PTH123
        print("ready", flush=True)
        while not done.is_set():
            started = time.time()
            found = windows()
            if new_only:
                if known is None:
                    known = {window[0] for window in found}
                found = [window for window in found if window[0] not in known]
            log.write(json.dumps({"t": started, "front": front(), "windows": found}) + "\n")
            log.flush()
            time.sleep(max(0.0, PERIOD - (time.time() - started)))


if __name__ == "__main__":
    main(sys.argv[1], new_only=sys.argv[2:] == ["new"])
```
