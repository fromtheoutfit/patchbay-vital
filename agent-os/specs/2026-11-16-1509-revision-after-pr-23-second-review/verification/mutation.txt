def run(tests: list[str]) -> tuple[list[tuple[str, str]], str]:
    """Run pytest in the copy; return the failed tests with their reasons, and the last line."""
    command = ["uv", "run", "--locked", "pytest", "-c", "pyproject.toml", "-p", "no:doctest", "-p", "no:cacheprovider", "-rfE", "--no-header", *tests]
    # wide, so that pytest's summary line has room for the reason after a long test name
    wide = {**os.environ, "COLUMNS": "400"}
    done = subprocess.run(command, cwd=COPY, capture_output=True, text=True, check=False, env=wide)
    # a parametrized test's id can hold spaces, so a name runs to the " - " before its reason
    found = re.findall(r"^(?:FAILED|ERROR) (.+?)(?: - (.*))?$", done.stdout, flags=re.MULTILINE)
    tail = done.stdout.strip().splitlines()[-1] if done.stdout.strip() else done.stderr.strip()[-200:]
    return [(name.split("::", 1)[1] if "::" in name else name, reason) for name, reason in found], f"({tail})"

"""The sample test's old form given the new form's sample: a throwaway, in the scratch copy alone.

The test as at 64863c6 draws 25 uniform rounds and no bound, so a mutant that differs only at a
bound passes it for want of the display. This module gives the old form, a load and then a set of
each display, the new form's 27 rounds, to tell the form from the sample. `mutate.sh` writes it
into the copy as tests/unit/test_old_form_on_the_new_sample.py; it is no part of the tree. It
takes the fake plugin and its fixtures from the tree's tests/unit/test_instrument.py.
"""

import numpy as np
from patchbay import Patch
from test_instrument import (  # the fixtures are taken by name, for pytest to find them here
    _FakeVital,
    _model_floats,
    _quantity,
    bus,
    host,
    instrument,
    plugin,
)

from patchbay_vital.instrument import VitalInstrument
from patchbay_vital.parameters import MappedParameter, parameter_map

__all__ = ["bus", "host", "instrument", "plugin"]


def _rounds() -> list[dict[str, object]]:
    """Return the new form's sample: 25 uniform rounds, seed 23, then each bound, as quantities."""
    groups: dict[str, MappedParameter] = {}
    for record in parameter_map().parameters:
        if record.parameter.kind == "continuous":
            groups.setdefault(record.scale.model_dump_json() + record.curve.model_dump_json(), record)
    rng = np.random.default_rng(23)
    rounds = [
        {r.id: float(rng.uniform(r.parameter.minimum, r.parameter.maximum)) for r in groups.values()}
        for _ in range(25)
    ]
    rounds.append({r.id: r.parameter.minimum for r in groups.values()})
    rounds.append({r.id: r.parameter.maximum for r in groups.values()})
    units = {r.id: r.parameter.unit for r in groups.values()}
    return [{key: _quantity(display, units[key]) for key, display in found.items()} for found in rounds]


def test_the_old_form_on_the_new_sample_a_set_after_the_load_agrees_with_the_load(
    instrument: VitalInstrument,
) -> None:
    """The old form's one assertion, over the new sample's 1,485 displays."""
    compared = 0
    for values in _rounds():
        instrument.load_patch(Patch(instrument="Vital", values=values))
        loaded = _model_floats(instrument.preset)
        for parameter_id, value in values.items():
            instrument.set(parameter_id, value)
            assert _model_floats(instrument.preset)[parameter_id] == loaded[parameter_id], value
            compared += 1
    assert compared == 1485


def test_the_old_form_on_the_new_sample_every_set_after_the_load_reaches_the_plugin(
    instrument: VitalInstrument, plugin: _FakeVital
) -> None:
    """What the old form never asked: was the set after the load a set, or the keep rule's no-op."""
    kept = []
    for values in _rounds():
        instrument.load_patch(Patch(instrument="Vital", values=values))
        for parameter_id, value in values.items():
            sent = len(plugin.sets)
            instrument.set(parameter_id, value)
            if len(plugin.sets) == sent:
                kept.append((parameter_id, value.value))
    assert kept == [], f"{len(kept)} set(s) after the load called no plugin: {kept}"
