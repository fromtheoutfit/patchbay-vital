"""What a set of the value `get` shows does to each float Vital upgrades, counted on Vital.

For each line of `tests/data/upgraded-floats.txt`, on its preset as `load_patch` loads it: the
model's float (the file's, or the parameter's default for a key the file lacks), the float Vital
holds before the set, what `set(id, get(id))` does, and the float Vital holds after it. Vital's
floats are read through the diagnostic read, `instrument.host.state()`. An outcome is one of:

- mended: the set reaches Vital, which then holds the model's float;
- refused: the set raises ParameterRangeError, the model and Vital unchanged;
- kept: the set changes nothing, the model the same object, and Vital holds what it held.

A count over the 46 recorded floats, not a sample. Run from the repository root with the locked
interpreter. Each preset is loaded once, on one instrument over one Vital.
"""

import importlib.metadata
import platform
from pathlib import Path

from patchbay import EventBus, ParameterId

from patchbay_vital import loads
from patchbay_vital.errors import ParameterRangeError
from patchbay_vital.format import split_modulations
from patchbay_vital.host import DEFAULT_PATH, VitalHost, _floats
from patchbay_vital.instrument import VitalInstrument
from patchbay_vital.parameters import parameter_map
from patchbay_vital.patch import patch_of

PRESETS = Path.home() / "Music" / "Vital"
RECORDED = Path("tests") / "data" / "upgraded-floats.txt"


def model_floats(instrument: VitalInstrument) -> dict[str, float]:
    """Return what the model says each parameter holds: its float, or the default without one."""
    preset = instrument.preset
    _, slots = split_modulations(preset.modulations)
    held = {str(key): value for key, value in (*preset.settings.values.items(), *slots.items())}
    return {r.preset_key: held.get(r.preset_key, r.preset_default) for r in parameter_map().parameters}


rows: dict[str, list[tuple[str, str, float]]] = {}
for line in RECORDED.read_text(encoding="utf-8").splitlines():
    if line.startswith("#"):
        continue
    preset_name, key, written, vital_holds = line.split("\t")
    rows.setdefault(preset_name, []).append((key, written, float(vital_holds)))

host = VitalHost()
instrument = VitalInstrument(EventBus(), host)
print(f"Vital {host.version} at {DEFAULT_PATH}, through pedalboard {importlib.metadata.version('pedalboard')}")
print(f"machine {platform.machine()}, Python {platform.python_version()}")
print(f"{RECORDED}: {sum(len(found) for found in rows.values())} floats on {len(rows)} presets, read from ~/Music/Vital")
print()
print("preset | key | in the file | the model's float | Vital before | the set | Vital after | the model after")
counts = {"mended": 0, "refused": 0, "kept": 0}
other: list[str] = []
for preset_name, found in rows.items():
    text = (PRESETS / preset_name).read_bytes().decode("utf-8")
    instrument.load_patch(patch_of(loads(text)))
    for key, written, recorded in found:
        before = instrument.preset
        model = model_floats(instrument)[key]
        vital_before = _floats(host.state())[key]
        shown = instrument.get(ParameterId(key))
        try:
            instrument.set(ParameterId(key), shown)
        except ParameterRangeError as error:
            outcome = f"refused: {error.given!r} is outside {error.minimum!r} to {error.maximum!r}"
            kind = "refused" if instrument.preset is before else "other"
        else:
            kind = ""
            outcome = ""
        vital_after = _floats(host.state())[key]
        model_after = model_floats(instrument)[key]
        if not kind:
            if vital_after == model_after:
                kind, outcome = "mended", "mended: it reaches Vital"
            elif instrument.preset is before and vital_after == vital_before:
                kind, outcome = "kept", "kept: the model the same object, Vital as before"
            else:
                kind, outcome = "other", "OTHER"
        if kind == "refused" and vital_after != vital_before:
            kind = "other"
        if vital_before != recorded:
            kind = "other"
            outcome += f" (VITAL HELD {vital_before!r} BEFORE, NOT THE RECORDED {recorded!r})"
        if kind == "other":
            other.append(f"{preset_name} {key}")
        else:
            counts[kind] += 1
        name = preset_name.rsplit("/", 1)[-1].removesuffix(".vital")
        print(f"{name} | {key} | {written} | {model!r} | {vital_before!r} | {outcome} | {vital_after!r} | {model_after!r}")
print()
print(f"mended {counts['mended']}, refused {counts['refused']}, kept {counts['kept']}, of {sum(len(found) for found in rows.values())}")
print(f"none of the three: {len(other)} {other if other else ''}".rstrip())
