"""Probe: notes of one pitch on channels of their own, with Vital's MPE setting off and on.

The bytes the rule proposed for finding 5 would send: a note struck while its pitch already
sounds on its channel goes out on the lowest channel where that pitch does not. Each case below
gives the channels explicitly, so `VitalInstrument.render` as it stands sends exactly those bytes,
and the one-channel form beside it is today's.

On the start preset at 48 kHz, 120 bpm (a beat is 0.5 s), two bars. Each render's envelope is its
RMS under a Hann window of `WINDOW` s every `HOP` s, both channels; a level is the envelope's
median over a span; an end is the last window centre at or above `END_FRACTION` of the quietest
level of a note alone in the same setting (each single's level over the span it alone sounds). A centroid is taken over `HELD` under one Hann window, both channels summed, as
the instrument spec's centroid test takes it. Every span starts past the opening, over by 0.102 s.

Three settings: as the start preset holds them; `mpe_enabled` on (1); and `velocity_track` at
100 % with MPE off, so velocity sets level and two notes on one sample can be told apart by it.

Run from the repository root with the locked interpreter, Vital installed.
"""

import numpy as np
import pedalboard
from patchbay import EventBus, Note, ParameterId, Pattern, Quantity, RenderRequest, Unit

from patchbay_vital.host import DEFAULT_PATH
from patchbay_vital.instrument import VitalInstrument
from patchbay_vital.midi import midi_of
from patchbay_vital.parameters import parameter_map

RATE = 48_000
WINDOW = 0.02  # s
HOP = 0.005  # s
END_FRACTION = 0.5
HELD = (0.65, 0.95)  # s: B alone below sounds from 0.5 s, so this is its steady part
REQUEST = RenderRequest(sample_rate=RATE, tail_seconds=0.0)
SPANS = ((0.15, 0.45), (0.55, 0.95), (1.05, 1.45), (1.55, 1.95), (2.05, 2.45), (2.55, 2.95))


def pattern(*notes: tuple[int, int, float, float, int]) -> Pattern:
    """Two bars of four beats, from (pitch, velocity, start, duration, channel)."""
    return Pattern(
        tempo_bpm=120.0,
        bars=2,
        beats_per_bar=4,
        notes=tuple(
            Note(pitch=p, velocity=v, start_beats=s, duration_beats=d, channel=c)
            for p, v, s, d, c in notes
        ),
    )


def envelope(audio: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    size, hop = round(WINDOW * RATE), round(HOP * RATE)
    window = np.hanning(size)
    norm = np.sqrt(np.mean(window**2))
    starts = np.arange(0, audio.shape[1] - size, hop)
    values = np.array(
        [np.sqrt(np.mean((audio[:, s : s + size] * window) ** 2)) / norm for s in starts]
    )
    return (starts + size / 2) / RATE, values


def level(times: np.ndarray, values: np.ndarray, start: float, end: float) -> float:
    inside = (times >= start) & (times <= end)
    return float(np.median(values[inside]))


def ended(times: np.ndarray, values: np.ndarray, reference: float) -> float:
    loud = np.nonzero(values >= END_FRACTION * reference)[0]
    return float(times[loud[-1]]) if loud.size else float("nan")


def centroid(audio: np.ndarray) -> float:
    start, end = (round(t * RATE) for t in HELD)
    mono = audio[:, start:end].sum(axis=0) * np.hanning(end - start)
    power = np.abs(np.fft.rfft(mono)) ** 2
    frequencies = np.fft.rfftfreq(end - start, 1 / RATE)
    total = float(power.sum())
    return float((frequencies * power).sum() / total) if total > 0 else float("nan")


CASES = {
    "B alone, channel 0: 60 beats 1 to 3": pattern((60, 100, 1.0, 2.0, 0)),
    "B alone, channel 1": pattern((60, 100, 1.0, 2.0, 1)),
    "B alone, channel 2": pattern((60, 100, 1.0, 2.0, 2)),
    "three of 60 overlapping on channel 0 (today): A beats 0-3, B 1-4, C 2-5": pattern(
        (60, 100, 0.0, 3.0, 0), (60, 100, 1.0, 3.0, 0), (60, 100, 2.0, 3.0, 0)
    ),
    "three of 60 overlapping on channels 0, 1, 2 (the rule's)": pattern(

"""Each note alone, with the span it sounds over steadily."""

SETTINGS = {
    "as the start preset holds them (mpe_enabled off, velocity_track 0 %)": {},
    "mpe_enabled on": {"mpe_enabled": Quantity(value=1.0, unit=Unit.none)},
    "velocity_track 100 %, mpe_enabled off": {
        "velocity_track": Quantity(value=100.0, unit=Unit.percent)
    },
}


def main() -> None:
    found = parameter_map()
    print(f"Vital {found.vital_version} at {DEFAULT_PATH}, through pedalboard {pedalboard.__version__}")
    print(f"machine {found.machine}, numpy {np.__version__}")
    print(
        f"window: Hann {WINDOW} s every {HOP} s; an end is the last centre at or above "
        f"{END_FRACTION} of one note's level; centroid over {HELD[0]} to {HELD[1]} s"
    )
    for setting, values in SETTINGS.items():
        instrument = VitalInstrument(EventBus())
        for key, value in values.items():
            instrument.set(ParameterId(key), value)
            print(f"\n#### {setting}: {key} set to {instrument.get(ParameterId(key))!r}")
        if not values:
            print(f"\n#### {setting}")
        renders = {}
        for name, case in CASES.items():
            audio = np.asarray(instrument.render(case, REQUEST).audio, dtype=np.float64)
            renders[name] = (envelope(audio), centroid(audio))
            print(f"\n== {name}")
            for message, time in midi_of(case, REQUEST):
                print(f"  midi {message.hex(' ')} at sample {round(time * RATE - 0.5)}")
            (times, env), cen = renders[name]
            spans = ", ".join(f"{a:.2f}-{b:.2f} s {level(times, env, a, b):.6f}" for a, b in SPANS)
            print(f"  levels: {spans}")
            print(f"  centroid {cen:.2f} Hz")
        singles = {name: span for name, span in SINGLES.items()}
        one = min(level(*renders[name][0], *span) for name, span in singles.items())
        print(f"\n== ends, against the quietest note alone in this setting, {one:.6f}")
        for name, ((times, env), _) in renders.items():
            print(f"  {ended(times, env, one):.3f} s  {name}")


if __name__ == "__main__":
    main()
