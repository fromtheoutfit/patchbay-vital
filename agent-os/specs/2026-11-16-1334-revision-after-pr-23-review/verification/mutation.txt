#!/usr/bin/env bash
# usage: mutate.sh   (from the repository root): copy what git lists into a scratch directory, sync it, run the mutants there.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
COPY="$R/mut/patchbay-vital"
rm -rf "$R/mut"; mkdir -p "$COPY"
git ls-files -z --cached --others --exclude-standard | while IFS= read -r -d '' f; do
  [ -e "$f" ] || continue
  mkdir -p "$COPY/$(dirname "$f")" && cp -p "$f" "$COPY/$f"
done
echo "the scratch copy holds $(find "$COPY" -type f | wc -l | tr -d ' ') files, what git lists of the tree; it is outside the tree and nothing in the tree is changed"
(cd "$COPY" && uv sync --locked 2>&1 | tail -2)
uv run --locked python -u -B "$R/mutate.py" "$COPY"
echo "exit=$?"

def run(tests: list[str]) -> list[str]:
    command = ["uv", "run", "--locked", "pytest", "-c", "pyproject.toml", "-p", "no:doctest", "-p", "no:cacheprovider", "-rfE", "--no-header", *tests]
    done = subprocess.run(command, cwd=COPY, capture_output=True, text=True, check=False)
    # a parametrized test's id can hold spaces, so a name runs to the " - " before its reason
    failed = re.findall(r"^(?:FAILED|ERROR) (.+?)(?: - .*)?$", done.stdout, flags=re.MULTILINE)
    tail = done.stdout.strip().splitlines()[-1] if done.stdout.strip() else done.stderr.strip()[-200:]
    return [name.split("::", 1)[1] if "::" in name else name for name in failed] + [f"({tail})"]

    if text.count(old) != 1:
        print(f"{name}: THE TEXT TO REPLACE IS THERE {text.count(old)} TIMES, not once: {old!r}")

    path.write_bytes(kept)
    caught = any(must_fail in test for test in failed)

    print(f"{name}: {what}")

    print(f"    must fail {must_fail}: {'yes' if caught else 'NO, SURVIVED'}; {len(failed)} failed {tail}")
