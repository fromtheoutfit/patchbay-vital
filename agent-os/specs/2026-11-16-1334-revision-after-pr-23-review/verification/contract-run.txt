"""Core's thirteen contract tests against Vital, a fresh VitalHost for each: not the gate's way.

A throwaway invocation for the record: the gate shares one Vital a module. Run from the repository
root with the locked interpreter.
"""

import importlib.metadata
import json
import platform
import time

from patchbay import EventBus
from patchbay.testing import CONTRACT_TESTS

from patchbay_vital.host import DEFAULT_PATH, VitalHost
from patchbay_vital.instrument import VitalInstrument

core = importlib.metadata.distribution("patchbay-core")
source = json.loads(core.read_text("direct_url.json") or "{}")
print(f"patchbay-core {core.version} at {source.get('vcs_info', {}).get('commit_id')} ({source.get('url')})")
print(f"pedalboard {importlib.metadata.version('pedalboard')}, machine {platform.machine()}, Python {platform.python_version()}")
passed = 0
for test in CONTRACT_TESTS:
    began = time.perf_counter()
    host = VitalHost()
    loaded = time.perf_counter()
    bus = EventBus()
    instrument = VitalInstrument(bus, host)
    test(instrument, bus)
    done = time.perf_counter()
    passed += 1
    print(f"PASSED  {test.__name__}  (Vital {host.version} at {DEFAULT_PATH}: loaded in {loaded - began:.2f} s, the test {done - loaded:.2f} s)")
    del instrument, host
print(f"{passed} passed of {len(CONTRACT_TESTS)}, each on a VitalHost of its own")
