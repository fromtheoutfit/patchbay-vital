"""What the guard's second decode costs, beside the load it guards (PR 23's finding 7).

On the largest installed preset the manifest lists, by bytes, and on the start preset: the time
of `check_loadable` (its decode of `wavetables`, `lfos` and `sample` among it), of the decode alone
of those three sections, and of the two loads `load_patch` makes through the host. Each the best
of `REPEATS` by `time.perf_counter`, the loads on the installed Vital.

Run from the repository root with the locked interpreter, Vital and its presets installed.
"""

import time
from pathlib import Path

from patchbay import PresetKey

from patchbay_vital import check_loadable, dumps, loads, parameter_map, start_preset
from patchbay_vital.host import VitalHost
from patchbay_vital.instrument import LOADS
from patchbay_vital.preset import DECODER

REPEATS = 5
PRESETS = Path.home() / "Music" / "Vital"
MANIFEST = [
    line
    for line in Path("tests/data/factory-manifest.txt").read_text(encoding="utf-8").splitlines()
    if not line.startswith("#")
]


def best(call) -> float:
    times = []
    for _ in range(REPEATS):
        started = time.perf_counter()
        call()
        times.append(time.perf_counter() - started)
    return min(times)


found = parameter_map()
host = VitalHost()
largest = max(MANIFEST, key=lambda name: (PRESETS / name).stat().st_size)
for name, preset in (
    ("the start preset", start_preset()),
    (largest, loads((PRESETS / largest).read_text(encoding="utf-8"))),
):
    text = dumps(preset).encode("utf-8")
    sections = [preset.opaque[PresetKey(key)] for key in ("wavetables", "lfos", "sample")]
    guard = best(lambda: check_loadable(preset, found))
    decode = best(lambda: [DECODER.decode(raw.decode("utf-8")) for raw in sections])
    load = best(lambda: [host.load_state(text) for _ in range(LOADS)])
    print(f"{name}: {len(text):,} bytes of text, the three sections {sum(map(len, sections)):,}")
    print(f"  check_loadable {guard * 1000:.2f} ms, of which the sections' decode {decode * 1000:.2f} ms")
    print(f"  the {LOADS} loads {load * 1000:.2f} ms; the guard is {guard / load:.1%} of them")
