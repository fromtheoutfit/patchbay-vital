"""A labelled sample: a load of an arbitrary display and a set of it leave the same float.

One record of each continuous group of the map, a group being a scale and a curve, linear and
indexed alike. For each, `SIZE` displays drawn by `numpy.random.default_rng(SEED).uniform` over
the parameter's bounds, in the groups' order and then the draws'. A load is `preset_of` of a patch
of values alone naming the display; a set is the float `VitalInstrument.set` stores,
`MappedParameter.to_preset` of the display, which the unit tests hold the set to. Each is compared
exactly, the sign of a zero included. A display that names its own float, the one case where the
two differ, is counted apart: a draw that hits one is not a disagreement but is listed.

A sample, not a count: it says nothing of displays not drawn. No plugin is used.
"""

import math

import numpy as np
from patchbay import ParameterId, Patch, Quantity

from patchbay_vital import parameter_map, preset_of
from patchbay_vital.format import split_modulations

SEED = 20261116
SIZE = 2_000

found = parameter_map()
groups: dict[str, object] = {}
for record in found.parameters:
    if record.parameter.kind == "continuous":
        groups.setdefault(record.scale.model_dump_json() + record.curve.model_dump_json(), record)
print(f"{len(groups)} continuous groups, {SIZE} displays each, seed {SEED}, numpy {np.__version__}")
rng = np.random.default_rng(SEED)
drawn = agree = named = 0
disagree: list[str] = []
for record in groups.values():
    for _ in range(SIZE):
        display = float(rng.uniform(record.parameter.minimum, record.parameter.maximum))
        value = Quantity(value=display, unit=record.parameter.unit)
        preset = preset_of(Patch(instrument="Vital", values={ParameterId(record.id): value}))
        _, slots = split_modulations(preset.modulations)
        loaded = {**preset.settings.values, **slots}[record.preset_key]
        held = record.to_preset(display)
        drawn += 1
        if loaded == held and math.copysign(1.0, loaded) == math.copysign(1.0, held):
            agree += 1
        elif record.to_display(loaded) == display:
            named += 1
            disagree.append(f"  named: {record.id} {display!r} loads {loaded!r}, a set holds {held!r}")
        else:
            disagree.append(f"  DISAGREE: {record.id} {display!r} loads {loaded!r}, a set {held!r}")
print(f"drawn {drawn}: the same float {agree}; a display naming its own float {named}")
print(f"other disagreements: {drawn - agree - named}")
for line in disagree:
    print(line)
for record in groups.values():
    print(f"  group of {record.id}: {record.scale.kind} scale {record.scale.minimum!r} to "
          f"{record.scale.maximum!r}, {record.curve.kind} curve")
