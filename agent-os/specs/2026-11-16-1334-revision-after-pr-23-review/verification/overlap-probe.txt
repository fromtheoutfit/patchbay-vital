"""Probe: what Vital makes of two notes of one pitch that overlap, with the instrument's bytes today.

On the start preset at 48 kHz, 120 bpm (a beat is 0.5 s), a render of two bars, each pattern
rendered by `VitalInstrument.render` as it stands, so the MIDI is `midi_of`'s. Each render's
envelope is its RMS under a Hann window of `WINDOW` s every `HOP` s, both channels. A note's level
is the envelope's median over a span where it alone sounds; a render's end is the last window
centre where the envelope is at or above `END_FRACTION` of the level. The opening transient of
the start preset is over by 0.102 s (the instrument spec's probes.txt), so every span starts past
`OPENING`.

Cases: each note alone, as references; two notes of one pitch overlapping (A beats 0 to 2, B 1
to 3); the same pitch struck on the same sample (lengths 1 and 3 beats, velocities 100 and 50,
in both orders); a control of two pitches overlapping; and the same overlap on two channels.

Run from the repository root with the locked interpreter, Vital installed.
"""

import numpy as np
import pedalboard
from patchbay import EventBus, Note, Pattern, RenderRequest

from patchbay_vital.host import DEFAULT_PATH
from patchbay_vital.instrument import VitalInstrument
from patchbay_vital.midi import midi_of
from patchbay_vital.parameters import parameter_map

RATE = 48_000
WINDOW = 0.02  # s, the Hann window's length
HOP = 0.005  # s, between window centres
END_FRACTION = 0.5  # of a note's level: where a render counts as having ended
OPENING = 0.15  # s: every span measured starts past the start preset's opening
BEAT = 0.5  # s at 120 bpm
REQUEST = RenderRequest(sample_rate=RATE, tail_seconds=0.0)


def pattern(*notes: tuple[int, int, float, float, int]) -> Pattern:
    """A pattern of two bars of four beats, from (pitch, velocity, start, duration, channel)."""
    return Pattern(
        tempo_bpm=120.0,
        bars=2,
        beats_per_bar=4,
        notes=tuple(
            Note(pitch=p, velocity=v, start_beats=s, duration_beats=d, channel=c)
            for p, v, s, d, c in notes
        ),
    )


def envelope(audio: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """Return window centres in seconds and the RMS under a Hann window at each."""
    size, hop = round(WINDOW * RATE), round(HOP * RATE)
    window = np.hanning(size)
    norm = np.sqrt(np.mean(window**2))
    starts = np.arange(0, audio.shape[1] - size, hop)
    values = np.array(
        [np.sqrt(np.mean((audio[:, s : s + size] * window) ** 2)) / norm for s in starts]
    )
    return (starts + size / 2) / RATE, values


def level(times: np.ndarray, values: np.ndarray, start: float, end: float) -> float:
    """The envelope's median between two times."""
    inside = (times >= start) & (times <= end)
    return float(np.median(values[inside]))


def ended(times: np.ndarray, values: np.ndarray, reference: float) -> float:
    """The last window centre at or above END_FRACTION of a level."""
    loud = np.nonzero(values >= END_FRACTION * reference)[0]
    return float(times[loud[-1]]) if loud.size else float("nan")


CASES = {
    "A alone: pitch 60, velocity 100, beats 0 to 2, channel 0": pattern((60, 100, 0.0, 2.0, 0)),
    "B alone: pitch 60, velocity 100, beats 1 to 3, channel 0": pattern((60, 100, 1.0, 2.0, 0)),
    "same pitch overlapping: A 60 beats 0 to 2, B 60 beats 1 to 3, channel 0": pattern(
        (60, 100, 0.0, 2.0, 0), (60, 100, 1.0, 2.0, 0)
    ),
    "same pitch, same sample: 60 velocity 100 for 1 beat, then 60 velocity 50 for 3 beats": pattern(
        (60, 100, 0.0, 1.0, 0), (60, 50, 0.0, 3.0, 0)
    ),
    "same pitch, same sample: 60 velocity 50 for 3 beats, then 60 velocity 100 for 1 beat": pattern(
        (60, 50, 0.0, 3.0, 0), (60, 100, 0.0, 1.0, 0)
    ),
    "60 velocity 100 alone for 1 beat": pattern((60, 100, 0.0, 1.0, 0)),
    "60 velocity 50 alone for 3 beats": pattern((60, 50, 0.0, 3.0, 0)),
    "control, two pitches: A 60 beats 0 to 2, B 67 beats 1 to 3, channel 0": pattern(
        (60, 100, 0.0, 2.0, 0), (67, 100, 1.0, 2.0, 0)
    ),
    "67 alone: beats 1 to 3": pattern((67, 100, 1.0, 2.0, 0)),
    "same pitch on two channels: A 60 ch 0 beats 0 to 2, B 60 ch 1 beats 1 to 3": pattern(
        (60, 100, 0.0, 2.0, 0), (60, 100, 1.0, 2.0, 1)
    ),
}


def main() -> None:
    found = parameter_map()
    print(f"Vital {found.vital_version} at {DEFAULT_PATH}, through pedalboard {pedalboard.__version__}")
    print(f"machine {found.machine}, numpy {np.__version__}")
    print(
        f"window: Hann {WINDOW} s every {HOP} s, both channels; an end is the last centre at or "
        f"above {END_FRACTION} of the level; spans start past {OPENING} s; a beat is {BEAT} s"
    )
    instrument = VitalInstrument(EventBus())
    renders = {}
    for name, case in CASES.items():
        audio = instrument.render(case, REQUEST).audio
        times, values = envelope(np.asarray(audio, dtype=np.float64))
        renders[name] = (times, values)
        print()
        print(f"== {name}")
        for message, time in midi_of(case, REQUEST):
            print(f"  midi {message.hex(' ')} at {time:.9f} s (sample {round(time * RATE - 0.5)})")
        print(f"  peak {float(np.max(np.abs(audio))):.4f}")
        for start, end in ((OPENING, 0.45), (0.55, 0.95), (1.05, 1.45), (1.55, 1.95)):
            print(f"  level {start:.2f} to {end:.2f} s: {level(times, values, start, end):.6f}")
    reference = level(*renders["A alone: pitch 60, velocity 100, beats 0 to 2, channel 0"], OPENING, 0.95)
    print()
    print(f"== ends, against the level of A alone from {OPENING} to 0.95 s, {reference:.6f}")
    for name, (times, values) in renders.items():
        print(f"  {ended(times, values, reference):.3f} s  {name}")


if __name__ == "__main__":
    main()
