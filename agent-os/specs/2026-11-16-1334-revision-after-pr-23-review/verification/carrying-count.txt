"""The carrying, counted again under PR 23's load rule: every float of every preset on hand.

For the start preset, the fixture and the 75 installed presets the manifest lists: how many floats
`patch_of` carries in `values` and in the preset entry; how many of those in `values` a load of
the patch restores exactly, by comparing `preset_of(patch)`'s float with the preset's own, the sign
of a zero included; and the text byte for byte through a patch and through the patch's JSON.

Beside it, the rule before this revision, restated here: on a linear scale the curve's float32
for the display, kept within the scale's ends, and on an indexed scale `to_preset`. A float in
`values` under that rule and not under this one would fail the revision's stop; one in `values`
under this one and not under that is listed by preset and key.

Then `PatchLoaded.parameter_count` for each form of patch, from a real `VitalInstrument` over the
installed Vital: a whole patch of the start preset, a patch of values alone, and a whole patch of a
preset carrying floats in the entry.

Run from the repository root with the locked interpreter, Vital and its presets installed.
"""

import json
import math
from pathlib import Path

from patchbay import EventBus, ParameterId, Patch, PatchLoaded, Quantity, Unit
from patchbay.testing import RecordingSubscriber

from patchbay_vital import dumps, loads, parameter_map, patch_of, preset_of, start_preset
from patchbay_vital.format import split_modulations
from patchbay_vital.instrument import VitalInstrument
from patchbay_vital.patch import PRESET_ENTRY

PRESETS = Path.home() / "Music" / "Vital"
FIXTURE = Path("tests/data/init-media-replaced.vital")
MANIFEST = [
    line
    for line in Path("tests/data/factory-manifest.txt").read_text(encoding="utf-8").splitlines()
    if not line.startswith("#")
]
found = parameter_map()
by_id = {record.id: record for record in found.parameters}


def same(one: float, other: float) -> bool:
    return one == other and math.copysign(1.0, one) == math.copysign(1.0, other)


def old_back(record, display: float) -> float:
    """The rule before this revision, as 63ad5ae's `_float_of` had it."""
    if record.scale.kind == "indexed":
        return record.to_preset(display)
    return min(max(record.curve.to_preset(display), record.scale.minimum), record.scale.maximum)


def old_carries(key: str, held: float) -> bool:
    record = by_id.get(key)
    if record is None:
        return False
    try:
        value = Quantity(value=record.to_display(held), unit=record.parameter.unit)
        record.parameter.check(value)
    except Exception:  # noqa: BLE001 - the old check's own refusals, restated
        return False
    return same(old_back(record, value.value), held)


def floats_of(preset) -> dict[str, float]:
    _, slots = split_modulations(preset.modulations)
    return {str(k): v for k, v in (*preset.settings.values.items(), *slots.items())}


texts = [("the start preset", dumps(start_preset())), (str(FIXTURE), FIXTURE.read_text("utf-8"))]
texts += [(name, (PRESETS / name).read_text("utf-8")) for name in MANIFEST]
total = in_values = in_entry = restored = through_patch = through_json = 0
old_in_values = 0
gained: list[tuple[str, str, float]] = []
lost: list[tuple[str, str, float]] = []
for name, text in texts:
    preset = loads(text)
    floats = floats_of(preset)
    patch = patch_of(preset)
    carried = json.loads(patch.opaque[PRESET_ENTRY])["settings"]
    back = floats_of(preset_of(patch))
    total += len(floats)
    in_values += len(patch.values)
    in_entry += len(carried)
    assert len(patch.values) + len(carried) == len(floats), name
    restored += sum(same(back[key], floats[key]) for key in patch.values)
    through_patch += dumps(preset_of(patch)) == text
    through_json += dumps(preset_of(Patch.model_validate_json(patch.model_dump_json()))) == text
    for key, held in floats.items():
        was, now = old_carries(key, held), key in patch.values
        old_in_values += was
        if now and not was:
            gained.append((name, key, held))
        if was and not now:
            lost.append((name, key, held))
print(f"{len(texts)} texts: the start preset, the fixture and {len(MANIFEST)} installed presets")
print(f"  floats: {total}")
print(f"  in values: {in_values} (under the rule before this revision: {old_in_values})")
print(f"  in the preset entry: {in_entry}")
print(f"  of those in values, restored exactly by a load of the patch: {restored}")
print(f"  byte for byte through a patch: {through_patch}")
print(f"  byte for byte through the patch's JSON: {through_json}")
print(f"  in values now and not before: {len(gained)}")
for name, key, held in gained:
    record = by_id[key]
    print(f"    {name}: {key} {held!r}, shown as {record.to_display(held)!r}")
print(f"  in values before and not now: {len(lost)}")
for name, key, held in lost:
    print(f"    {name}: {key} {held!r}")

print()
print("PatchLoaded.parameter_count for each form of patch, on a VitalInstrument over Vital")
bus = EventBus()
recorder = RecordingSubscriber()
bus.subscribe(None, recorder)
instrument = VitalInstrument(bus)
whole = instrument.patch()
instrument.load_patch(whole)
alone = Patch(
    instrument="Vital",
    values={
        ParameterId("filter_1_on"): Quantity(value=1.0, unit=Unit.none),
        ParameterId("filter_1_cutoff"): Quantity(value=12.0, unit=Unit.semitones),
    },
)
instrument.load_patch(alone)
ah_eh = patch_of(loads((PRESETS / "Factory/Presets/Ah Eh Ee Oh.vital").read_text("utf-8")))
instrument.load_patch(ah_eh)
counts = [event.parameter_count for event in recorder.of_type(PatchLoaded)]
entry = len(json.loads(ah_eh.opaque[PRESET_ENTRY])["settings"])
print(f"  a whole patch of the start preset: {counts[0]}, its values {len(whole.values)}")
print(f"  a patch of two values alone: {counts[1]}")
print(
    f"  a whole patch of Ah Eh Ee Oh, {entry} float in its entry: {counts[2]}, its values "
    f"{len(ah_eh.values)}"
)
