```python
"""The revision after PR 14's fifth review: `main`'s shipped methods against this branch's, directly.

Nothing is reproduced. Two pools of processes each import `patchbay_vital` from one tree, `main`'s
(an extract of its commit) or this branch's, by putting that tree's `src` first on `sys.path`
before the first import, and each calls its own tree's scalar methods on every float32 normalized
value r of a record, as a caller would:

  P0 = scale.to_preset(r)            the preset the parameter holds
  D0 = record.from_normalized(r)     its display
  TP = record.to_preset(D0)          the display's preset
  TN = record.to_normalized(D0)      the normalized value a set of the display sends
  HN = scale.to_normalized(P0)       the held preset's normalized value

This process compares the two trees' five values on every r and counts, by binade of r:

  S!=, D!=   P0 or D0 differs (the scale and the curve forward, unchanged since `main`)
  PRESET     TP differs from `main`'s
  NORM       TN differs from `main`'s
  HELDNORM   HN differs from `main`'s
  ANY        any of the five differs

and, from this branch's own result TP on a linear scale, which branch of `to_preset`'s rule the
value took. P1 is the curve's float32 for D0, clamped to the scale; `landed` is what a set of P1
holds, scale.to_preset(scale.to_normalized(P1)); `other` is the float32 next to P1 away from
`landed`. The three are distinct wherever `landed` is not P1, so TP names its branch:

  KEPT       TP == P1 and a set of it holds it (0 <= TN <= 1 and scale.to_preset(TN) == TP)
  PLUGIN     TP == landed != P1         the first branch
  OTHER      TP == other, landed != P1  the second branch
  UNHELD     TP == P1 and a set of it does not hold it: the third branch
  UNKNOWN    none of these
  PN         of PLUGIN, scale.to_normalized(P1) != TN: the normalized value computed on the way
             to the first branch is not the one the scale gives the preset returned

Every count is also split at `volume`'s floor exactly, by P0 <= 2 ** -43 and P0 > 2 ** -43, value
by value, never by binade.

OUT counts what the run stops on, at once:
  on `volume`    a difference from `main` on a value that is not PLUGIN or OTHER, any HELDNORM,
                 any UNHELD or UNKNOWN;
  on an envelope record
                 a difference from `main` where P0 is not a power of two, any PLUGIN, OTHER,
                 UNHELD or UNKNOWN.

Run from the repository root:
  uv run --locked python direct_count.py --main <main's src> --branch <this branch's src>
      --record <id> [--from <binade> --to <binade>] [--time [--budget <hours>]] [--label <text>]
      [--main-processes <n>] [--branch-processes <n>]
"""

import argparse
import math
import multiprocessing
import sys
import time

import numpy as np

FLOOR = 2.0**-43
CHUNK = 1 << 16
SPANS = [("zero", 0, 1), ("sub", 1, 1 << 23)]
SPANS += [(str(e - 127), e << 23, (e + 1) << 23) for e in range(1, 127)]
SPANS += [("one", 0x3F800000, 0x3F800001)]
COMPARED = ("S!=", "D!=", "PRESET", "NORM", "HELDNORM", "ANY")
BRANCHES = ("KEPT", "PLUGIN", "OTHER", "UNHELD", "UNKNOWN")
COLUMNS = ("n", *COMPARED, *BRANCHES, "PN", "OUT")

_TREE = {}


def use(src):
    """A worker's initializer: import `patchbay_vital` from one tree, before anything else does."""
    sys.path.insert(0, src)
    import patchbay  # noqa: PLC0415
    import patchbay_vital.float32  # noqa: PLC0415
    import patchbay_vital.parameters  # noqa: PLC0415

    _TREE["parameters"] = patchbay_vital.parameters
    _TREE["float32"] = patchbay_vital.float32
    _TREE["id"] = patchbay.ParameterId


def where():
    """Return the files this worker's tree was imported from."""
    return _TREE["parameters"].__file__, _TREE["float32"].__file__


def work(job):
    """Call one tree's methods on every float32 normalized value of a bit-pattern range."""
    name, lo, hi, classify = job
    record = _TREE["parameters"].parameter_map().find(_TREE["id"](name))
    scale, curve = record.scale, record.curve
    linear = scale.kind == "linear"
    values = np.arange(lo, hi, dtype=np.uint32).view(np.float32).astype(np.float64).tolist()
    out = np.empty((5, len(values)))
    branch = np.full(len(values), -1, np.int8)
    on_the_way = np.zeros(len(values), bool)
    step = _TREE["float32"].next_float32 if classify else None
    for i, r in enumerate(values):
        p0 = scale.to_preset(r)
        d0 = record.from_normalized(r)
        tp = record.to_preset(d0)
        tn = record.to_normalized(d0)
        out[:, i] = (p0, d0, tp, tn, scale.to_normalized(p0))
        if not (classify and linear):
            continue
        p1 = min(max(curve.to_preset(d0), scale.minimum), scale.maximum)
        if tp == p1:
            branch[i] = 0 if 0.0 <= tn <= 1.0 and scale.to_preset(tn) == tp else 3
            continue
        way = scale.to_normalized(p1)
        landed = scale.to_preset(way)
        if landed == p1:
            branch[i] = 4
        elif tp == landed:
            branch[i] = 1
            on_the_way[i] = way != tn
        elif tp == step(p1, math.copysign(math.inf, p1 - landed)):
            branch[i] = 2
        else:
            branch[i] = 4
    return lo, out, branch, on_the_way


def tally(record_kind, ours, theirs, branch, on_the_way):
    """Return the counts at or below the floor and above it, and the values the run stops on."""
    p0 = ours[0]
    differs = [ours[k] != theirs[k] for k in range(5)]
    anything = differs[0] | differs[1] | differs[2] | differs[3] | differs[4]
    classes = [branch == b for b in range(len(BRANCHES))]
    if record_kind == "volume":
        moved = classes[1] | classes[2]
        out = (anything & ~moved) | differs[4] | classes[3] | classes[4]
    else:
        power = np.frexp(p0)[0] == 0.5  # noqa: PLR2004 - a power of two's significand
        out = (anything & ~power) | classes[1] | classes[2] | classes[3] | classes[4]
    flags = (*differs, anything, *classes, on_the_way, out)
    below = p0 <= FLOOR
    rows = []
    for side in (below, ~below):
        rows.append([int(np.count_nonzero(side)), *(int(np.count_nonzero(f & side)) for f in flags)])
    return np.array(rows, dtype=np.int64), np.flatnonzero(out)


def spans(first, last):
    labels = [label for label, _, _ in SPANS]
    lo = 0 if first is None else labels.index(first)
    hi = len(SPANS) - 1 if last is None else labels.index(last)
    return SPANS[lo : hi + 1]


def line(label, row):
    return f"{label:>12} " + " ".join(f"{int(v):>11}" for v in row)


def timing(pools, name, where_from):
    """Time each tree on 200,000 values, half at the floor binades and half over every binade."""
    rng = np.random.default_rng(21)
    report = {}
    for side, pool in pools.items():
        started = time.time()
        jobs = []
        for start in rng.integers(0x10000000, 0x3F000000, 4, dtype=np.uint32).tolist():
            jobs.append(pool.apply_async(work, ((name, start, start + 50_000, side == "branch"),)))
        for job in jobs:
            job.get()
        report[side] = (time.time() - started) / 50_000 * 1e6
        print(f"  {side} ({where_from[side]}): {report[side]:.1f} us a value, 4 ranges of 50,000"
              " values in 4 processes at once")
    return report


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--main", required=True)
    parser.add_argument("--branch", required=True)
    parser.add_argument("--record", required=True)
    parser.add_argument("--from", dest="first")
    parser.add_argument("--to", dest="last")
    parser.add_argument("--main-processes", type=int, default=5)
    parser.add_argument("--branch-processes", type=int, default=7)
    parser.add_argument("--time", action="store_true")
    parser.add_argument("--label", default="")
    parser.add_argument("--budget", type=float, default=4.0)
    args = parser.parse_args()
    kind = "volume" if args.record == "volume" else "envelope"
    context = multiprocessing.get_context("spawn")
    sizes = {"main": args.main_processes, "branch": args.branch_processes}
    sources = {"main": args.main, "branch": args.branch}
    pools = {s: context.Pool(sizes[s], initializer=use, initargs=(sources[s],)) for s in sizes}
    where_from = {}
    for side, pool in pools.items():
        files = pool.apply(where)
        where_from[side] = files[0]
        print(f"== the {side} pool, {sizes[side]} processes, imports {files[0]}")
        print(f"   and {files[1]}")
        if not all(f.startswith(sources[side]) for f in files):
            print("   NOT the tree asked for: stopping")
            return 2
    if args.time:
        print(f"== timing on {args.record}")
        took = timing(pools, args.record, where_from)
        total = 1_065_353_217
        wall = max(took["main"] / sizes["main"], took["branch"] / sizes["branch"]) * total / 3.6e9
        cpu = (took["main"] + took["branch"]) * total / 3.6e9
        print(f"   every float32 normalized value of one record, {total:,}: about {cpu:.1f}"
              f" CPU-hours, {wall:.2f} hours wall on these pools")
        if wall > args.budget:
            print(f"   more than the {args.budget} hours a record the owner allowed (8 for the two):"
                  " stopping")
            return 3
        return 0
    chosen = spans(args.first, args.last)
    print(f"== {args.record}{args.label}: binades {chosen[0][0]} to {chosen[-1][0]},"
          f" {sum(hi - lo for _, lo, hi in chosen):,} float32 normalized values, each through both"
          " trees")
    print(line("binade", ()) + " ".join(f"{c:>11}" for c in COLUMNS))
    started = time.time()
    total = np.zeros((2, len(COLUMNS)), dtype=np.int64)
    quiet = []
    for label, lo, hi in chosen:
        row = np.zeros((2, len(COLUMNS)), dtype=np.int64)
        jobs = []
        for start in range(lo, hi, CHUNK):
            end = min(hi, start + CHUNK)
            jobs.append((
                pools["main"].apply_async(work, ((args.record, start, end, False),)),
                pools["branch"].apply_async(work, ((args.record, start, end, True),)),
            ))
        for theirs_job, ours_job in jobs:
            start, theirs, _, _ = theirs_job.get()
            _, ours, branch, on_the_way = ours_job.get()
            counted, stops = tally(kind, ours, theirs, branch, on_the_way)
            row += counted
            if stops.size:
                print(line(label, row.sum(axis=0)))
                print(f"== OUT: {stops.size} values in the range from bit pattern {start}; the first:")
                for i in stops[:5].tolist():
                    bits = np.array([start + i], dtype=np.uint32).view(np.float32)[0]
                    print(f"   r {float(bits)!r}: this branch's P0, D0, TP, TN, HN"
                          f" {[float(v) for v in ours[:, i]]!r}, main's"
                          f" {[float(v) for v in theirs[:, i]]!r}, branch"
                          f" {BRANCHES[branch[i]] if branch[i] >= 0 else 'none'}")
                print("== stopped at once, as the owner asked")
                for pool in pools.values():
                    pool.terminate()
                return 1
        total += row
        whole = row.sum(axis=0)
        if whole[1:].any() and not (whole[7] == whole[0] and not whole[1:7].any()):
            if row[0, 0] and row[1, 0]:
                print(line(label, whole))
                print(line("<= 2**-43", row[0]))
                print(line("> 2**-43", row[1]))
            else:
                print(line(label, whole))
        else:
            quiet.append(label)
        sys.stdout.flush()
    print(line("total", total.sum(axis=0)))
    print(line("P0 <= 2**-43", total[0]))
    print(line("P0 > 2**-43", total[1]))
    print(f"  every value KEPT and nothing differing from main in {len(quiet)} binades:"
          f" {', '.join(quiet) if quiet else 'none'}")
    print(f"  {time.time() - started:.0f} s")
    for pool in pools.values():
        pool.close()
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
import inspect
import textwrap

    (NormalizedScale, "to_preset"), (NormalizedScale, "to_normalized"), (NormalizedScale, "holds"),

def source(owner, name):
    member = inspect.getattr_static(owner, name)
    member = member.fget if isinstance(member, property) else member
    member = getattr(member, "__wrapped__", member)
    lines, first = inspect.getsourcelines(member)
    return ast.parse(textwrap.dedent("".join(lines))), first


def reads(tree):
    """Return every `<name>.<attribute>` read in a function, as written."""
    found = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Attribute) and isinstance(node.value, ast.Name):
            found.add(f"{node.value.id}.{node.attr}")
        elif isinstance(node, ast.Attribute) and isinstance(node.value, ast.Attribute):
            inner = node.value
            if isinstance(inner.value, ast.Name):
                found.add(f"{inner.value.id}.{inner.attr}.{node.attr}")
    return sorted(found)


def main():

    records = [r for r in m.parameters
```
