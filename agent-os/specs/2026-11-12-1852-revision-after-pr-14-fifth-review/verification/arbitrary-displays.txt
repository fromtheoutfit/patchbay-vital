"""The revision after PR 14's fifth review: how often an arbitrary display takes the third branch.

`vital/parameter-map.md` says that of 20,000 displays drawn uniformly over `volume`'s range (seed
8), 5.4% take `to_preset`'s third branch, with no script recorded. This draws them, with numpy's
`default_rng(8).uniform` over the parameter's bounds, and reads the branch each takes from the
shipped `to_preset`'s result, as the direct count does: P1 the curve's float32 for the display,
clamped; `landed` what a set of P1 holds; `other` the float32 next to P1 away from `landed`. The
same is counted on the sample the fourth-review sweep checked its reproduction on, to say how many
of its 200,010 values reach each branch (finding 3). Run from the repository root:
uv run --locked python <scratchpad>/r5/arbitrary_displays.py <the fourth-review scratch folder>
"""

import math
import sys
from collections import Counter

import numpy as np
from patchbay import ParameterId
from patchbay_vital.float32 import next_float32
from patchbay_vital.parameters import parameter_map


def branch(rec, display):
    scale, curve = rec.scale, rec.curve
    back = rec.to_preset(display)
    p1 = min(max(curve.to_preset(display), scale.minimum), scale.maximum)
    if back == p1:
        return "KEPT" if scale.holds(p1) else "UNHELD (the third branch)"
    landed = scale.to_preset(scale.to_normalized(p1))
    if landed != p1 and back == landed:
        return "PLUGIN (the first branch)"
    if landed != p1 and back == next_float32(p1, math.copysign(math.inf, p1 - landed)):
        return "OTHER (the second branch)"
    return "UNKNOWN"


def main():
    rec = parameter_map().find(ParameterId("volume"))
    p = rec.parameter
    displays = np.random.default_rng(8).uniform(p.minimum, p.maximum, 20_000).tolist()
    counts = Counter(branch(rec, d) for d in displays)
    print(f"== 1. volume, 20,000 displays uniform over [{p.minimum!r}, {p.maximum!r}],"
          " numpy default_rng(8):")
    for name, n in sorted(counts.items()):
        print(f"   {name}: {n} ({n / 200:.2f}%)")
    shown = sum(rec.to_display(rec.to_preset(d)) == d for d in displays)
    print(f"   displays some preset the map gives shows exactly: {shown}")

    sys.path.insert(0, sys.argv[1])
    import sweep  # noqa: PLC0415 - the fourth-review revision's script, for its sample()

    values = sweep.sample().tolist()
    counts = Counter(branch(rec, rec.from_normalized(x)) for x in values)
    print(f"== 2. volume, the {len(values):,} values the fourth-review sweep checked its"
          " reproduction on (its sample(): 200,000 bit patterns uniform over [0, 1], seed 15, the"
          " seven pinned values, 0, 1 and 2 ** -149), by the branch each takes:")
    for name, n in sorted(counts.items()):
        print(f"   {name}: {n}")


if __name__ == "__main__":
    main()
