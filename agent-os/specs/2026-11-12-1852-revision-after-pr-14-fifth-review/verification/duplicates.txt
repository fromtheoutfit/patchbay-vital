"""The revision after PR 14's fifth review: each record counted beside its group, table for table.

The fourth-review revision's `sweep.py` counted the six records with a curve offset and the 12
envelope delay and hold records directly, each beside its group's count. Its `duplicates.py`
compared each such record's binade rows and total with its group's first record's, and skipped
each table's closing line, which holds the number of binades where nothing but n was counted and
the number of values given to the exact scalar computation. This reads the same transcript and
compares each table whole: the line after the record's name (its scale, curve and bounds), the
column header, every binade row, the total, and the closing line with only its run time, the
seconds at its end, removed. Run from the repository root:
uv run --locked python <scratchpad>/r5/duplicates.py <the fourth-review sweep.txt>
"""

import re
import sys

from patchbay import ParameterId
from patchbay_vital.parameters import parameter_map


def group_key(rec):
    p, s, c = rec.parameter, rec.scale, rec.curve
    return (p.kind, s.kind, s.minimum, s.maximum, c.kind, c.multiply, c.offset, p.minimum, p.maximum)


def tables(path):
    """Return each counted record's table: its heading less its name, and every line to its last."""
    out, name, rows = {}, None, []
    with open(path, encoding="utf-8") as transcript:
        for line in transcript:
            line = line.rstrip("\n")
            if line.startswith("== the script, run from the repository root beside"):
                break  # the exponential section, which counts each group once
            if line.startswith("-- ") and ": scale " in line:
                name, rows = line[3:].split(":", 1)[0], [line[3:].split(":", 1)[1]]
            elif name and line.startswith("  nothing but n counted"):
                closing, seconds = re.subn(r"; [0-9]+ s$", "", line)
                assert seconds == 1, line
                out[name] = [*rows, closing]
                name = None
            elif name:
                rows.append(line)
    return out


def main():
    counted = tables(sys.argv[1])
    groups = {}
    for rec in parameter_map().parameters:
        groups.setdefault(group_key(rec), []).append(rec.id)
    first = {rid: ids[0] for ids in groups.values() for rid in ids}
    direct = [name for name in counted if first[ParameterId(name)] != name]
    print(f"== {len(counted)} tables read; {len(direct)} records counted beside their group, each"
          " against its group's first record")
    same = 0
    for name in direct:
        group = first[ParameterId(name)]
        ours, theirs = counted[name], counted[group]
        equal = ours == theirs
        same += equal
        print(f"  {name} against {group}: {len(ours)} and {len(theirs)} lines (heading, header,"
              f" binades, total, closing line), identical {equal}")
        print(f"    closing line: {ours[-1].strip()}")
        for a, b in zip(ours, theirs, strict=False):
            if a != b:
                print(f"    differs: {a!r}\n         from {b!r}")
    print(f"== {same} of {len(direct)} identical, every line of each table, the run time alone left"
          " out")


if __name__ == "__main__":
    main()
