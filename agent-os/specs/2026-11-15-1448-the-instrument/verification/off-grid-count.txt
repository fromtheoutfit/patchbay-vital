"""The floats a set cannot land on, counted, and how far a set of each one's own display moves it.

For the start preset and each installed preset: every float inside its parameter's bounds that
the scale does not hold, a float off the host's grid. For each, the display get returns, the
preset a set of that display holds (the map's to_preset, the instrument's rule), how many float32
steps that is from the float, and how far its display is from the first, in the display's unit.

Run from the repository root with the locked interpreter. No plugin is used.
"""

import struct
from collections import Counter
from pathlib import Path

from patchbay import PatchbayError, Quantity

from patchbay_vital import loads
from patchbay_vital.format import split_modulations
from patchbay_vital.parameters import parameter_map

PRESETS = Path.home() / "Music" / "Vital"
MANIFEST = [
    line
    for line in Path("tests/data/factory-manifest.txt").read_text(encoding="utf-8").splitlines()
    if not line.startswith("#")
]
found = parameter_map()
by_id = {record.id: record for record in found.parameters}


def steps(one: float, other: float) -> int:
    a = struct.unpack("<i", struct.pack("<f", one))[0]
    b = struct.unpack("<i", struct.pack("<f", other))[0]
    return abs(a - b) if (one >= 0) == (other >= 0) else abs(a) + abs(b)


def count(text: str) -> tuple[int, Counter[int], int, int, float, list[tuple[str, float, float, int]]]:
    preset = loads(text)
    _, slots = split_modulations(preset.modulations)
    off = 0
    by_steps: Counter[int] = Counter()
    same_display = 0
    held_count = 0
    most = 0.0
    examples = []
    for key, held in {**preset.settings.values, **slots}.items():
        record = by_id.get(key)
        if record is None:
            continue
        display = record.to_display(held)
        try:
            record.parameter.check(Quantity(value=display, unit=record.parameter.unit))
        except PatchbayError:
            continue
        if record.scale.holds(held):
            held_count += 1
            continue
        off += 1
        landed = record.to_preset(display)
        moved = steps(held, landed)
        by_steps[moved] += 1
        shown = record.to_display(landed)
        same_display += shown == display
        relative = abs(shown - display) / max(abs(display), 1e-300)
        most = max(most, relative)
        examples.append((key, held, landed, moved))
    return off, by_steps, same_display, held_count, most, examples


start = Path("src/patchbay_vital/data/start.vital").read_text(encoding="utf-8")
off, by_steps, same, held_count, most, examples = count(start)
print(f"the start preset: {off} floats off the grid of {off + held_count} inside their bounds")
print(f"  float32 steps from each to what a set of its own display holds: {dict(sorted(by_steps.items()))}")
print(f"  of them, a set of the display shows that display again on {same}; the largest relative move of a display {most:.3g}")
for key, held, landed, moved in examples:
    print(f"    {key}: {held!r} -> {landed!r} ({moved} step{'s' if moved != 1 else ''})")
total_off = total_held = total_same = 0
total_steps: Counter[int] = Counter()
presets_with = 0
largest = 0.0
for relative in MANIFEST:
    off, by_steps, same, held_count, most, _ = count((PRESETS / relative).read_bytes().decode("utf-8"))
    total_off += off
    total_held += held_count
    total_same += same
    total_steps.update(by_steps)
    presets_with += off > 0
    largest = max(largest, most)
print(f"the {len(MANIFEST)} installed presets: {total_off} floats off the grid of {total_off + total_held} inside their bounds, in {presets_with} presets")
print(f"  float32 steps from each to what a set of its own display holds: {dict(sorted(total_steps.items()))}")
print(f"  of them, a set of the display shows that display again on {total_same}; the largest relative move of a display {largest:.3g}")

# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two

#!/usr/bin/env bash
# usage: machine.sh   Prints one line: the machine, its system, the locked Python and the load.
set -euo pipefail
echo "$(uname -m), $(sysctl -n machdep.cpu.brand_string), $(sysctl -n hw.ncpu) cores, macOS $(sw_vers -productVersion) ($(sw_vers -buildVersion)), Python $(uv run --locked python -c 'import platform; print(platform.python_version())'), load $(sysctl -n vm.loadavg | tr -d '{}' | xargs), at $(date '+%Y-%m-%d %H:%M %Z')"
```

--- <scratchpad>/i8/transcript.sh
```
#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/i8/$(basename "$s")"; done
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/i8/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
