from patchbay_vital.format import split_modulations
from patchbay_vital.parameters import parameter_map
from patchbay_vital.patch import PRESET_ENTRY, patch_of, preset_of

PRESETS = Path.home() / "Music" / "Vital"
MANIFEST = [
    line
    for line in Path("tests/data/factory-manifest.txt").read_text(encoding="utf-8").splitlines()
    if not line.startswith("#")
]
files = [("the start preset", Path("src/patchbay_vital/data/start.vital")), ("the fixture", Path("tests/data/init-media-replaced.vital"))]
files += [(relative, PRESETS / relative) for relative in MANIFEST]
found = parameter_map()
total = Counter()
by_key: Counter[str] = Counter()
rows = []
for label, path in files:
    text = path.read_bytes().decode("utf-8")
    preset = loads(text)
    _, slots = split_modulations(preset.modulations)
    floats = {**preset.settings.values, **slots}
    patch = patch_of(preset)
    carried = json.loads(patch.opaque[PRESET_ENTRY])["settings"]
    assert not set(patch.values) & set(carried)
    assert set(patch.values) | set(carried) == set(floats)
    exact = dumps(preset_of(patch)) == text
    through_json = dumps(preset_of(Patch.model_validate_json(patch.model_dump_json()))) == text
    total["texts"] += 1
    total["floats"] += len(floats)
    total["in values"] += len(patch.values)
    total["in the preset entry"] += len(carried)
    total["byte for byte through a patch"] += exact
    total["byte for byte through the patch's JSON"] += through_json
    total["keys absent"] += len(found.parameters) - sum(1 for record in found.parameters if record.preset_key in floats)
    for key, held in carried.items():
        by_key[key] += 1
        record = next((r for r in found.parameters if r.id == key), None)
        why = "no parameter" if record is None else (
            f"outside its scale, {record.scale.minimum!r} to {record.scale.maximum!r}, shown as {record.to_display(held)!r} against {record.parameter.minimum!r} to {record.parameter.maximum!r}"
            if not record.scale.minimum <= held <= record.scale.maximum
            else "inside its scale, its display cannot carry it"
        )
        rows.append((label, key, held, why))
print(f"{total['texts']} texts: the start preset, the fixture and {len(MANIFEST)} installed presets")
for name in ("floats", "in values", "in the preset entry", "keys absent", "byte for byte through a patch", "byte for byte through the patch's JSON"):
    print(f"  {name}: {total[name]}")
print(f"the floats that took the opaque route, {len(rows)} on {len({label for label, *_ in rows})} texts, by key: {dict(by_key.most_common())}")
for label, key, held, why in rows:
    print(f"  {label}: {key} {held!r}: {why}")

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"

#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/i8/$(basename "$s")"; done
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/i8/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
