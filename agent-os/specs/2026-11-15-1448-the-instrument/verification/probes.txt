def centroid(audio: np.ndarray) -> float:
    mono = audio.mean(axis=0)[7200:]
    magnitude = np.abs(np.fft.rfft(mono))
    return float((np.fft.rfftfreq(len(mono), 1 / RATE) * magnitude).sum() / magnitude.sum())


def rms(audio: np.ndarray) -> float:
    held = audio.mean(axis=0)[int(0.2 * RATE) : int(1.4 * RATE)]
    return float(np.sqrt((held**2).mean()))

"""One load, in a process of its own: what it leaves in the plugin, or nothing if it never returns.

usage: probe_load.py <case>. Prints one line per fact. The caller gives the process a time limit:
a load Vital answers with a modal alert does not return.
"""

import json
import sys
from pathlib import Path

from patchbay import ParameterId

from patchbay_vital import loads
from patchbay_vital.errors import VitalError
from patchbay_vital.format import split_modulations
from patchbay_vital.host import VitalHost, _floats
from patchbay_vital.parameters import parameter_map
from patchbay_vital.start import start_text

MAP = parameter_map()
START = start_text(MAP).encode("utf-8")
PRESETS = Path.home() / "Music" / "Vital"


def variant(top: dict[str, object] | None = None, **settings: object) -> bytes:
    changed = json.loads(START)
    for key, value in settings.items():
        if value is None:
            del changed["settings"][key]
        else:
            changed["settings"][key] = value
    changed.update(top or {})
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


def no_sections() -> bytes:
    changed = json.loads(START)
    for name in ("wavetables", "lfos", "sample", "custom_warps", "random_values"):
        del changed["settings"][name]
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


def routed(source: str, destination: str) -> bytes:
    changed = json.loads(START)
    changed["settings"]["modulations"][0] = {"source": source, "destination": destination}
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


CASES = {
    # texts the package's own writer can write, from a patch someone built
    "the start preset rewritten by json.dumps (the control)": lambda: variant(),
    "no section at all": no_sections,
    "wavetables an empty array": lambda: variant(wavetables=[]),
    "wavetables an array of one empty object": lambda: variant(wavetables=[{}]),
    "wavetables an object": lambda: variant(wavetables={"x": 1}),
    "lfos an empty array": lambda: variant(lfos=[]),
    "sample an empty object": lambda: variant(sample={}),
    "a routing whose source and destination are no names of Vital's": lambda: routed("nonsense", "nowhere"),
    "synth_version 99.0.0": lambda: variant({"synth_version": "99.0.0"}),
    "synth_version abc": lambda: variant({"synth_version": "abc"}),
    "without filter_1_cutoff": lambda: variant(filter_1_cutoff=None),
    "volume at 1e30, far outside its scale": lambda: variant(volume=1e30),
    # texts that pass the guard below, their fault inside a section, which the package does not read
    "three wavetables whose groups is a number": lambda: variant(wavetables=[{"groups": 5}] * 3),
    "eight lfos whose points is a string": lambda: variant(lfos=[{"name": "x", "num_points": 2, "points": "x", "powers": [0.0, 0.0], "smooth": False}] * 8),
    "a sample with its four keys, its length a string": lambda: variant(sample={"length": "x", "name": "x", "sample_rate": 44100, "samples": ""}),
    # texts the writer cannot write, for reference
    "an empty object": lambda: b"{}",
    "settings empty": lambda: b'{"settings":{}}',
    "a string where a float belongs": lambda: variant(volume="loud"),
    "wavetables a string": lambda: variant(wavetables="x"),
    "without modulations": lambda: variant(modulations=None),
}


WRITABLE = {
    "an empty object": "no: a preset's models require its metadata and settings, so the writer never writes this",
    "settings empty": "no: the format requires the routing list and each slot's floats",
    "a string where a float belongs": "no: a setting is a float in the model",
    "wavetables a string": "no: a section is the text of a JSON array or object in the model",
    "without modulations": "no: the format requires the routing list",
}
VERSION = tuple(int(part) for part in MAP.vital_version.split("."))


def guard(text: bytes) -> str:
    """Say what the instrument's guard, as decided, does with a text: the reason it refuses, or passes."""
    preset = json.loads(text)
    version = preset.get("synth_version")
    if not isinstance(version, str) or not all(part.isascii() and part.isdigit() for part in version.split(".")):
        return f"refuses: synth_version {version!r} is not digits and dots"
    if tuple(int(part) for part in version.split(".")) > VERSION:
        return f"refuses: synth_version {version} is newer than the plugin's {MAP.vital_version}"
    settings = preset.get("settings", {})
    wavetables, lfos, sample = settings.get("wavetables"), settings.get("lfos"), settings.get("sample")
    if not (isinstance(wavetables, list) and len(wavetables) == 3 and all(isinstance(w, dict) for w in wavetables)):
        return "refuses: wavetables is not an array of 3 objects"
    if not (isinstance(lfos, list) and len(lfos) == 8 and all(isinstance(l, dict) for l in lfos)):
        return "refuses: lfos is not an array of 8 objects"
    if not (isinstance(sample, dict) and {"length", "name", "sample_rate", "samples"} <= set(sample)):
        return "refuses: sample is not an object holding length, name, sample_rate and samples"
    return "passes"


def sections_of(text: bytes) -> dict[str, str]:
    settings = json.loads(text)["settings"]
    return {k: json.dumps(v, sort_keys=True) for k, v in settings.items() if not isinstance(v, float)}


def one_load(label: str) -> None:
    text = CASES[label]()
    print(f"can the package's writer write it: {WRITABLE.get(label, 'yes')}")
    try:
        print(f"the guard: {guard(text)}", flush=True)
    except (ValueError, AttributeError) as error:
        print(f"the guard: is not reached ({type(error).__name__})", flush=True)
    host = VitalHost()
    host.load_state(START)
    host.load_state(START)
    marker = ParameterId("filter_1_cutoff")
    host.set(marker, 0.9)
    before, before_sections = _floats(host.state()), sections_of(host.state())
    print("before the load: the marker filter_1_cutoff", before[marker], flush=True)
    raised = "no error"
    try:
        host.load_state(text)
    except VitalError as error:
        raised = f"{type(error).__name__}: {str(error)[:90]}"
    after, after_sections = _floats(host.state()), sections_of(host.state())
    moved = sorted(k for k in before if after.get(k) != before[k])
    gone = sorted(k for k in before if k not in after)
    sections_moved = sorted(k for k in before_sections if after_sections.get(k) != before_sections[k])
    print(f"returned: {raised}")
    print(f"floats changed {len(moved)} of {len(before)} {moved[:3]}, gone {len(gone)}; the marker -> {after.get(marker)}; sections changed {sections_moved}")


def lacking_keys() -> None:
    manifest = [
        line
        for line in Path("tests/data/factory-manifest.txt").read_text(encoding="utf-8").splitlines()
        if not line.startswith("#")
    ]
    old_text = (PRESETS / manifest[0]).read_bytes()
    old = loads(old_text.decode("utf-8"))
    _, old_slots = split_modulations(old.modulations)
    held = {**old.settings.values, **old_slots}
    lacking = [record for record in MAP.parameters if record.id not in held]
    print(f"{manifest[0]} (synth_version {old.synth_version}) lacks {len(lacking)} of the 903 keys; its sections: {sorted(old.opaque)}")
    host = VitalHost()
    host.load_state(START)
    host.load_state(START)
    for record in lacking:
        host.set(record.id, 1.0 if record.scale.to_normalized(record.preset_default) < 0.5 else 0.0)
    before = _floats(host.state())
    away = sum(before[r.preset_key] != r.preset_default for r in lacking)
    host.load_state(old_text)
    after = _floats(host.state())
    kept = [r.id for r in lacking if after[r.preset_key] == before[r.preset_key] and before[r.preset_key] != r.preset_default]
    default = [r.id for r in lacking if after[r.preset_key] == r.preset_default]
    other = [r.id for r in lacking if r.id not in kept and r.id not in default]
    print(f"each of those {len(lacking)} was first set away from its default ({away} moved); after the load: at its default {len(default)}, still at the value set {len(kept)}, at another value {len(other)}")
    print(f"    still at the value set (first eight): {kept[:8]}")
    print(f"    at another value (first six): {[(k, before[k], after[k], MAP.find(ParameterId(k)).preset_default) for k in other[:6]]}")
    start_sections, now = sections_of(START), sections_of(host.state())
    for name in ("custom_warps", "random_values"):
        print(f"    section {name}, which the old preset lacks: in the state after the load {name in now}; the start preset's own text {now.get(name) == start_sections[name]}")
    host.load_state(START)
    after_start = _floats(host.state())
    wrong = sorted(r.preset_key for r in MAP.parameters if after_start[r.preset_key] != r.preset_default)
    print(f"then one load of the start preset: floats not the start's {len(wrong)} {wrong[:4]}")


if __name__ == "__main__":
    if sys.argv[1] == "--list":
        print("\n".join(CASES))
    elif sys.argv[1] == "--lacking":
        lacking_keys()
    else:
        one_load(sys.argv[1])

#!/usr/bin/env bash
# usage: probe_loads.sh   (from the repository root)
# Runs each load of probe_load.py in a process of its own with a time limit. A load that does not
# return is sampled once, to show what it waits on, and killed.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
LIMIT=6
sampled=0
echo "== 5. what a load the plugin cannot use leaves in the plugin, each in a process of its own, $LIMIT s allowed"
echo "before each: a fresh Vital, the start preset loaded twice, filter_1_cutoff set to normalized 0.9 as a marker."
echo "a load that has not returned after $LIMIT s is killed, by design: Vital has put up a modal alert and the caller would wait for ever."
echo "beside each text: whether the package's own writer can write it, and what the instrument's guard, as decided, does with it."
uv run --locked python -B "$R/probe_load.py" --list > "$R/out/cases.txt"
while IFS= read -r label; do
  echo
  echo "-- $label"
  uv run --locked python -u -B "$R/probe_load.py" "$label" > "$R/out/case.txt" 2>&1 &
  pid=$!
  waited=0
  while kill -0 "$pid" 2>/dev/null && [ "$waited" -lt "$LIMIT" ]; do sleep 1; waited=$((waited + 1)); done
  if kill -0 "$pid" 2>/dev/null; then
    cat "$R/out/case.txt"
    echo "DID NOT RETURN within $LIMIT s: the process is waiting, and is killed, by design"
    child=$(pgrep -P "$pid" | head -1)
    if [ "$sampled" -eq 0 ] && [ -n "$child" ]; then
      sampled=1
      echo "   what it waits on (sample of the process, the frames from Vital's setStateInformation down to AppKit's modal loop):"
      sample "$child" 1 2>/dev/null | grep -E "setStateInformation|AlertWindow|OSXMessageBox|NSAlert runModal|runModalForWindow" | sed 's/^[ +!:|]*/      /' | cut -c1-170 | head -8
    fi
    pkill -P "$pid" 2>/dev/null; kill "$pid" 2>/dev/null; wait "$pid" 2>/dev/null
  else
    wait "$pid"; status=$?
    cat "$R/out/case.txt"
    [ "$status" -eq 0 ] || echo "exit=$status"
  fi
done < "$R/out/cases.txt"
echo
echo "== 5b. a key the text lacks: the default, or what the plugin held before?"
uv run --locked python -u -B "$R/probe_load.py" --lacking 2>&1

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"
