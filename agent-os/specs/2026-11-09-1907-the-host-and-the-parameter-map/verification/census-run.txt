    print(f"   {n:4d}  {kind:21s} {unit:10s} x{mul:g} {off:+g}")
print("choices disambiguated:")

print("choice offsets (a choice's index is the preset less this):", dict(Counter(r.scale.minimum for r in records if r.parameter.kind == "choice")))

rng = np.random.default_rng(11)
edges = [0.0, 1.0, float(np.float32(1e-30)), float(np.float32(1e-20)), float(np.float32(1e-12)), float(np.float32(2.0**-24)), float(np.nextafter(np.float32(1.0), np.float32(0.0)))]
display_bad, preset_bad, checked = Counter(), Counter(), 0
for r in records:
    points = edges + [float(x) for x in rng.random(300).astype(np.float32)]
    lo = r.scale.minimum
    for normalized in points:
        preset = r.scale.to_preset(normalized)
        display = r.from_normalized(normalized)
        back = r.to_preset(display)
        checked += 1
        display_bad[r.id] += r.to_display(back) != display
        preset_bad[r.id] += back != preset
    # presets just above the scale's minimum, one float32 step at a time, and tiny ones
    for k in range(1, 2001):
        preset = float(np.float32(lo) + np.float32(k) * np.spacing(np.float32(lo if lo else 1e-38)))
        if not lo <= preset <= r.scale.maximum or r.scale.kind == "indexed":
            continue
        checked += 1
        display = r.to_display(preset)
        back = r.to_preset(min(max(display, r.parameter.minimum), r.parameter.maximum))
        display_bad[r.id] += r.to_display(back) != display
        preset_bad[r.id] += back != preset
print(f"sweep: {checked} conversions over {len(records)} records (both ends, tiny normalized values, 300 random, 2000 float32 steps above each linear scale's minimum; an indexed scale holds whole numbers only)")

print("  a preset not recovered:", {k: v for k, v in preset_bad.items() if v})
print("anchors (normalized, preset, display) at the minimum, the default and the maximum:")

    print(f"   {key} ({r.parameter.kind}, {r.parameter.unit.value}, {r.scale.kind}, {r.curve.kind} x{r.curve.multiply:g} {r.curve.offset:+g}): {row}")
