# ---- the .vstpreset container and the component it holds, as the probes read them


def component(preset_data):
    assert preset_data[:4] == b"VST3"
    list_at = struct.unpack("<q", preset_data[40:48])[0]
    assert preset_data[list_at : list_at + 4] == b"List"
    return preset_data[48:list_at]


def chunk(preset_data):
    comp = component(preset_data)
    size = struct.unpack(">i", comp[172:176])[0]
    return comp[176 : 176 + size]


def text_of(preset_data):
    body = chunk(preset_data)
    _, end = json.JSONDecoder().raw_decode(body.decode("latin-1"))
    return body[:end], body[end:]


def settings(plugin):
    return json.loads(text_of(plugin.preset_data)[0])["settings"]


def wrap(template, text, trailer=TRAILER):
    body = text + trailer
    comp = component(template)
    header = bytearray(comp[16:176])
    struct.pack_into(">i", header, 4, len(body) + 152)
    struct.pack_into(">i", header, 156, len(body))
    new = comp[:16] + bytes(header) + body
    list_at = 48 + len(new)
    out = bytearray(template[:48])
    struct.pack_into("<q", out, 40, list_at)
    entries = b"Comp" + struct.pack("<qq", 48, len(new)) + b"Cont" + struct.pack("<qq", list_at, 0)
    return bytes(out + new + b"List" + struct.pack("<i", 2) + entries)


def dumps(obj):
    return json.dumps(obj, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode()


JUCE64 = ".ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+"


def juce_base64(text):
    size_text, _, body = text.partition(".")
    size = int(size_text)
    out = bytearray(size)
    bit = 0
    for character in body:
        value = JUCE64.index(character)
        for k in range(6):
            if bit < size * 8 and value >> k & 1:
                out[bit >> 3] |= 1 << (bit & 7)
            bit += 1
    return bytes(out)


# ---- float32 arithmetic, exact


def round_f32(value):
    near = np.float32(float(value))
    best = None
    for candidate in (np.nextafter(near, np.float32(-np.inf)), near, np.nextafter(near, np.float32(np.inf))):
        distance = abs(Fraction(float(candidate)) - value)
        even = int(np.array([candidate], dtype=np.float32).view(np.int32)[0]) % 2 == 0
        if best is None or distance < best[0] or (distance == best[0] and even):
            best = (distance, candidate)
    return float(best[1])


def f32(x):
    return float(np.float32(x))


def fused(r, span, minimum):
    return round_f32(Fraction(f32(r)) * Fraction(span) + Fraction(minimum))


def separate(r, span, minimum):
    return float(np.float32(np.float32(np.float32(r) * np.float32(span)) + np.float32(minimum)))


def ulps(a, b):
    bits = np.array([a, b], dtype=np.float32).view(np.int32)
    return abs(int(bits[0]) - int(bits[1]))
