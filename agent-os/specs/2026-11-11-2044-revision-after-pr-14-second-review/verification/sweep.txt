excuses two shapes by value predicates, each consulted only where its claim fails, and this run
counts, per group, that every failure has its excuse's shape: then the set each excuse covers is
the failures exactly. It also counts HELD, which reads the scale alone, on the 9 exponential
groups, whose curves it cannot reproduce. The rows are read from tests/data/ through
tests/conftest.py's `recorded_rows`, the reader the tests use.

Counted for every scale/curve group of the committed map whose curve this arithmetic reproduces
bit for bit (linear, quadratic, quartic, square_root), for each of the six records with a curve
offset and each of the 12 envelope delay and hold records directly; HELD alone for the 9
exponential groups.

P0 = scale.to_preset(r), the preset the parameter holds; D0 = to_display(P0); P1 = to_preset(D0);

  A3     P1 != P0                  the preset back

`bound!` counts what breaks the property as this revision states it: A1, HELDx, A3x or SETx.

The arithmetic is the map's, exactly: r * span is exact in a double (24 x 24 bits); a sum is

CHUNK = 1 << 22
SIGN = np.int64(0x7FFFFFFFFFFFFFFF)


def two_sum(a, b):
    s = a + b
    bp = s - a
    ap = s - bp
    return s, (a - ap) + (b - bp)


def round32(s, err):
    """Return the float32 nearest s + err exactly, s a double and err its exact remainder."""
    c32 = s.astype(np.float32)
    c = c32.astype(np.float64)
    diff = s - c
    toward = np.where(diff > 0, np.float32(np.inf), np.float32(-np.inf))
    nb = np.nextafter(c32, toward).astype(np.float64)
    away = (diff != 0) & (2 * diff == nb - c) & (err != 0) & (np.sign(err) == np.sign(diff))
    return np.where(away, nb, c)


def half_away(v):
    return np.copysign(np.floor(np.abs(v) + 0.5), v)


def ordered(x):
    """Return each double's place in value order as an int64, both zeros at 0."""
    bits = x.view(np.int64)
    return np.where(bits < 0, -(bits & SIGN), bits)


class Vec:
    def __init__(self, rec, scale_only=False):
        s, c = rec.scale, rec.curve
        self.rec = rec
        self.mn, self.mx, self.span, self.ix = s.minimum, s.maximum, s.span, s.kind == "indexed"
        self.kind, self.mul, self.off = c.kind, c.multiply, c.offset
        assert scale_only or self.kind in REPRODUCED
        self.scalar = 0

    def S(self, r):
        s, err = two_sum(r * self.span, np.float64(self.mn))
        v = round32(s, err)
        return half_away(v) if self.ix else v

    def N(self, p):
        a, ea = two_sum(p, np.float64(-self.mn))
        q = (a / self.span).astype(np.float32)
        best = None
        unsure = ea != 0
        for c32 in (np.nextafter(q, np.float32(-np.inf)), q, np.nextafter(q, np.float32(np.inf))):
            c = c32.astype(np.float64)
            d, ed = two_sum(c * self.span, -a)
            unsure |= ed != 0
            d = np.abs(d)
            even = (c32.view(np.uint32) & 1) == 0
            if best is None:
                best, bd, be = c, d, even
            else:
                take = (d < bd) | ((d == bd) & even & ~be)
                best, bd, be = np.where(take, c, best), np.where(take, d, bd), np.where(take, even, be)
        if unsure.any():
            index = np.flatnonzero(unsure)
            self.scalar += index.size
            best = best.copy()
            best[index] = [self.rec.scale.to_normalized(float(x)) for x in p[index]]
        return best

    def skew(self, p):
        if self.kind == "linear":
            return p
        if self.kind == "quadratic":
            return p * p
        if self.kind == "quartic":
            s = p * p
            return s * s
        return np.sqrt(np.maximum(p, 0.0))

    def unskew(self, y):
        if self.kind == "linear":
            return y
        if self.kind == "quadratic":
            return np.sqrt(np.maximum(y, 0.0))
        if self.kind == "quartic":
            return np.sqrt(np.maximum(np.sqrt(np.maximum(y, 0.0)), 0.0))
        return y * y

    def disp(self, p):
        return self.mul * self.skew(p) + self.off

    def inv(self, d):
        v = self.unskew((d - self.off) / self.mul)
        p = np.minimum(np.maximum(v.astype(np.float32).astype(np.float64), self.mn), self.mx)
        return half_away(p) if self.ix else p

    def paths(self, r):
        p0 = self.S(r)
        d0 = self.disp(p0)
        p1 = self.inv(d0)
        p2 = self.S(self.N(p1))
        held = self.S(self.N(p0))
        return p0, d0, p1, p2, held

    def share(self, p0, d0):
        """Return where D0 is the display of the float32 just below or just above P0."""
        p32 = p0.astype(np.float32)
        below = np.nextafter(p32, np.float32(-np.inf)).astype(np.float64)
        above = np.nextafter(p32, np.float32(np.inf)).astype(np.float64)
        return (self.disp(below) == d0) | (self.disp(above) == d0)


def short(p0, held):
    """Return where P0 is a positive power of two and `held` the float32 just below it."""
    power = (p0 > 0) & (np.frexp(p0)[0] == 0.5)
    under = np.nextafter(p0.astype(np.float32), np.float32(-np.inf)).astype(np.float64)
    return power & (held == under)


def scalar_paths(rec, x):
    d = rec.from_normalized(x)
    return (
        rec.scale.to_preset(x),
        d,
        rec.to_preset(d),
        rec.scale.to_preset(rec.to_normalized(d)),
        rec.scale.to_preset(rec.scale.to_normalized(rec.scale.to_preset(x))),
    )


def check(name):
    """Return mismatches and scalar fallbacks of the vectorised paths against the scalar methods."""
    rec = parameter_map().find(ParameterId(name))
    v = Vec(rec)
    rng = np.random.default_rng(14)
    bits = rng.integers(0, 0x3F800001, 200_000, dtype=np.uint32)
    named = np.array([*ROWS, 0.0, 1.0, 2.0**-149], dtype=np.float32)
    sample = np.concatenate([bits.view(np.float32), named]).astype(np.float64)
    vector = v.paths(sample)
    shares = v.share(vector[0], vector[1])
    bad = 0
    for i in range(sample.size):
        try:
            scalar = scalar_paths(rec, float(sample[i]))
            shared = test_parameters._shares_its_display(rec, scalar[0])
        except Exception:  # noqa: BLE001 - a raise is a mismatch, counted
            bad += 1
            continue
        bad += (*scalar, shared) != (*(float(column[i]) for column in vector), bool(shares[i]))
    return sample.size, bad, v.scalar


def check_scale(name):
    """Return mismatches of the vectorised P0 and S(N(P0)) against the scalar methods."""
    rec = parameter_map().find(ParameterId(name))
    v = Vec(rec, scale_only=True)
    rng = np.random.default_rng(14)
    bits = rng.integers(0, 0x3F800001, 200_000, dtype=np.uint32)
    named = np.array([*ROWS, 0.0, 1.0, 2.0**-149], dtype=np.float32)
    sample = np.concatenate([bits.view(np.float32), named]).astype(np.float64)
    p0 = v.S(sample)
    held = v.S(v.N(p0))
    bad = 0
    for i in range(sample.size):
        x = float(sample[i])
        preset = rec.scale.to_preset(x)
        bad += (preset, rec.scale.to_preset(rec.scale.to_normalized(preset))) != (p0[i], held[i])
    return sample.size, bad, v.scalar

def count(name):
    """Count every float32 normalized value through one record's paths, by binade."""
    started = time.time()
    checked, mismatches, check_scalar = check(name)
    result = {"name": name, "checked": checked, "mismatches": mismatches,
              "check_scalar": check_scalar, "rows": [], "quiet": []}
    if mismatches:
        return result
    rec = parameter_map().find(ParameterId(name))
    v = Vec(rec)
    total = np.zeros(len(COLUMNS) + 1, dtype=np.int64)
    for label, lo, hi in SPANS:
        row = np.zeros(len(COLUMNS) + 1, dtype=np.int64)
        for start in range(lo, hi, CHUNK):
            r = np.arange(start, min(hi, start + CHUNK), dtype=np.uint32)
            r = r.view(np.float32).astype(np.float64)
            p0, d0, p1, p2, held = v.paths(r)
            shares = v.share(p0, d0)
            a1, hd, a2, a3 = v.disp(p1) != d0, held != p0, p2 != p1, p1 != p0
            pow2 = short(p0, held)
            steps = ordered(v.disp(p2)) - ordered(d0)
            s0, sp, sm = steps == 0, steps == 1, steps == -1
            far = ~(s0 | sp | sm)
            held_x = hd & ~pow2
            a3_x = a3 & ~shares
            set_x = ~s0 & ~((shares & (sp | sm)) | pow2)
            bound = a1 | held_x | a3_x | set_x
            share0 = shares & ~a3 & s0
            flags = (shares, share0, a1, hd, held_x, a2, a3, a3_x, s0, sp, sm, far, set_x, bound)
            row += [r.size, *(int(np.count_nonzero(f)) for f in flags)]
            if pow2.any():
                exponents = np.frexp(p0[pow2])[1] - 1
                result["short"] = sorted({*result.get("short", ()), *exponents.tolist()})
        total += row
        # a binade is printed where anything but n and SET0 is counted
        if row[1:9].any() or row[10:].any():
            result["rows"].append((label, row.tolist()))
        else:
            result["quiet"].append(label)
    result["total"] = total.tolist()
    result["scalar"] = v.scalar
    result["seconds"] = time.time() - started
    return result


def count_held(name):
    """Count HELD on every float32 normalized value of a record whose curve is not reproduced."""
    started = time.time()
    checked, mismatches, check_scalar = check_scale(name)
    result = {"name": name, "checked": checked, "mismatches": mismatches,
              "check_scalar": check_scalar, "rows": [], "quiet": []}
    if mismatches:
        return result
    rec = parameter_map().find(ParameterId(name))
    v = Vec(rec, scale_only=True)
    total = np.zeros(len(HELD_COLUMNS) + 1, dtype=np.int64)
    for label, lo, hi in SPANS:
        row = np.zeros(len(HELD_COLUMNS) + 1, dtype=np.int64)
        for start in range(lo, hi, CHUNK):
            r = np.arange(start, min(hi, start + CHUNK), dtype=np.uint32)
            r = r.view(np.float32).astype(np.float64)
            p0 = v.S(r)
            held = v.S(v.N(p0))
            hd = held != p0
            row += [r.size, int(np.count_nonzero(hd)), int(np.count_nonzero(hd & ~short(p0, held)))]
        total += row
        if row[1:].any():
            result["rows"].append((label, row.tolist()))
        else:
            result["quiet"].append(label)
    result["total"] = total.tolist()
    result["scalar"] = v.scalar
    result["seconds"] = time.time() - started
    return result


def count_any(job):
    kind, name = job
    return (kind, count(name) if kind == "full" else count_held(name))

    p, s, c = rec.parameter, rec.scale, rec.curve
    return (p.kind, s.kind, s.minimum, s.maximum, c.kind, c.multiply, c.offset, p.minimum, p.maximum)
