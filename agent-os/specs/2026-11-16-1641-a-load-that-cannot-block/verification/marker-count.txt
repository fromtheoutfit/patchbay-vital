"""Issue #20, the shipped remedy: the marker on every good load, and the two-load result with it.

For the start preset, the fixture and each of the 75 installed presets, on one Vital in this
process: these are texts Vital loads, the ones the integration tests load in their own process.
Each is loaded twice by the host as shipped, every load checked with the marker, and twice by the
same host with the check off, as a load was made before issue #20; the plugin's state after each
is read, and the two compared: all 903 floats and every section. Under two histories:

  start   the start preset loaded twice before the text, in the same way as the text
  sets    every parameter set to an end of its range and a short render before the text, the
          history under which issue #8 found one load is not enough

With the check on, the marker is read after each load: it reads the text's value where the load
was made. The check is turned off by telling the host it sought the marker and found none, which
is what it does where the map is not the plugin's.
"""

import importlib.metadata
import platform
import time

import common as c
from patchbay import ParameterId
from patchbay_vital import host as host_module
from patchbay_vital.errors import PluginStateError
from patchbay_vital.host import DEFAULT_PATH, MARKER, VitalHost

SHORT = 4_800
ENDS = {
    record.id: 1.0 if record.scale.to_normalized(record.preset_default) < 0.5 else 0.0  # noqa: PLR2004
    for record in c.MAP.parameters
}


def head(title: str) -> None:
    print(f"\n== {title}", flush=True)


def checking(host: VitalHost, on: bool) -> None:
    """Turn the host's marker check on, by having it seek the marker again, or off."""
    host._marker_sought = not on  # noqa: SLF001 - the probe's reach, to compare with and without
    host._marker_found = None  # noqa: SLF001


def history(host: VitalHost, which: str) -> tuple[float, float]:
    """Make a history before a text; the milliseconds its sets took, and its render."""
    if which == "start":
        host.load_state(c.START)
        host.load_state(c.START)
        return 0.0, 0.0
    began = time.perf_counter()
    for parameter_id, normalized in ENDS.items():
        host.set(ParameterId(parameter_id), normalized)
    sets = time.perf_counter()
    host.render([], SHORT, 48_000)
    return (sets - began) * 1000.0, (time.perf_counter() - sets) * 1000.0


def main() -> None:
    host = VitalHost()
    record = c.MAP.find(MARKER)
    print(f"Vital {host.version} at {DEFAULT_PATH}, through {host.host}")
    print(f"machine {platform.machine()}, Python {platform.python_version()}, numpy {importlib.metadata.version('numpy')}")
    print(f"the watch on a state write here: {type(host_module._ALERTS).__name__}; the marker: {MARKER}, host id {record.host_id}, a {record.parameter.kind} from {record.scale.minimum:g} to {record.scale.maximum:g}, its default {record.preset_default:g}")  # noqa: SLF001
    texts = c.good_texts()
    print(f"the texts: the start preset, the fixture and {len(texts) - 2} installed presets, {len(texts)} in all")
    for which in ("start", "sets"):
        head(f"the history {which!r}: each text loaded twice with the check, and twice without it")
        rows = []
        moved = loads = raised = 0
        with_ms, without_ms, sets_ms, render_ms = [], [], [], []
        for name, text in texts:
            text_floats = c.floats_of(text)
            target = record.scale.to_normalized(text_floats.get(str(record.preset_key), record.preset_default))
            states = {}
            for on in (True, False):
                checking(host, on)
                took_sets, took_render = history(host, which)
                sets_ms.append(took_sets)
                render_ms.append(took_render)
                for _ in range(2):
                    began = time.perf_counter()
                    try:
                        host.load_state(text)
                    except PluginStateError as error:
                        raised += 1
                        print(f"  RAISED on {name}: {error}")
                    (with_ms if on else without_ms).append((time.perf_counter() - began) * 1000.0)
                    if on:
                        loads += 1
                        moved += host.get(MARKER) == target
                states[on] = host.state()
            checked, plain = c.floats_of(states[True]), c.floats_of(states[False])
            floats_differ = sum(checked.get(key) != plain.get(key) for key in c.KEYS)
            sections_differ = c.sections_off(c.sections_of(states[True]), c.sections_of(states[False]))
            held, lacking = c.floats_off(checked, text_floats)
            rows.append((len(text), floats_differ, sections_differ, len(held) + len(lacking), name))
        checking(host, True)
        print("      bytes  floats of 903 that differ, with the check and without  sections that differ  floats not the text's, with the check   the text")
        for size, floats_differ, sections_differ, off, name in sorted(rows):
            print(f"  {size:>9}  {floats_differ:>58}  {sections_differ!s:>20}  {off:>37}   {name}")
        count = len(rows)
        print(f"\nloads made with the check: {loads}; the marker read the text's value after the load on {moved}; a load raised: {raised}")
        print(f"all 903 floats the same with the check as without: {sum(row[1] == 0 for row in rows)} of {count} texts; every section the same: {sum(not row[2] for row in rows)} of {count}")
        off = [row[3] for row in rows]
        print(f"floats not the text's after the two checked loads: none on {sum(n == 0 for n in off)} texts, {sum(off)} floats on {sum(n > 0 for n in off)} (issue #8 counted none on 66 and 46 on 11, with no check)")
        print(f"a load's time with the check: {c.spread(with_ms)} ms; without: {c.spread(without_ms)} ms; the check's share, the medians apart: {(sorted(with_ms)[len(with_ms) // 2] - sorted(without_ms)[len(without_ms) // 2]) * 1000:.0f} microseconds")
        if which == "sets":
            print(f"the history itself, a labelled sample: the 903 sets {c.spread(sets_ms)} ms; the render of {SHORT} samples {c.spread(render_ms)} ms")


if __name__ == "__main__":
    main()

"""What the probes of issue #20 share: the texts, the reads of the plugin, and the checks.

Run from the repository root. Nothing here loads a text into Vital; the callers do, and a load
that may block is made in a child the caller can kill.

import ctypes
import json
import statistics
import sys
import time
from collections.abc import Callable
from pathlib import Path

from patchbay import ParameterId

from patchbay_vital.host import _LIBC, VitalHost, _floats
from patchbay_vital.parameters import parameter_map
from patchbay_vital.start import start_text

MAP = parameter_map()
START = start_text(MAP).encode("utf-8")
PRESETS = Path.home() / "Music" / "Vital"
ROOT = Path.cwd()
FIXTURE = ROOT / "tests" / "data" / "init-media-replaced.vital"
MANIFEST = ROOT / "tests" / "data" / "factory-manifest.txt"
KEYS = tuple(str(record.preset_key) for record in MAP.parameters)
DEFAULTS = {str(record.preset_key): record.preset_default for record in MAP.parameters}

"""Seconds a load is given, counted from the child saying it is about to make it."""
MARKERS = ("filter_1_cutoff", "filter_1_on")
"""The parameters planted before a load: one continuous, one a switch; Vital upgrades neither."""


def good_texts() -> list[tuple[str, bytes]]:
    """The 77 texts Vital is known to load: the start preset, the fixture, the 75 installed."""
    manifest = [
        line
        for line in MANIFEST.read_text(encoding="utf-8").splitlines()
        if line and not line.startswith("#")
    ]
    texts = [("the start preset", START), ("the fixture", FIXTURE.read_bytes())]
    texts += [(name, (PRESETS / name).read_bytes()) for name in manifest]
    return texts


def variant(top: dict[str, object] | None = None, **settings: object) -> bytes:
    """The start preset with settings replaced (None removes one) and top-level keys set."""
    changed = json.loads(START)
    for key, value in settings.items():
        if value is None:
            del changed["settings"][key]
        else:
            changed["settings"][key] = value
    changed.update(top or {})
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


def no_sections() -> bytes:
    changed = json.loads(START)
    for name in ("wavetables", "lfos", "sample", "custom_warps", "random_values"):
        del changed["settings"][name]
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


def routed(source: str, destination: str) -> bytes:
    changed = json.loads(START)
    changed["settings"]["modulations"][0] = {"source": source, "destination": destination}
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


_LFO = {"name": "x", "num_points": 2, "points": "x", "powers": [0.0, 0.0], "smooth": False}
_SAMPLE = {"length": "x", "name": "x", "sample_rate": 44100, "samples": ""}

CASES: dict[str, Callable[[], bytes]] = {
    # The 20 texts of the instrument spec's probes.txt, section 5, built as they were there.
    "the start preset rewritten by json.dumps (the control)": lambda: variant(),
    "no section at all": no_sections,
    "wavetables an empty array": lambda: variant(wavetables=[]),
    "wavetables an array of one empty object": lambda: variant(wavetables=[{}]),

def floats_of(text: bytes) -> dict[str, float]:
    """A text's float settings by key, read as the package reads JSON; none where it has none."""
    try:
        return dict(_floats(text))
    except Exception:  # noqa: BLE001  a probe's read of a text that may be malformed
        return {}


def sections_of(text: bytes) -> dict[str, object]:
    """A text's settings that are not numbers: its sections, parsed."""
    settings = json.loads(text).get("settings", {})
    if not isinstance(settings, dict):
        return {}
    return {
        key: value
        for key, value in settings.items()
        if isinstance(value, (dict, list, str))
    }


def environment() -> bytes:
    """The process's floating-point environment, as fegetenv gives it."""
    saved = ctypes.create_string_buffer(512)
    _LIBC.fegetenv(saved)
    return saved.raw


def subnormals_kept() -> bool:
    """Whether a subnormal result is still a subnormal, so flush-to-zero is off."""
    tiny = sys.float_info.min
    return tiny / 4.0 != 0.0


def plant(host: VitalHost, text: dict[str, float]) -> dict[str, tuple[float, float]]:
    """Set each marker to the end of its range the text's value is not near.

    Returns, by marker, the normalized value planted and the one the text's value has, the
    parameter's default where the text lacks the key.
    """
    planted = {}
    for key in MARKERS:
        record = MAP.find(ParameterId(key))
        target = record.scale.to_normalized(text.get(key, record.preset_default))
        normalized = 0.0 if target > 0.5 else 1.0
        host.set(ParameterId(key), normalized)
        planted[key] = (normalized, target)
    return planted


def markers_read(host: VitalHost, planted: dict[str, tuple[float, float]]) -> tuple[int, int]:
    """How many markers moved off the planted value, and how many read the text's exactly."""
    moved = exact = 0
    for key, (normalized, target) in planted.items():
        now = host.get(ParameterId(key))
        moved += now != normalized
        exact += now == target
    return moved, exact


def floats_off(state: dict[str, float], text: dict[str, float]) -> tuple[list[str], list[str]]:
    """The map's keys whose float in the state is not the text's, and, for keys the text lacks,
    those not at the parameter's default."""
    held = [key for key in KEYS if key in text and state.get(key) != text[key]]
    lacking = [key for key in KEYS if key not in text and state.get(key) != DEFAULTS[key]]
    return held, lacking


def sections_off(state: dict[str, object], text: dict[str, object]) -> list[str]:
    """The names of the text's sections the state does not hold as the text does."""
    return sorted(name for name in text if state.get(name) != text[name])


def timed(call: Callable[[], object]) -> tuple[float, object]:
    """Run a call; return its milliseconds and what it gave."""
    start = time.perf_counter()
    result = call()
    return (time.perf_counter() - start) * 1000.0, result


def spread(values: list[float]) -> str:
    """A sample's least, median and greatest, to one decimal."""
    return (
        f"min {min(values):.1f}, median {statistics.median(values):.1f}, max {max(values):.1f}"
    )
