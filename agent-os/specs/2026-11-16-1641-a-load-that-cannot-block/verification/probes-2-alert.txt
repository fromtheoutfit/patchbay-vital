```
and the front application 20 times a second, so what showed is measured.

def head(title: str) -> None:
    print(f"\n== {title}", flush=True)


class Screen:
    """The poller, and what it saw of one process."""

    def __init__(self) -> None:
        self.process = subprocess.Popen(  # noqa: S603  the probe's own interpreter and script
            [sys.executable, "-u", "-B", POLLER, str(WINDOWS)],
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
        )
        assert self.process.stdout is not None
        self.process.stdout.readline()

    def polls(self) -> list[dict[str, object]]:
        with WINDOWS.open(encoding="utf-8") as log:
            return [json.loads(line) for line in log if line.endswith("\n")]

    def of(self, pid: int, start: float, zero: float | None, end: float | None) -> str:
        """What was on screen of a process from `start` on, in ms from `zero`."""
        time.sleep(SETTLE)
        rows = [poll for poll in self.polls() if poll["t"] >= start]
        seen = [
            (poll["t"], [window for window in poll["windows"] if window[0] == pid])
            for poll in rows
        ]
        shown = [(at, windows) for at, windows in seen if windows]
        front = [poll["t"] for poll in rows if poll["front"] == pid]
        fronts = {poll["front"] for poll in rows}
        others = len(fronts - {pid})
        said = f"{len(rows)} polls; "
        if not shown:
            said += "no window of the child was on screen at any of them"
        else:
            origin = zero if zero is not None else start
            largest = max((window for _, windows in shown for window in windows), key=lambda w: w[2] * w[3])
            said += (
                f"a window of the child on screen at {len(shown)} of them, from "
                f"{(shown[0][0] - origin) * 1000:+.0f} ms to {(shown[-1][0] - origin) * 1000:+.0f} ms "
                f"after the load began, at most {max(len(windows) for _, windows in shown)} at once, "
                f"the largest {largest[2]} by {largest[3]} at layer {largest[1]}"
            )
            if end is not None:
                said += f"; last seen {(shown[-1][0] - end) * 1000:+.0f} ms from the kill"
        said += (
            f"; the child was the front application at {len(front)} polls"
            f"; distinct front applications seen besides the child: {others}"
        )
        return said

    def stop(self) -> None:
        assert self.process.stdin is not None
        self.process.stdin.close()
        self.process.wait(timeout=10)


def stamp(lines: list[str], prefix: str) -> float | None:
    """The wall-clock time of the last line that starts with a prefix, as the child printed it."""
    for line in reversed(lines):
        if line.startswith(prefix + " at "):
            return float(line.removeprefix(prefix + " at ").split(":")[0].split(" ")[0])
    return None


def pid_of(lines: list[str], child: Child) -> int:
    return next((int(line.split()[1]) for line in lines if line.startswith("PID ")), child.process.pid)


def result_of(lines: list[str]) -> dict[str, object] | None:
    for line in lines:

    return None
```

```
"""A child of probe run 2: the alert itself, and a load made under each of the two watches.

usage: child2.py <mode> [<argument>...] <the time it was started>

  baseline <case>            the text loaded with no watch, a thread ticking and an alarm set
  variant <watch> <case>     the text loaded under a watch, then what the plugin holds and plays
  control <watch>            each of the 77 good texts loaded twice under a watch
  offmain <watch|none> <case>  the text loaded from a thread that is not the main one

<watch> is `timer` or `answer` (objc.py). Before a load that may block the child says ABOUT TO
LOAD and after it RETURNED, each with its wall-clock time; its parent kills it when the second
does not follow in time, and the dead-man timer of child.py ends it whatever its parent does.
"""

import faulthandler
import json
import os
import signal
import sys
import threading
import time
from pathlib import Path

import numpy as np

import common as c
import objc
from child import ABOUT_TO_LOAD, DEAD_MAN, RETURNED
from patchbay_vital.errors import VitalError

HERE = Path(__file__).resolve().parent
RATE = 48_000
SAMPLES = 96_000
HELD = slice(9_600, 67_200)
"""0.2 s to 1.4 s of a note held from 0 to 1.5 s: the window of the instrument's own measure."""
NOTE = [(bytes([0x90, 60, 100]), 0.5 / RATE), (bytes([0x80, 60, 0]), 72_000.5 / RATE)]


def twice(host: c.VitalHost, text: bytes) -> None:
    host.load_state(text)
    host.load_state(text)


def rms_of_a_note(host: c.VitalHost) -> float:
    """The RMS of the channels' mean over the held part of one C4 at velocity 100."""
    audio = host.render(NOTE, SAMPLES, RATE)
    held = audio.mean(axis=0)[HELD].astype(np.float64)
    return float(np.sqrt(np.mean(held * held)))


def watched(
    host: c.VitalHost, watch: object, text: bytes, what: str | None
) -> tuple[float, bool, str, float]:
    """One load under a watch: its milliseconds, whether the watch fired, what was raised, and
    the microseconds arming the watch and taking it down took. Said when `what` is given."""
    if what is not None:
        print(f"{ABOUT_TO_LOAD} at {time.time()!r}: {what}", flush=True)
    faulthandler.dump_traceback_later(DEAD_MAN, exit=True)
    raised = "no error"
    before = time.perf_counter()
    watch.arm()  # type: ignore[attr-defined]
    armed = time.perf_counter()
    try:
        host.load_state(text)
    except VitalError as error:
        raised = f"{type(error).__name__}: {str(error)[:100]}"
    finally:
        loaded = time.perf_counter()
        fired = watch.disarm()  # type: ignore[attr-defined]
        down = time.perf_counter()
        faulthandler.cancel_dump_traceback_later()
    taken = (loaded - armed) * 1000.0
    if what is not None:
        print(f"{RETURNED} at {time.time()!r} in {taken:.1f} ms", flush=True)
    return taken, fired, raised, ((armed - before) + (down - loaded)) * 1e6


def baseline(label: str) -> None:
    """The text loaded with no watch, while a thread ticks and an alarm is set."""
    notes = HERE / "out" / f"ticks-{os.getpid()}.txt"
    descriptor = os.open(notes, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o644)

    def note(what: str) -> None:
        os.write(descriptor, f"{what} {time.time()!r}\n".encode())

    def tick() -> None:
        while True:
            note("tick")
            time.sleep(0.1)

    signal.signal(signal.SIGALRM, lambda *_: note("alarm"))
    host = c.VitalHost()
    twice(host, c.START)
    text = c.CASES[label]()
    threading.Thread(target=tick, daemon=True).start()
    signal.setitimer(signal.ITIMER_REAL, 0.25, 0.25)
    time.sleep(1.0)
    print(f"NOTES {notes.name}", flush=True)
    note("load-begins")
    began = time.time()
    print(f"{ABOUT_TO_LOAD} at {began!r}: the text, no watch", flush=True)
    faulthandler.dump_traceback_later(DEAD_MAN, exit=True)
    host.load_state(text)
    faulthandler.cancel_dump_traceback_later()
    note("load-ended")
    ended = time.time()
    print(f"{RETURNED} at {ended!r} in {(ended - began) * 1000:.1f} ms", flush=True)


def variant(kind: str, label: str) -> None:
    """The text loaded under a watch; then what the plugin holds, and whether it still plays."""
    watch = objc.WATCHES[kind]()
    host = c.VitalHost()
    twice(host, c.START)
```

```
"""A child of a probe run: it hosts its own Vital and makes the loads its parent must not.

usage: child.py <mode> [<argument>...] <the time it was started, as the parent's time.time()>

  fresh <dead-man seconds>    read one text from stdin, load it twice, print one line of marks
  kept <dead-man seconds>     load each text sent on stdin twice, one line a text, until stdin ends
  detect <case>               section 5's case loaded twice over the start preset, with markers

Before a load that may block it says ABOUT TO LOAD and after it RETURNED; its parent kills it
when the second does not follow the first in time. It also ends itself: a dead-man timer that
needs no GIL (faulthandler's) exits the process when a load outlives it, parent or no parent.
"""

import time

MAIN = time.time()

import ctypes  # noqa: E402
import faulthandler  # noqa: E402
import json  # noqa: E402
import struct  # noqa: E402
import sys  # noqa: E402

ABOUT_TO_LOAD = "ABOUT TO LOAD"
RETURNED = "RETURNED"
DEAD_MAN = 8.0
"""Seconds a load that may block is given before this process ends itself."""


def _environment(library: ctypes.CDLL) -> bytes:
    saved = ctypes.create_string_buffer(512)
    library.fegetenv(saved)
    return saved.raw


def _subnormals_kept() -> bool:
    tiny = sys.float_info.min
    return tiny / 4.0 != 0.0


def _load(host: object, text: bytes, dead_man: float) -> float:
    """One load under the dead-man timer; its milliseconds."""
    faulthandler.dump_traceback_later(dead_man, exit=True)
    start = time.perf_counter()
    host.load_state(text)  # type: ignore[attr-defined]
    taken = (time.perf_counter() - start) * 1000.0
    faulthandler.cancel_dump_traceback_later()
    return taken


def said(host: object, text: bytes, what: str) -> float:
    """One load that may block, said before and after."""
    print(f"{ABOUT_TO_LOAD}: {what}", flush=True)
    taken = _load(host, text, DEAD_MAN)
    print(f"{RETURNED} in {taken:.1f} ms", flush=True)
    return taken


def fresh(dead_man: float, spawned: float) -> None:
    """What a trial in a child of its own costs: only the host is imported, as a remedy would."""
    from patchbay_vital.host import _LIBC, VitalHost  # noqa: PLC0415

    marks: dict[str, object] = {"spawned": spawned, "main": MAIN, "imported": time.time()}
    host = VitalHost()
    marks["hosted"] = time.time()
    text = sys.stdin.buffer.read()
    marks["read"] = time.time()
    before = _environment(_LIBC)
    marks["first_ms"] = _load(host, text, dead_man)
    marks["second_ms"] = _load(host, text, dead_man)
    marks["loaded"] = time.time()
    marks["environment_kept"] = _environment(_LIBC) == before
    marks["subnormals_kept"] = _subnormals_kept()
    marks["done"] = time.time()
    print(json.dumps(marks), flush=True)


def kept(dead_man: float, spawned: float) -> None:
    """A child kept up: each text sent as an 8-byte length and its bytes, one line back."""
    from patchbay_vital.host import _LIBC, VitalHost  # noqa: PLC0415

    imported = time.time()
    host = VitalHost()
    ready = {"spawned": spawned, "main": MAIN, "imported": imported, "hosted": time.time()}
    print(json.dumps(ready), flush=True)
    stdin = sys.stdin.buffer
    while True:
        head = stdin.read(8)
        if len(head) < 8:  # noqa: PLR2004
            return
        (size,) = struct.unpack(">Q", head)
        text = stdin.read(size)
        before = _environment(_LIBC)
        first = _load(host, text, dead_man)
        second = _load(host, text, dead_man)
        reply = {
            "first_ms": first,
            "second_ms": second,
            "environment_kept": _environment(_LIBC) == before,
            "subnormals_kept": _subnormals_kept(),
        }
        print(json.dumps(reply), flush=True)


def detect(label: str) -> None:
    """A text section 5 saw return wrong: what each check through the diagnostic read sees."""
    import common as c  # noqa: PLC0415

    host = c.VitalHost()
    host.load_state(c.START)
    host.load_state(c.START)
    text = c.CASES[label]()
    text_floats, text_sections = c.floats_of(text), c.sections_of(text)
    planted = c.plant(host, text_floats)
    before = host.state()
    before_floats, before_sections = c.floats_of(before), c.sections_of(before)
    said(host, text, "the first load")
    said(host, text, "the second load")
    moved, exact = c.markers_read(host, planted)
    after = host.state()
    after_floats, after_sections = c.floats_of(after), c.sections_of(after)
    changed = sorted(k for k in c.KEYS if after_floats.get(k) != before_floats.get(k))
    held, lacking = c.floats_off(after_floats, text_floats)
    result = {
        "markers": len(planted),
        "moved": moved,
        "exact": exact,
        "floats_changed": len(changed),
        "floats_not_the_text_s": len(held) + len(lacking),
        "first_floats_not_the_text_s": (held + lacking)[:4],
        "sections_changed": sorted(
            k for k in before_sections if after_sections.get(k) != before_sections[k]
        ),
        "sections_not_the_text_s": c.sections_off(after_sections, text_sections),
    }
    print(f"RESULT {json.dumps(result)}", flush=True)


if __name__ == "__main__":
    mode, *arguments, started = sys.argv[1:]
    if mode == "fresh":
        fresh(float(arguments[0]), float(started))
    elif mode == "kept":
```

```
"""The Objective-C runtime and AppKit through ctypes, and two ways to answer Vital's alert.

Standard library only. Each message is sent through objc_msgSend under a fixed prototype, as
arm64 needs. Both watches are armed on the main thread, before a load, and taken down after it:

- `ModalTimer`, the watchdog as a timer: an NSTimer on the main run loop in the modal panel's
  run-loop mode alone, whose target is the application and whose selector is abortModal. It can
  fire only while a modal loop runs; whether it fired is read from the timer afterwards.
- `AlertAnswered`, the alert answered before it opens: -[NSAlert runModal] is replaced by a
  function that records the call and the alert's two texts and returns the first button's code
  at once; the method is put back afterwards.
"""

import ctypes
import ctypes.util

_objc = ctypes.CDLL(ctypes.util.find_library("objc"))
_appkit = ctypes.CDLL(ctypes.util.find_library("AppKit"))
_foundation = ctypes.CDLL(ctypes.util.find_library("Foundation"))

_objc.objc_getClass.restype = ctypes.c_void_p
_objc.objc_getClass.argtypes = [ctypes.c_char_p]
_objc.sel_registerName.restype = ctypes.c_void_p
_objc.sel_registerName.argtypes = [ctypes.c_char_p]
_objc.class_getInstanceMethod.restype = ctypes.c_void_p
_objc.class_getInstanceMethod.argtypes = [ctypes.c_void_p, ctypes.c_void_p]
_objc.method_setImplementation.restype = ctypes.c_void_p
_objc.method_setImplementation.argtypes = [ctypes.c_void_p, ctypes.c_void_p]
_objc.method_getImplementation.restype = ctypes.c_void_p
_objc.method_getImplementation.argtypes = [ctypes.c_void_p]
_objc.objc_autoreleasePoolPush.restype = ctypes.c_void_p
_objc.objc_autoreleasePoolPop.argtypes = [ctypes.c_void_p]

FIRST_BUTTON = 1000
"""NSAlertFirstButtonReturn."""


def cls(name: str) -> int:
    found = _objc.objc_getClass(name.encode())
    if not found:
        raise LookupError(name)
    return int(found)


def sel(name: str) -> int:
    return int(_objc.sel_registerName(name.encode()))


def send(receiver: int | None, selector: str, restype: object = ctypes.c_void_p, *typed: tuple[object, object]) -> object:
    """Send one message: each argument given as (its ctypes type, its value)."""
    prototype = ctypes.CFUNCTYPE(restype, ctypes.c_void_p, ctypes.c_void_p, *(kind for kind, _ in typed))
    function = prototype(("objc_msgSend", _objc))
    return function(receiver, sel(selector), *(value for _, value in typed))


def text_of(string: int | None) -> str:
    """An NSString's text, empty for nil."""
    if not string:
        return ""
    raw = send(string, "UTF8String", ctypes.c_char_p)
    return raw.decode("utf-8", "replace") if isinstance(raw, bytes) else ""


def application() -> int:
    """The shared NSApplication, made if the process has none."""
    return int(send(cls("NSApplication"), "sharedApplication"))  # type: ignore[arg-type]


def modal_mode() -> int:
    """NSModalPanelRunLoopMode, the run-loop mode a modal panel's loop runs in."""
    return int(ctypes.c_void_p.in_dll(_appkit, "NSModalPanelRunLoopMode").value or 0)


class ModalTimer:
    """An NSTimer in the modal panel's run-loop mode alone that calls abortModal."""

    name = "the watchdog as a timer"

    def __init__(self) -> None:
        self._timer: int | None = None
        self.fired = 0

    def arm(self) -> None:
        pool = _objc.objc_autoreleasePoolPush()
        timer = send(
            cls("NSTimer"),
            "timerWithTimeInterval:target:selector:userInfo:repeats:",
            ctypes.c_void_p,
            (ctypes.c_double, 0.0),
            (ctypes.c_void_p, application()),
            (ctypes.c_void_p, sel("abortModal")),
            (ctypes.c_void_p, None),
            (ctypes.c_bool, False),
        )
        send(timer, "retain")  # type: ignore[arg-type]
        loop = send(cls("NSRunLoop"), "mainRunLoop")
        send(loop, "addTimer:forMode:", None, (ctypes.c_void_p, timer), (ctypes.c_void_p, modal_mode()))  # type: ignore[arg-type]
        _objc.objc_autoreleasePoolPop(pool)
        self._timer = int(timer)  # type: ignore[arg-type]

    def disarm(self) -> bool:
        """Take the timer down; return whether it fired, which a timer that does not repeat
        shows by being no longer valid."""
        assert self._timer is not None
        fired = not send(self._timer, "isValid", ctypes.c_bool)
        send(self._timer, "invalidate", None)
        send(self._timer, "release", None)
        self._timer = None
        self.fired += fired
        return fired


_ANSWER = ctypes.CFUNCTYPE(ctypes.c_long, ctypes.c_void_p, ctypes.c_void_p)


class AlertAnswered:
    """-[NSAlert runModal] replaced, for the length of a load, by a function that returns at once."""

    name = "the alert answered before it opens"

    def __init__(self) -> None:
        self.alerts: list[tuple[str, str]] = []
        self._function = _ANSWER(self._answer)
        self._method = _objc.class_getInstanceMethod(cls("NSAlert"), sel("runModal"))
        self._original: int | None = None
        self._before = 0
        self.fired = 0

    def _answer(self, alert: int, _selector: int) -> int:
        try:
            message = text_of(send(alert, "messageText"))  # type: ignore[arg-type]
            informative = text_of(send(alert, "informativeText"))  # type: ignore[arg-type]
        except Exception as error:  # noqa: BLE001  nothing may leave a callback
            message, informative = f"<{type(error).__name__}>", ""
        self.alerts.append((message, informative))
        return FIRST_BUTTON

    def arm(self) -> None:
        self._before = len(self.alerts)
        self._original = _objc.method_setImplementation(
            self._method, ctypes.cast(self._function, ctypes.c_void_p)
        )

    def disarm(self) -> bool:
        """Put the method back; return whether an alert was answered since it was armed."""
        assert self._original is not None
        _objc.method_setImplementation(self._method, self._original)
        self._original = None
        fired = len(self.alerts) > self._before
        self.fired += fired
        return fired

    def in_place(self) -> bool:
        """Whether the method's implementation is the one this object put there."""
        now = _objc.method_getImplementation(self._method)
        return now == ctypes.cast(self._function, ctypes.c_void_p).value


WATCHES = {"timer": ModalTimer, "answer": AlertAnswered}
```

```
"""A second process that watches the screen while a probe run's children load.

usage: poller.py <out file>

20 times a second it writes one JSON line: the time, the front application's pid (lsappinfo),
and every window on screen as [owner pid, layer, width, height] (CGWindowListCopyWindowInfo,
which gives a window's owner and bounds without the Screen Recording permission). It ends when
its input closes. It loads no plugin and opens no window.
"""

import ctypes
import ctypes.util
import json
import subprocess
import sys
import threading
import time

_cg = ctypes.CDLL(ctypes.util.find_library("CoreGraphics"))
_cf = ctypes.CDLL(ctypes.util.find_library("CoreFoundation"))

_cg.CGWindowListCopyWindowInfo.restype = ctypes.c_void_p
_cg.CGWindowListCopyWindowInfo.argtypes = [ctypes.c_uint32, ctypes.c_uint32]
_cf.CFArrayGetCount.restype = ctypes.c_long
_cf.CFArrayGetCount.argtypes = [ctypes.c_void_p]
_cf.CFArrayGetValueAtIndex.restype = ctypes.c_void_p
_cf.CFArrayGetValueAtIndex.argtypes = [ctypes.c_void_p, ctypes.c_long]
_cf.CFDictionaryGetValue.restype = ctypes.c_void_p
_cf.CFDictionaryGetValue.argtypes = [ctypes.c_void_p, ctypes.c_void_p]
_cf.CFNumberGetValue.restype = ctypes.c_bool
_cf.CFNumberGetValue.argtypes = [ctypes.c_void_p, ctypes.c_int, ctypes.c_void_p]
_cf.CFRelease.argtypes = [ctypes.c_void_p]


class _Rect(ctypes.Structure):
    _fields_ = (("x", ctypes.c_double), ("y", ctypes.c_double), ("w", ctypes.c_double), ("h", ctypes.c_double))


_cg.CGRectMakeWithDictionaryRepresentation.restype = ctypes.c_bool
_cg.CGRectMakeWithDictionaryRepresentation.argtypes = [ctypes.c_void_p, ctypes.POINTER(_Rect)]

ON_SCREEN_ONLY = 1
LONG_LONG = 11
"""kCFNumberLongLongType."""
PERIOD = 0.05


def _key(name: str) -> int:
    return int(ctypes.c_void_p.in_dll(_cg, name).value or 0)


OWNER, LAYER, BOUNDS = _key("kCGWindowOwnerPID"), _key("kCGWindowLayer"), _key("kCGWindowBounds")


def _number(dictionary: int, key: int) -> int:
    value = ctypes.c_longlong(0)
    number = _cf.CFDictionaryGetValue(dictionary, key)
    if number:
        _cf.CFNumberGetValue(number, LONG_LONG, ctypes.byref(value))
    return int(value.value)


def windows() -> list[list[int]]:
    """Every window on screen: its owner's pid, its layer, its width and its height."""
    found = []
    array = _cg.CGWindowListCopyWindowInfo(ON_SCREEN_ONLY, 0)
    if not array:
        return found
    for index in range(_cf.CFArrayGetCount(array)):
        entry = _cf.CFArrayGetValueAtIndex(array, index)
        rect = _Rect()
        bounds = _cf.CFDictionaryGetValue(entry, BOUNDS)
        if bounds:
            _cg.CGRectMakeWithDictionaryRepresentation(bounds, ctypes.byref(rect))
        found.append([_number(entry, OWNER), _number(entry, LAYER), round(rect.w), round(rect.h)])
    _cf.CFRelease(array)
    return found


def front() -> int | None:
    """The front application's pid, as lsappinfo gives it."""
    asn = subprocess.run(["/usr/bin/lsappinfo", "front"], capture_output=True, text=True, check=False).stdout.strip()  # noqa: S603
    if not asn:
        return None
    line = subprocess.run(["/usr/bin/lsappinfo", "info", "-only", "pid", asn], capture_output=True, text=True, check=False).stdout  # noqa: S603
    digits = line.strip().rpartition("=")[2]
    return int(digits) if digits.isdigit() else None


def main(out: str) -> None:
    done = threading.Event()
    threading.Thread(target=lambda: (sys.stdin.read(), done.set()), daemon=True).start()
    with open(out, "w", encoding="utf-8") as log:  # noqa: PTH123
        print("ready", flush=True)
        while not done.is_set():
            started = time.time()
            log.write(json.dumps({"t": started, "front": front(), "windows": windows()}) + "\n")
            log.flush()
            time.sleep(max(0.0, PERIOD - (time.time() - started)))


if __name__ == "__main__":
    main(sys.argv[1])
```

```
"""What the probes of issue #20 share: the texts, the reads of the plugin, and the checks.

Run from the repository root. Nothing here loads a text into Vital; the callers do, and a load
that may block is made in a child the caller can kill.
"""

import ctypes
import json
import statistics
import sys
import time
from collections.abc import Callable
from pathlib import Path

from patchbay import ParameterId

from patchbay_vital.host import _LIBC, VitalHost, _floats
from patchbay_vital.parameters import parameter_map
from patchbay_vital.start import start_text

MAP = parameter_map()
START = start_text(MAP).encode("utf-8")
PRESETS = Path.home() / "Music" / "Vital"
ROOT = Path.cwd()
FIXTURE = ROOT / "tests" / "data" / "init-media-replaced.vital"
MANIFEST = ROOT / "tests" / "data" / "factory-manifest.txt"
KEYS = tuple(str(record.preset_key) for record in MAP.parameters)
DEFAULTS = {str(record.preset_key): record.preset_default for record in MAP.parameters}
LIMIT = 3.0
"""Seconds a load is given, counted from the child saying it is about to make it."""
MARKERS = ("filter_1_cutoff", "filter_1_on")
"""The parameters planted before a load: one continuous, one a switch; Vital upgrades neither."""


def good_texts() -> list[tuple[str, bytes]]:
    """The 77 texts Vital is known to load: the start preset, the fixture, the 75 installed."""
    manifest = [
        line
        for line in MANIFEST.read_text(encoding="utf-8").splitlines()
        if line and not line.startswith("#")
    ]
    texts = [("the start preset", START), ("the fixture", FIXTURE.read_bytes())]
    texts += [(name, (PRESETS / name).read_bytes()) for name in manifest]
    return texts


def variant(top: dict[str, object] | None = None, **settings: object) -> bytes:
    """The start preset with settings replaced (None removes one) and top-level keys set."""
    changed = json.loads(START)
    for key, value in settings.items():
        if value is None:
            del changed["settings"][key]
        else:
            changed["settings"][key] = value
    changed.update(top or {})
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


def no_sections() -> bytes:
    changed = json.loads(START)
    for name in ("wavetables", "lfos", "sample", "custom_warps", "random_values"):
        del changed["settings"][name]
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


def routed(source: str, destination: str) -> bytes:
    changed = json.loads(START)
    changed["settings"]["modulations"][0] = {"source": source, "destination": destination}
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")


_LFO = {"name": "x", "num_points": 2, "points": "x", "powers": [0.0, 0.0], "smooth": False}

def floats_of(text: bytes) -> dict[str, float]:
    """A text's float settings by key, read as the package reads JSON; none where it has none."""
    try:
        return dict(_floats(text))
    except Exception:  # noqa: BLE001  a probe's read of a text that may be malformed
        return {}


def sections_of(text: bytes) -> dict[str, object]:
    """A text's settings that are not numbers: its sections, parsed."""
    settings = json.loads(text).get("settings", {})
    if not isinstance(settings, dict):
        return {}
    return {
        key: value
        for key, value in settings.items()
        if isinstance(value, (dict, list, str))
    }


def environment() -> bytes:
    """The process's floating-point environment, as fegetenv gives it."""
    saved = ctypes.create_string_buffer(512)
    _LIBC.fegetenv(saved)
    return saved.raw


def subnormals_kept() -> bool:
    """Whether a subnormal result is still a subnormal, so flush-to-zero is off."""
    tiny = sys.float_info.min
    return tiny / 4.0 != 0.0


def plant(host: VitalHost, text: dict[str, float]) -> dict[str, tuple[float, float]]:
    """Set each marker to the end of its range the text's value is not near.

    Returns, by marker, the normalized value planted and the one the text's value has, the
    parameter's default where the text lacks the key.
    """
    planted = {}
    for key in MARKERS:
        record = MAP.find(ParameterId(key))
        target = record.scale.to_normalized(text.get(key, record.preset_default))
        normalized = 0.0 if target > 0.5 else 1.0
        host.set(ParameterId(key), normalized)
        planted[key] = (normalized, target)
    return planted


def markers_read(host: VitalHost, planted: dict[str, tuple[float, float]]) -> tuple[int, int]:
    """How many markers moved off the planted value, and how many read the text's exactly."""
    moved = exact = 0
    for key, (normalized, target) in planted.items():
        now = host.get(ParameterId(key))
        moved += now != normalized
        exact += now == target
    return moved, exact


def floats_off(state: dict[str, float], text: dict[str, float]) -> tuple[list[str], list[str]]:
    """The map's keys whose float in the state is not the text's, and, for keys the text lacks,
    those not at the parameter's default."""
    held = [key for key in KEYS if key in text and state.get(key) != text[key]]
    lacking = [key for key in KEYS if key not in text and state.get(key) != DEFAULTS[key]]
    return held, lacking


def sections_off(state: dict[str, object], text: dict[str, object]) -> list[str]:
    """The names of the text's sections the state does not hold as the text does."""
    return sorted(name for name in text if state.get(name) != text[name])


def timed(call: Callable[[], object]) -> tuple[float, object]:
    """Run a call; return its milliseconds and what it gave."""
    start = time.perf_counter()
    result = call()
    return (time.perf_counter() - start) * 1000.0, result


def spread(values: list[float]) -> str:
    """A sample's least, median and greatest, to one decimal."""
    return (
        f"min {min(values):.1f}, median {statistics.median(values):.1f}, max {max(values):.1f}"
    )
```

```
"""How a probe run starts a child, reads it under a time limit, and kills what does not return."""

import queue
import subprocess
import sys
import threading
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
CHILD = str(HERE / "child.py")
ABOUT_TO_LOAD = "ABOUT TO LOAD"
RETURNED = "RETURNED"
START_UP = 60.0
"""Seconds a child is given for everything that is not a load that may block."""


class Child:
    """A child process whose lines are read under a time limit."""

    every: list["Child"] = []
    """Each child a run started, so the run can show at its end that none is left."""

    def __init__(self, *arguments: str, stdin: bool = False, script: str = CHILD) -> None:
        Child.every.append(self)
        self.started = time.time()
        self.killed: float | None = None
        self.process = subprocess.Popen(  # noqa: S603  the probe's own interpreter and script
            [sys.executable, "-u", "-B", script, *arguments, repr(self.started)],
            stdin=subprocess.PIPE if stdin else subprocess.DEVNULL,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
        )
        self._lines: queue.Queue[bytes | None] = queue.Queue()
        threading.Thread(target=self._read, daemon=True).start()

    def _read(self) -> None:
        assert self.process.stdout is not None
        for line in self.process.stdout:
            self._lines.put(line)
        self._lines.put(None)

    def line(self, limit: float) -> str | None:
        """The child's next line, or None when it ended or said nothing within the limit."""
        try:
            line = self._lines.get(timeout=limit)
        except queue.Empty:
            return None
        return None if line is None else line.decode("utf-8", "replace").rstrip("\n")

    def kill(self) -> float:
        """Kill the child; return the milliseconds until it was gone."""
        start = time.perf_counter()
        self.killed = time.time()
        self.process.kill()
        self.process.wait()
        return (time.perf_counter() - start) * 1000.0

    def send(self, text: bytes) -> None:
        """Send a kept child one text: an 8-byte length, then its bytes."""
        assert self.process.stdin is not None
        self.process.stdin.write(len(text).to_bytes(8, "big"))
        self.process.stdin.write(text)
        self.process.stdin.flush()

    def close(self) -> int:
        """Close the child's input and wait for it; kill it if it stays."""
        if self.process.stdin is not None:
            self.process.stdin.close()
        try:
            return self.process.wait(timeout=START_UP)
        except subprocess.TimeoutExpired:
            self.kill()
            return self.process.returncode


def watch(child: Child, limit: float, indent: str = "    ") -> tuple[bool, list[str]]:
    """Print a child's lines until it ends; kill it when a load does not return within the limit.

    The child says ABOUT TO LOAD before each load that may block and RETURNED after it. Returns
    whether every load returned, and the lines.
    """
    lines: list[str] = []
    waiting = False
    while True:
        line = child.line(limit if waiting else START_UP)
        if line is None:
            if child.process.poll() is not None and not waiting:
                return True, lines
            if child.process.poll() is not None:
                print(f"{indent}THE CHILD ENDED DURING A LOAD, exit {child.process.returncode}")
                return False, lines
            what = f"within {limit:g} s" if waiting else f"within {START_UP:g} s of starting"
            gone = child.kill()
            print(
                f"{indent}DID NOT RETURN {what}: the child is killed, by design "
                f"(gone {gone:.0f} ms after the kill)"
            )
            return False, lines
        if line.startswith(ABOUT_TO_LOAD):
            waiting = True
        elif line.startswith(RETURNED):
            waiting = False
        lines.append(line)
        print(f"{indent}{line}", flush=True)


def memory(pid: int) -> float:
    """A process's resident memory in MiB, as ps gives it."""
    done = subprocess.run(  # noqa: S603
        ["/bin/ps", "-o", "rss=", "-p", str(pid)], capture_output=True, text=True, check=False
    )
    return int(done.stdout.strip() or 0) / 1024.0
```

```
#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"
```

```
#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*" | sed "s|$R|<scratchpad>/i20|g"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/i20/$(basename "$s")"; done
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/i20/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
```
