"""A child of the runs made on the shipped remedy: one model, or one text, through the package.

usage: child5.py <mode> <argument> <the time it was started>

  rollback <index>   good text <index> (common.good_texts) as the instrument's model; then a patch
                     of it whose sample Vital cannot read; what load_patch raises, and whether the
                     plugin holds every float and every section as before it
  remedy <index>     text <index> (texts3.every_text) through the shipped code: through the
                     instrument where the package's writer can write it, else through the host

Before the load that may block the child says ABOUT TO LOAD and after it RETURNED; its parent
kills it when the second does not follow in time, and child.py's dead-man timer ends it whatever
its parent does. Nothing here answers an alert but the package's own host.
"""

import faulthandler
import json
import os
import sys
import time

import common as c
import texts3
from child import ABOUT_TO_LOAD, DEAD_MAN, RETURNED
from patchbay import EventBus, PresetKey
from patchbay_vital import dumps, loads
from patchbay_vital.errors import VitalError
from patchbay_vital.instrument import VitalInstrument
from patchbay_vital.patch import patch_of

UNREAD_SAMPLE = {"length": "x", "name": "x", "sample_rate": 44100, "samples": ""}
"""Section 5's sample with its four keys, its length a string: the guard passes it, Vital alerts."""


def said(call, what: str) -> tuple[str, float]:
    """One call that may block, said before and after: what it raised, and its milliseconds."""
    print(f"{ABOUT_TO_LOAD} at {time.time()!r}: {what}", flush=True)
    faulthandler.dump_traceback_later(DEAD_MAN, exit=True)
    raised = "no error"
    start = time.perf_counter()
    try:
        call()
    except VitalError as error:
        cause = error.__cause__
        raised = f"{type(error).__name__}: {error}"
        if cause is not None:
            raised += f" (from {type(cause).__name__})"
    finally:
        taken = (time.perf_counter() - start) * 1000.0
        faulthandler.cancel_dump_traceback_later()
    print(f"{RETURNED} at {time.time()!r} in {taken:.1f} ms; fired False", flush=True)
    return raised, taken


def rollback(index: int) -> None:
    _, text = c.good_texts()[index]
    model = loads(text.decode("utf-8"))
    instrument = VitalInstrument(EventBus())
    instrument.load_patch(patch_of(model))
    before = instrument.host.state()
    unread = {**model.opaque, PresetKey("sample"): json.dumps(UNREAD_SAMPLE).encode("utf-8")}
    bad = patch_of(model.model_copy(update={"opaque": unread}))
    raised, taken = said(lambda: instrument.load_patch(bad), f"good text {index}, its sample unreadable")
    after = instrument.host.state()
    floats_before, floats_after = c.floats_of(before), c.floats_of(after)
    result = {
        "i": index,
        "raised": raised,
        "ms": taken,
        "floats": len(floats_before),
        "floats_off": sum(floats_after.get(key) != held for key, held in floats_before.items()),
        "sections_off": c.sections_off(c.sections_of(after), c.sections_of(before)),
        "model_kept": dumps(instrument.preset) == dumps(model),
    }
    print(f"T {json.dumps(result)}", flush=True)


def remedy(index: int) -> None:
    _, _, text = texts3.every_text()[index]
    writer, guard = texts3.judged(text)
    instrument = VitalInstrument(EventBus())
    host = instrument.host
    before = host.state()
    if writer == "yes":
        patch = patch_of(loads(text.decode("utf-8")))
        raised, taken = said(lambda: instrument.load_patch(patch), f"text {index}, through the instrument")
        way = "the instrument"
    else:
        raised, taken = said(lambda: host.load_state(text), f"text {index}, through the host")
        way = "the host"
    after = host.state()
    floats_before, floats_after = c.floats_of(before), c.floats_of(after)
    result = {
        "i": index,
        "way": way,
        "guard": guard.split(" (")[0],
        "raised": raised,
        "ms": taken,
        "floats_off": sum(floats_after.get(key) != held for key, held in floats_before.items()),
        "sections_off": c.sections_off(c.sections_of(after), c.sections_of(before)),
    }
    if way == "the host" and raised != "no error":
        host.load_state(c.START)
        host.load_state(c.START)
        again = host.state()
        result["floats_off_after_the_start_preset"] = sum(
            c.floats_of(again).get(key) != held for key, held in floats_before.items()
        )
    print(f"T {json.dumps(result)}", flush=True)


if __name__ == "__main__":
    print(f"PID {os.getpid()}", flush=True)
    mode, argument, _started = sys.argv[1:]
    if mode == "rollback":
        rollback(int(argument))
    elif mode == "remedy":
        remedy(int(argument))
    else:
        sys.exit(f"no such mode: {mode}")

"""Issue #20, probes, run 4: the wide counts under the watch, a crash counted as a verdict.

Run 3 stopped at the first text whose child died. Here a child that dies of a signal during a
load is a verdict, with its signal, counted and not retried within its pass. Every load that may
block is made in a child this run can kill, on the main thread, under the watch run 2 saw leave
nothing on screen (the alert answered before it opens), and given LIMIT seconds from the child's
word. The run stops only at a load that does not return, or at a slip in the probe's own scripts
(a child that exits with a status of its own, which no signal gives). This process loads no
plugin. poller.py records the windows of the run's children throughout.

A verdict is a letter for each of a text's two loads, and a mark:
  A  the watch answered an alert in the load      -  the load returned and no alert was raised
  X  the child died of a signal during the load   ?  the load was not made, the child being dead
  +  after both letters: the child died after both loads returned, before it reported
  !! the child died before the text's first load was said, while its history was being made

The run's order: each child mode once on the control text; every text in a fresh child; every
text on one instance after each of three histories, the pass restarted on a new instance after a
text that killed it, and the restart marked; every text that crashed anywhere, in RERUNS more
fresh children; a soak; then macOS's crash reports of the children that died.

`collect` runs the children and `report` prints what they found, so `report` and the restart of
a pass can be run on made-up results and a fake child first (selftest4.py).
"""

import collections
import importlib.metadata
import json
import os
import platform
import signal
import subprocess
import sys
import time
from collections.abc import Callable
from dataclasses import dataclass, field
from pathlib import Path

import common as c
import texts3
from runner import ABOUT_TO_LOAD, RETURNED, START_UP, Child

HERE = Path(__file__).resolve().parent
CHILD3 = str(HERE / "child3.py")
POLLER = str(HERE / "poller.py")
WINDOWS = HERE / "out" / "windows-run4.jsonl"
WATCH = "answer"
HISTORIES = ("start", "sets", "older")
HELD_AS_GIVEN = ("custom_warps", "lfos", "modulations", "random_values")
"""The sections run 1 saw every good text come back holding as the text holds them."""
OLDER = "Factory/Presets/Ah Eh Ee Oh.vital"
RERUNS = 10
REPORTS = Path.home() / "Library" / "Logs" / "DiagnosticReports"
REPORT_WAIT = 45.0
"""Seconds given to the system to finish writing its crash reports before they are read."""

Result = dict[str, object]
Record = dict[str, object]
"""One text in one source: its verdict, the child's result where it reported one, the signal
that killed the child where one did, and that child's pid."""
Data = dict[str, object]


class StoppedError(Exception):
    """The run stopped: a load did not return, or the probe's own script failed."""


def head(title: str) -> None:
    print(f"\n== {title}", flush=True)


def clock() -> str:
    return time.strftime("%H:%M:%S")


def release(child: Child) -> None:
    """Wait for a child that is ending and close its pipes, so hundreds of children hold none."""
    try:
        child.process.wait(timeout=START_UP)
    except subprocess.TimeoutExpired:
        child.kill()
    deadline = time.time() + 5.0
    while not child.ended and time.time() < deadline:
        time.sleep(0.01)
    for pipe in (child.process.stdin, child.process.stdout):
        if pipe is not None and child.ended:
            pipe.close()


@dataclass
class Trace:
    """What a child did, read to its end."""

    results: list[Result] = field(default_factory=list)
    end: Result | None = None
    pending: list[tuple[str, bool, bool]] = field(default_factory=list)
    """The loads said since the last result: what each was, whether it returned, whether the
    watch fired in it."""
    waited_out: bool = False
    code: int | None = None
    last: str = ""

    @property
    def died(self) -> bool:
        """Killed by a signal that the run did not send."""
        return not self.waited_out and self.code is not None and self.code < 0

    @property
    def failed(self) -> bool:
        """Ended with a status of its own: a slip in the probe's script, not a crash."""
        return not self.waited_out and self.code is not None and self.code > 0

    @property
    def signal(self) -> str:
        return signal.Signals(-self.code).name if self.code is not None and self.code < 0 else ""


def follow(child: Child, limit: float, on_result: Callable[[Result], None] | None = None) -> Trace:
    """Read a child to its end; kill it when a load it said it would make does not return."""
    trace = Trace()
    waiting = False
    while True:
        line = child.line(limit if waiting else START_UP)
        if line is None:
            if not child.ended:
                child.kill()
                trace.waited_out = True
            release(child)
            trace.code = child.process.returncode
            return trace
        if line.startswith(ABOUT_TO_LOAD):
            waiting, trace.last = True, line
            trace.pending.append((line.partition(": ")[2], False, False))
        elif line.startswith(RETURNED):
            waiting = False
            if trace.pending:
                trace.pending[-1] = (trace.pending[-1][0], True, line.rstrip().endswith("fired True"))
        elif line.startswith(("T ", "RESULT ")):
            result = json.loads(line.partition(" ")[2])
            trace.results.append(result)
            trace.pending = []
            if on_result is not None:
                on_result(result)
        elif line.startswith("E "):
            trace.end = json.loads(line[2:])
        elif not line.startswith("PID "):
            print(f"    the child said: {line}", flush=True)


class Screen:
    """The poller, recording only the windows of processes new since it started."""

    def __init__(self) -> None:
        self.process = subprocess.Popen(  # noqa: S603  the probe's own interpreter and script
            [sys.executable, "-u", "-B", POLLER, str(WINDOWS), "new"],
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
        )
        assert self.process.stdout is not None
        self.process.stdout.readline()

    def polls(self) -> list[dict[str, object]]:
        with WINDOWS.open(encoding="utf-8") as log:
            return [json.loads(line) for line in log if line.endswith("\n")]

    def seen(self, children: list[Child]) -> str:
        """Over some children: how many had a window of theirs on screen at any poll."""
        time.sleep(0.3)
        pids = {child.process.pid for child in children}
        shown: dict[int, int] = {}
        sizes: set[tuple[int, int]] = set()
        front = 0
        for poll in self.polls():
            front += poll["front"] in pids
            for pid, _layer, width, height in poll["windows"]:  # type: ignore[union-attr]
                if pid in pids:
                    shown[pid] = shown.get(pid, 0) + 1
                    sizes.add((width, height))
        return (
            f"of {len(children)} children, {len(shown)} had a window of theirs on screen at a poll "
            f"(polls with one, by child: {sorted(shown.values())}; sizes seen: {sorted(sizes)}); "
            f"one of them was the front application at {front} polls"
        )

    def others(self, children: list[Child]) -> str:
        """Windows of processes new since the poller started that are no child of the run."""
        pids = {child.process.pid for child in children}
        polls = self.polls()
        strangers = sum(any(window[0] not in pids for window in poll["windows"]) for poll in polls)  # type: ignore[union-attr]
        return f"a window of a process new since the run began that is no child of it: at {strangers} of {len(polls)} polls"

    def stop(self) -> None:
        assert self.process.stdin is not None
        self.process.stdin.close()
        self.process.wait(timeout=10)


def stopped(what: str, child: Child, trace: Trace, screen: Screen) -> StoppedError:
    if trace.failed:
        return StoppedError(
            f"THE PROBE'S OWN SCRIPT FAILED: {what}: its child ended with status {trace.code}, which no "
            f"signal gives. Nothing is concluded from it. The run stops here; nothing further is run."
        )
    return StoppedError(
        f"A LOAD DID NOT RETURN within {c.LIMIT:g} s and its child was killed by the run: {what}. The "
        f"child's last word: {trace.last!r}. What was on screen: {screen.seen([child])}. The run stops "
        f"here; nothing further is run."
    )


def mark(fired: object) -> str:
    """Two loads' verdicts as two letters: A an alert, - none."""
    return "".join("A" if one else "-" for one in fired)  # type: ignore[union-attr]


def verdict_of(pending: list[tuple[str, bool, bool]]) -> str:
    """The verdict of a text whose child died before reporting it, from the loads it said."""
    if not pending:
        return "!!"
    letters = "".join(("A" if fired else "-") if returned else "X" for _, returned, fired in pending)
    if all(returned for _, returned, _ in pending) and len(pending) >= 2:  # noqa: PLR2004
        return letters[:2] + "+"
    return letters.ljust(2, "?")


def crashed(record: Record) -> bool:
    return bool(record["signal"])


def preflight(script: str) -> None:
    """Run each mode of the child script once on the control text, a text that loads."""
    head("0. the probe's own scripts: each mode of the child, once, on the control text (text 0, which loads)")
    os.environ["PROBE_FIRST"] = "1"
    try:
        modes = (
            ("one", WATCH, "0"),
            ("batch", WATCH, "start"),
            ("batch", WATCH, "sets"),
            ("batch", WATCH, "older", "0"),
            ("soak", WATCH),
        )
        for arguments in modes:
            child = Child(*arguments, script=script)
            trace = follow(child, c.LIMIT)
            ran = trace.code == 0 and len(trace.results) == 1 and not trace.waited_out
            print(f"  {' '.join(arguments)}: {'ran' if ran else 'FAILED'} (exit {trace.code}, results {len(trace.results)})", flush=True)
            if not ran:
                msg = "THE PROBE'S OWN SCRIPTS FAILED on the control text: nothing was measured, and the run stops here."
                raise StoppedError(msg)
    finally:
        del os.environ["PROBE_FIRST"]


def fresh_child(index: int, label: str, screen: Screen, script: str, children: list[Child]) -> Record:
    """One text in a child of its own."""
    child = Child("one", WATCH, str(index), script=script)
    children.append(child)
    trace = follow(child, c.LIMIT)
    if trace.waited_out or trace.failed:
        raise stopped(f"text {index}, {label}, in a fresh child", child, trace, screen)
    if trace.died:
        return {"verdict": verdict_of(trace.pending), "result": None, "signal": trace.signal, "pid": child.process.pid}
    if len(trace.results) != 1:
        msg = f"THE PROBE'S OWN SCRIPT FAILED: text {index}, {label}: its fresh child ended with status 0 and {len(trace.results)} results. The run stops here."
        raise StoppedError(msg)
    result = trace.results[0]
    return {"verdict": mark(result["fired"]), "result": result, "signal": "", "pid": child.process.pid}


def one_pass(
    history: str, count: int, labels: list[str], screen: Screen, script: str, children: list[Child]
) -> tuple[list[Record], Result | None, list[tuple[int, str]]]:
    """Every text in turn on one instance; a new instance after a text that killed the last.

    Returns each text's record, the last instance's closing result, and each restart: the text
    the new instance began at, and why.
    """
    records: list[Record | None] = [None] * count
    restarts: list[tuple[int, str]] = []
    start = 0
    stuck = -1
    while True:
        child = Child("batch", WATCH, history, str(start), script=script)
        children.append(child)
        pid = child.process.pid

        def got(result: Result, pid: int = pid) -> None:
            records[int(result["i"])] = {"verdict": mark(result["fired"]), "result": result, "signal": "", "pid": pid}  # type: ignore[call-overload]

        trace = follow(child, c.LIMIT, got)
        following = next((index for index in range(start, count) if records[index] is None), count)
        if trace.waited_out or trace.failed:
            raise stopped(f"the pass after the history {history!r}, at text {following}, {labels[min(following, count - 1)]}", child, trace, screen)
        if not trace.died:
            if following < count:
                msg = f"the pass after the history {history!r} ended by itself at text {following} of {count}: THE PROBE'S OWN SCRIPT FAILED. The run stops here."
                raise StoppedError(msg)
            return [record for record in records if record is not None], trace.end, restarts
        if trace.pending:
            verdict = verdict_of(trace.pending)
            records[following] = {"verdict": verdict, "result": None, "signal": trace.signal, "pid": pid}
            why = f"text {following}, {labels[following]}, killed its instance ({trace.signal}, verdict {verdict})"
            start = following + 1
        elif stuck == following:
            records[following] = {"verdict": "!!", "result": None, "signal": trace.signal, "pid": pid}
            why = f"the instance died again ({trace.signal}) before text {following}'s first load was said; the text is passed over"
            start = following + 1
        else:
            stuck = following
            why = f"the instance died ({trace.signal}) with no load said, after text {following - 1} and before text {following}'s first load; that text is taken again"
            start = following
        restarts.append((start, why))
        print(f"  {clock()}  RESTART of the pass after the history {history!r}: {why}. A new instance takes the texts from {start} on; they meet a shorter history.", flush=True)
        if start >= count:
            return [record for record in records if record is not None], None, restarts


def collect(screen: Screen, count: int, labels: list[str], data: Data, script: str = CHILD3) -> None:
    """Run every child of the run into `data`; StoppedError at a load that does not return."""
    screens: dict[str, str] = {}
    data["screens"] = screens
    sources: dict[str, list[Record]] = {}
    ends: dict[str, Result | None] = {}
    restarts: dict[str, list[tuple[int, str]]] = {}
    data["sources"], data["ends"], data["restarts"] = sources, ends, restarts
    children: list[Child] = []
    data["children"] = children

    head("the run's children, as they go")
    fresh: list[Record] = []
    sources["fresh"] = fresh
    mine: list[Child] = []
    for index in range(count):
        record = fresh_child(index, labels[index], screen, script, mine)
        fresh.append(record)
        if crashed(record):
            print(f"  {clock()}  text {index}, {labels[index]}: its fresh child died ({record['signal']}), verdict {record['verdict']}", flush=True)
        if (index + 1) % 50 == 0 or index + 1 == count:
            print(f"  {clock()}  {index + 1} of {count} fresh children done", flush=True)
    children += mine
    screens["fresh"] = screen.seen(mine)
    for history in HISTORIES:
        mine = []
        sources[history], ends[history], restarts[history] = one_pass(history, count, labels, screen, script, mine)
        children += mine
        screens[history] = screen.seen(mine)
        print(f"  {clock()}  the pass after the history {history!r} done: {count} texts on {len(mine)} instances", flush=True)

    crashing = sorted({index for records in sources.values() for index, record in enumerate(records) if crashed(record)})
    again: dict[int, list[Record]] = {}
    data["again"] = again
    mine = []
    for index in crashing:
        again[index] = [fresh_child(index, labels[index], screen, script, mine) for _ in range(RERUNS)]
        died = sum(crashed(record) for record in again[index])
        print(f"  {clock()}  text {index}, {labels[index]}, again in {RERUNS} fresh children: died in {died}", flush=True)
    children += mine
    screens["again"] = screen.seen(mine)

    child = Child("soak", WATCH, script=script)
    children.append(child)
    trace = follow(child, c.LIMIT)
    if trace.waited_out or trace.failed or len(trace.results) != 1:
        if trace.died:
            data["soak_died"] = f"{trace.signal} after {len(trace.pending)} loads said"
        else:
            raise stopped("the soak", child, trace, screen)
    else:
        data["soak"] = trace.results[0]
    screens["soak"] = screen.seen([child])
    print(f"  {clock()}  the soak done", flush=True)
    screens["others"] = screen.others(children)


def changed_sections(text: bytes, start: dict[str, object]) -> list[str]:
    """The sections a text does not hold as the start preset does, those it lacks among them."""
    held = c.sections_of(text)
    return sorted(name for name in set(start) | set(held) if held.get(name) != start.get(name))


def kind_of(result: Result, changed: list[str]) -> str:
    """What a load that raised no alert did, by the diagnostic read after it (history `older`)."""
    if result["moved"] < result["markers"]:  # type: ignore[operator]
        return "ignored whole"
    if not changed:
        return "loaded; the text changes no section"
    status: dict[str, str] = result["status"]  # type: ignore[assignment]
    not_given = set(result["not_given"])  # type: ignore[call-overload]
    found = set()
    for name in changed:
        if name in HELD_AS_GIVEN and name not in not_given and status.get(name) is not None:
            found.add("given")
        else:
            found.add(status.get(name, "absent"))
    if found == {"given"}:
        return "loaded; the changed section held as the text gives it"
    if "before" in found:
        return "loaded; a changed section KEPT from the preset before"
    if found <= {"plain", "given"}:
        return "loaded; the changed section read as the unchanged one"
    if found <= {"plain", "given", "either"}:
        return "loaded; the changed section as before and as unchanged alike"
    return "loaded; a changed section made into something else"


def percent(end: object) -> str:
    return f"{float(end['rms_ratio']) * 100:.1f}" if isinstance(end, dict) else "none"  # type: ignore[arg-type]


def place(symbol: str) -> str:
    """A symbol without its arguments."""
    return symbol.split("(")[0].strip()


def crash_groups(dead: dict[int, tuple[int, str]], since: float, labels: list[str]) -> None:
    """Print macOS's crash reports of the run's dead children, grouped by where in Vital they died.

    `dead` gives, for each dead child's pid, the text it was loading and the source it ran in.
    """
    today = time.strftime("%Y-%m-%d")
    deadline = time.time() + REPORT_WAIT
    found: dict[int, dict[str, object]] = {}
    while True:
        for path in REPORTS.glob(f"python3.12-{today}-*.ips"):
            if path.stat().st_mtime < since:
                continue
            try:
                with path.open(encoding="utf-8", errors="replace") as report:
                    report.readline()
                    body = json.loads(report.read())
            except (OSError, ValueError):
                continue
            if body.get("pid") in dead:
                found[int(body["pid"])] = body
        if len(found) == len(dead) or time.time() > deadline:
            break
        time.sleep(3.0)
    print(f"children of the run that died of a signal: {len(dead)}; a crash report of the system's found for {len(found)} of them")
    groups: dict[str, dict[str, object]] = {}
    for pid, body in sorted(found.items()):
        images: list[dict[str, object]] = body.get("usedImages", [])  # type: ignore[assignment]
        thread = body["threads"][body.get("faultingThread", 0)]  # type: ignore[index]
        frames = [(str(images[frame["imageIndex"]].get("name", "?")), str(frame.get("symbol", "<no symbol>"))) for frame in thread.get("frames", [])]
        vital = [place(symbol) for image, symbol in frames if image == "Vital"]
        where = " < ".join(vital[:3]) if vital else f"no frame of Vital on the faulting thread; its top frame: {frames[0][0]} {place(frames[0][1])}" if frames else "no frame at all"
        exception: dict[str, object] = body.get("exception", {})  # type: ignore[assignment]
        group = groups.setdefault(where, {"reports": 0, "texts": set(), "sources": collections.Counter(), "signals": collections.Counter(), "stack": frames, "main": collections.Counter()})
        group["reports"] = int(group["reports"]) + 1  # type: ignore[call-overload]
        index, source = dead[pid]
        group["texts"].add(index)  # type: ignore[union-attr]
        group["sources"][source] += 1  # type: ignore[index]
        group["signals"][f"{exception.get('type')} {exception.get('signal')} {exception.get('subtype', '')}".strip()] += 1  # type: ignore[index]
        group["main"]["the main thread" if body.get("faultingThread") == 0 else "another thread"] += 1  # type: ignore[index]
    print(f"grouped by where in Vital the faulting thread was (its innermost frames of Vital, innermost first): {len(groups)} groups")
    for where, group in sorted(groups.items(), key=lambda item: -len(item[1]["texts"])):  # type: ignore[arg-type]
        texts = sorted(group["texts"])  # type: ignore[call-overload]
        print(f"\n  {where}")
        print(f"      texts: {len(texts)}; reports: {group['reports']}; by source: {dict(group['sources'])}; {dict(group['signals'])}; the faulting thread: {dict(group['main'])}")  # type: ignore[call-overload]
        for index in texts:
            print(f"        {index:>3}  {labels[index]}")
        print("      the faulting thread's stack in the first of these reports, each frame its image and its symbol:")
        for image, symbol in group["stack"][:24]:  # type: ignore[index]
            print(f"          {image}  {symbol[:120]}")

def dead_children(data: Data) -> dict[int, tuple[int, str]]:

        for record in records:
            if crashed(record):
                dead[int(record["pid"])] = (index, "again")  # type: ignore[call-overload]

"""A child of a probe run: it hosts its own Vital and makes the loads its parent must not.

usage: child.py <mode> [<argument>...] <the time it was started, as the parent's time.time()>

  fresh <dead-man seconds>    read one text from stdin, load it twice, print one line of marks
  kept <dead-man seconds>     load each text sent on stdin twice, one line a text, until stdin ends
  detect <case>               section 5's case loaded twice over the start preset, with markers

Before a load that may block it says ABOUT TO LOAD and after it RETURNED; its parent kills it
when the second does not follow the first in time. It also ends itself: a dead-man timer that
needs no GIL (faulthandler's) exits the process when a load outlives it, parent or no parent.
"""

import time

MAIN = time.time()

import ctypes  # noqa: E402
import faulthandler  # noqa: E402
import json  # noqa: E402
import struct  # noqa: E402
import sys  # noqa: E402

ABOUT_TO_LOAD = "ABOUT TO LOAD"
RETURNED = "RETURNED"
DEAD_MAN = 8.0
"""Seconds a load that may block is given before this process ends itself."""


def _environment(library: ctypes.CDLL) -> bytes:
    saved = ctypes.create_string_buffer(512)
    library.fegetenv(saved)
    return saved.raw


def _subnormals_kept() -> bool:
    tiny = sys.float_info.min
    return tiny / 4.0 != 0.0


def _load(host: object, text: bytes, dead_man: float) -> float:
    """One load under the dead-man timer; its milliseconds."""
    faulthandler.dump_traceback_later(dead_man, exit=True)
    start = time.perf_counter()
    host.load_state(text)  # type: ignore[attr-defined]
    taken = (time.perf_counter() - start) * 1000.0
    faulthandler.cancel_dump_traceback_later()
    return taken


def said(host: object, text: bytes, what: str) -> float:
    """One load that may block, said before and after."""
    print(f"{ABOUT_TO_LOAD}: {what}", flush=True)
    taken = _load(host, text, DEAD_MAN)
    print(f"{RETURNED} in {taken:.1f} ms", flush=True)
    return taken


def fresh(dead_man: float, spawned: float) -> None:
    """What a trial in a child of its own costs: only the host is imported, as a remedy would."""
    from patchbay_vital.host import _LIBC, VitalHost  # noqa: PLC0415

    marks: dict[str, object] = {"spawned": spawned, "main": MAIN, "imported": time.time()}
    host = VitalHost()
    marks["hosted"] = time.time()
    text = sys.stdin.buffer.read()
    marks["read"] = time.time()
    before = _environment(_LIBC)
    marks["first_ms"] = _load(host, text, dead_man)
    marks["second_ms"] = _load(host, text, dead_man)
    marks["loaded"] = time.time()
    marks["environment_kept"] = _environment(_LIBC) == before
    marks["subnormals_kept"] = _subnormals_kept()
    marks["done"] = time.time()
    print(json.dumps(marks), flush=True)


def kept(dead_man: float, spawned: float) -> None:
    """A child kept up: each text sent as an 8-byte length and its bytes, one line back."""
    from patchbay_vital.host import _LIBC, VitalHost  # noqa: PLC0415

    imported = time.time()
    host = VitalHost()
    ready = {"spawned": spawned, "main": MAIN, "imported": imported, "hosted": time.time()}
    print(json.dumps(ready), flush=True)
    stdin = sys.stdin.buffer
    while True:
        head = stdin.read(8)
        if len(head) < 8:  # noqa: PLR2004

"""The generated set of malformed texts for run 3 of issue #20's probes.

Each is the start preset with one value inside one section replaced by another JSON type, or
removed, or its array made one entry shorter or longer. Every value of every section is reached,
down to DEPTH levels below the section, through the first entry of each array (the entries of an
array in the start preset are alike). Section 5's 20 texts come first, under their own names.
Nothing here loads a text into Vital.
"""

import copy
import json
from collections.abc import Iterator

import common as c
from patchbay_vital import check_loadable, loads
from patchbay_vital.errors import VitalError

SECTIONS = ("custom_warps", "lfos", "modulations", "random_values", "sample", "wavetables")
DEPTH = 8
"""Levels below a section that are reached: a wavetable's keyframe values are at the seventh."""
REPLACEMENTS: dict[str, object] = {
    "a number": 5,
    "a string": "x",
    "an empty array": [],
    "an empty object": {},
    "null": None,
    "true": True,
}

Path = tuple[str | int, ...]


def kind(value: object) -> str:
    if isinstance(value, bool):
        return "true"
    if isinstance(value, (int, float)):
        return "a number"
    if isinstance(value, str):
        return "a string"
    if isinstance(value, list):
        return "an empty array" if not value else "an array"
    if isinstance(value, dict):
        return "an empty object" if not value else "an object"
    return "null"


def nodes(value: object, path: Path = (), depth: int = 0) -> Iterator[tuple[Path, object]]:
    """Every value at or below one, each array through its first entry, to DEPTH levels."""
    yield path, value
    if depth == DEPTH:
        return
    if isinstance(value, dict):
        for key, inner in value.items():
            yield from nodes(inner, (*path, key), depth + 1)
    elif isinstance(value, list) and value:
        yield from nodes(value[0], (*path, 0), depth + 1)


def spelled(section: str, path: Path) -> str:
    return section + "".join(f"[{step}]" if isinstance(step, int) else f".{step}" for step in path)


def changed(section: object, path: Path, change: str) -> object:
    """A copy of a section with the value at a path replaced, removed, shortened or lengthened."""
    if not path:
        return _one(section, change)
    root = copy.deepcopy(section)
    parent = root
    for step in path[:-1]:
        parent = parent[step]  # type: ignore[index]
    last = path[-1]
    if change == "removed":
        del parent[last]  # type: ignore[attr-defined]
    else:
        parent[last] = _one(parent[last], change)  # type: ignore[index]
    return root


def _one(value: object, change: str) -> object:
    if change == "one entry fewer":
        return copy.deepcopy(value[:-1])  # type: ignore[index]
    if change == "one entry more":
        return copy.deepcopy([*value, value[-1]])  # type: ignore[misc, index]
    return copy.deepcopy(REPLACEMENTS[change])


def changes_of(path: Path, value: object) -> list[str]:
    """What is done to one value: each other JSON type, and what its place allows."""
    found = [name for name in REPLACEMENTS if name != kind(value)]
    if path and isinstance(path[-1], str):
        found.append("removed")
    if isinstance(value, list) and value:
        found += ["one entry fewer", "one entry more"]
    return found


def generated() -> list[tuple[str, str, bytes]]:
    """Each generated text: its label, the section it changes, and its bytes."""
    start = json.loads(c.START)
    texts = []
    for section in SECTIONS:
        for path, value in nodes(start["settings"][section]):
            for change in changes_of(path, value):
                text = copy.deepcopy(start)
                text["settings"][section] = changed(start["settings"][section], path, change)
                label = f"{spelled(section, path)}: {change}"
                raw = json.dumps(text, sort_keys=True, separators=(",", ":")).encode("utf-8")
                texts.append((label, section, raw))
    return texts


def every_text() -> list[tuple[str, str | None, bytes]]:
    """Section 5's 20 texts, then the generated ones."""
    named: list[tuple[str, str | None, bytes]] = [
        (label, None, build()) for label, build in c.CASES.items()
    ]
    return named + list(generated())


def judged(text: bytes) -> tuple[str, str]:
    """Whether the package's writer can write a text, and what the instrument's guard does."""
    try:
        preset = loads(text.decode("utf-8"))
    except VitalError as error:
        return f"no ({type(error).__name__})", "not reached"
    except Exception as error:  # noqa: BLE001  a probe's read of a text that may be malformed
        return f"no ({type(error).__name__}, not the package's error)", "not reached"
    try:
        check_loadable(preset, c.MAP)
    except VitalError as error:
        return "yes", f"refuses ({str(error)[:60]})"
    return "yes", "passes"


if __name__ == "__main__":
    texts = every_text()
    made = generated()
    print(f"texts: {len(texts)}, of them generated {len(made)}, distinct {len({raw for _, _, raw in texts})}")
    by_section: dict[str, int] = {}
    for _, section, _ in made:
        by_section[section] = by_section.get(section, 0) + 1
    print(f"generated, by section: {by_section}")
    verdicts: dict[tuple[str, str], int] = {}
    for _, _, raw in texts:
        writer, guard = judged(raw)
        key = (writer.split(" (")[0], guard.split(" (")[0])
        verdicts[key] = verdicts.get(key, 0) + 1
    print(f"(the writer can write it, the guard): {verdicts}")
    for label, _, _ in made[:6] + made[-6:]:
        print("  ", label)

"""A second process that watches the screen while a probe run's children load.

usage: poller.py <out file> [new]

With `new`, only the windows of processes that owned none at the first poll are written: those
of a run's children, and not those of what was on screen before it.

20 times a second it writes one JSON line: the time, the front application's pid (lsappinfo),
and every window on screen as [owner pid, layer, width, height] (CGWindowListCopyWindowInfo,
which gives a window's owner and bounds without the Screen Recording permission). It ends when
its input closes. It loads no plugin and opens no window.
"""

import ctypes
import ctypes.util
import json
import subprocess
import sys
import threading
import time

_cg = ctypes.CDLL(ctypes.util.find_library("CoreGraphics"))
_cf = ctypes.CDLL(ctypes.util.find_library("CoreFoundation"))

_cg.CGWindowListCopyWindowInfo.restype = ctypes.c_void_p
_cg.CGWindowListCopyWindowInfo.argtypes = [ctypes.c_uint32, ctypes.c_uint32]
_cf.CFArrayGetCount.restype = ctypes.c_long
_cf.CFArrayGetCount.argtypes = [ctypes.c_void_p]
_cf.CFArrayGetValueAtIndex.restype = ctypes.c_void_p
_cf.CFArrayGetValueAtIndex.argtypes = [ctypes.c_void_p, ctypes.c_long]
_cf.CFDictionaryGetValue.restype = ctypes.c_void_p
_cf.CFDictionaryGetValue.argtypes = [ctypes.c_void_p, ctypes.c_void_p]
_cf.CFNumberGetValue.restype = ctypes.c_bool
_cf.CFNumberGetValue.argtypes = [ctypes.c_void_p, ctypes.c_int, ctypes.c_void_p]
_cf.CFRelease.argtypes = [ctypes.c_void_p]


class _Rect(ctypes.Structure):
    _fields_ = (("x", ctypes.c_double), ("y", ctypes.c_double), ("w", ctypes.c_double), ("h", ctypes.c_double))


_cg.CGRectMakeWithDictionaryRepresentation.restype = ctypes.c_bool
_cg.CGRectMakeWithDictionaryRepresentation.argtypes = [ctypes.c_void_p, ctypes.POINTER(_Rect)]

ON_SCREEN_ONLY = 1
LONG_LONG = 11
"""kCFNumberLongLongType."""
PERIOD = 0.05


def _key(name: str) -> int:
    return int(ctypes.c_void_p.in_dll(_cg, name).value or 0)


OWNER, LAYER, BOUNDS = _key("kCGWindowOwnerPID"), _key("kCGWindowLayer"), _key("kCGWindowBounds")


def _number(dictionary: int, key: int) -> int:
    value = ctypes.c_longlong(0)
    number = _cf.CFDictionaryGetValue(dictionary, key)
    if number:
        _cf.CFNumberGetValue(number, LONG_LONG, ctypes.byref(value))
    return int(value.value)


def windows(everything: bool = False) -> list[list[int]]:
    """Every window on screen, or every window there is: owner's pid, layer, width, height."""
    found = []
    array = _cg.CGWindowListCopyWindowInfo(0 if everything else ON_SCREEN_ONLY, 0)
    if not array:
        return found
    for index in range(_cf.CFArrayGetCount(array)):
        entry = _cf.CFArrayGetValueAtIndex(array, index)
        rect = _Rect()
        bounds = _cf.CFDictionaryGetValue(entry, BOUNDS)
        if bounds:
            _cg.CGRectMakeWithDictionaryRepresentation(bounds, ctypes.byref(rect))
        found.append([_number(entry, OWNER), _number(entry, LAYER), round(rect.w), round(rect.h)])
    _cf.CFRelease(array)
    return found


def front() -> int | None:
    """The front application's pid, as lsappinfo gives it."""
    asn = subprocess.run(["/usr/bin/lsappinfo", "front"], capture_output=True, text=True, check=False).stdout.strip()  # noqa: S603
    if not asn:
        return None
    line = subprocess.run(["/usr/bin/lsappinfo", "info", "-only", "pid", asn], capture_output=True, text=True, check=False).stdout  # noqa: S603
    digits = line.strip().rpartition("=")[2]
    return int(digits) if digits.isdigit() else None


def main(out: str, new_only: bool = False) -> None:
    known: set[int] | None = None
    done = threading.Event()
    threading.Thread(target=lambda: (sys.stdin.read(), done.set()), daemon=True).start()
    with open(out, "w", encoding="utf-8") as log:  # noqa: PTH123
        print("ready", flush=True)
        while not done.is_set():
            started = time.time()
            found = windows()
            if new_only:
                if known is None:
                    known = {window[0] for window in found}
                found = [window for window in found if window[0] not in known]
            log.write(json.dumps({"t": started, "front": front(), "windows": found}) + "\n")
            log.flush()
            time.sleep(max(0.0, PERIOD - (time.time() - started)))


if __name__ == "__main__":
    main(sys.argv[1], new_only=sys.argv[2:] == ["new"])

"""How a probe run starts a child, reads it under a time limit, and kills what does not return."""

import queue
import subprocess
import sys
import threading
import time
from pathlib import Path

HERE = Path(__file__).resolve().parent
CHILD = str(HERE / "child.py")
ABOUT_TO_LOAD = "ABOUT TO LOAD"
RETURNED = "RETURNED"
START_UP = 60.0
"""Seconds a child is given for everything that is not a load that may block."""


class Child:
    """A child process whose lines are read under a time limit."""

    every: list["Child"] = []
    """Each child a run started, so the run can show at its end that none is left."""

    def __init__(self, *arguments: str, stdin: bool = False, script: str = CHILD) -> None:
        Child.every.append(self)
        self.started = time.time()
        self.killed: float | None = None
        self.ended = False
        """Whether the child closed its output, which a line of None alone does not say."""
        self.process = subprocess.Popen(  # noqa: S603  the probe's own interpreter and script
            [sys.executable, "-u", "-B", script, *arguments, repr(self.started)],
            stdin=subprocess.PIPE if stdin else subprocess.DEVNULL,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
        )
        self._lines: queue.Queue[bytes | None] = queue.Queue()
        threading.Thread(target=self._read, daemon=True).start()

    def _read(self) -> None:
        assert self.process.stdout is not None
        for line in self.process.stdout:
            self._lines.put(line)
        self.ended = True
        self._lines.put(None)

    def line(self, limit: float) -> str | None:
        """The child's next line, or None when it ended or said nothing within the limit."""
        try:
            line = self._lines.get(timeout=limit)
        except queue.Empty:
            return None
        return None if line is None else line.decode("utf-8", "replace").rstrip("\n")

    def kill(self) -> float:
        """Kill the child; return the milliseconds until it was gone."""
        start = time.perf_counter()
        self.killed = time.time()
        self.process.kill()
        self.process.wait()
        return (time.perf_counter() - start) * 1000.0

    def send(self, text: bytes) -> None:
        """Send a kept child one text: an 8-byte length, then its bytes."""
        assert self.process.stdin is not None
        self.process.stdin.write(len(text).to_bytes(8, "big"))
        self.process.stdin.write(text)
        self.process.stdin.flush()

    def close(self) -> int:
        """Close the child's input and wait for it; kill it if it stays."""
        if self.process.stdin is not None:
            self.process.stdin.close()
        try:
            return self.process.wait(timeout=START_UP)
        except subprocess.TimeoutExpired:
            self.kill()
            return self.process.returncode


def watch(child: Child, limit: float, indent: str = "    ") -> tuple[bool, list[str]]:
    """Print a child's lines until it ends; kill it when a load does not return within the limit.

    The child says ABOUT TO LOAD before each load that may block and RETURNED after it. Returns
    whether every load returned, and the lines.
    """
    lines: list[str] = []
    waiting = False
    while True:
        line = child.line(limit if waiting else START_UP)
        if line is None:
            if child.process.poll() is not None and not waiting:
                return True, lines
            if child.process.poll() is not None:
                print(f"{indent}THE CHILD ENDED DURING A LOAD, exit {child.process.returncode}")
                return False, lines
            what = f"within {limit:g} s" if waiting else f"within {START_UP:g} s of starting"
            gone = child.kill()
            print(
                f"{indent}DID NOT RETURN {what}: the child is killed, by design "
                f"(gone {gone:.0f} ms after the kill)"
            )
            return False, lines
        if line.startswith(ABOUT_TO_LOAD):
            waiting = True
        elif line.startswith(RETURNED):
            waiting = False
        lines.append(line)
        print(f"{indent}{line}", flush=True)


def memory(pid: int) -> float:
    """A process's resident memory in MiB, as ps gives it."""
    done = subprocess.run(  # noqa: S603
        ["/bin/ps", "-o", "rss=", "-p", str(pid)], capture_output=True, text=True, check=False
    )
    return int(done.stdout.strip() or 0) / 1024.0

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"

#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*" | sed "s|$R|<scratchpad>/i20|g"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/i20/$(basename "$s")"; done
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/i20/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
