ALERTING = "MW4"
"""The one mutant under which a real alert opens: its watch answers nothing, so the test's child
waits on the alert until it is killed at the test's own limit. It is run first, and the clock is
written to two files as it starts and as it ends, so the person at the screen can be told."""
MUTANTS.sort(key=lambda mutant: mutant[0] != ALERTING)
OUT = Path(__file__).resolve().parent / "out"

FAILED = re.compile(r"^(?:FAILED|ERROR) (\S+)", re.MULTILINE)


def copy_of(where: Path, source: Path) -> None:
    for name in ("src", "tests"):
        shutil.copytree(source / name, where / name, ignore=shutil.ignore_patterns("__pycache__"))
    shutil.copy(source / "pyproject.toml", where / "pyproject.toml")


def run(where: Path) -> tuple[int, list[str], str]:
    """Run the unit tests of a copy; the exit status, the tests that failed, the last line."""
    environment = {**os.environ, "PYTHONPATH": str(where / "src"), "PYTHONDONTWRITEBYTECODE": "1"}
    done = subprocess.run(  # noqa: S603 - this interpreter, on the probe's own copy
        [sys.executable, "-m", "pytest", "-c", "pyproject.toml", "-p", "no:doctest", "-p", "no:cacheprovider", "-rfE", "tests/unit"],
        cwd=where, env=environment, capture_output=True, text=True, check=False,
    )
    failed = sorted({name.split("::", 1)[1].split("[")[0] for name in FAILED.findall(done.stdout) if "::" in name})
    errors = sorted({name for name in FAILED.findall(done.stdout) if "::" not in name})
    return done.returncode, failed + errors, done.stdout.strip().splitlines()[-1] if done.stdout.strip() else done.stderr.strip()[-200:]


def main() -> None:
    with tempfile.TemporaryDirectory() as scratch:
        base = Path(scratch)
        print("== 0. the copy unchanged, and the shipped tests on the source before the remedy")
        clean = base / "clean"
        copy_of(clean, ROOT)
        status, failed, last = run(clean)
        print(f"the copy unchanged: exit {status}; failed {failed}; {last}")
        if status != 0:
            sys.exit("THE COPY UNCHANGED DOES NOT PASS: no mutant means anything")
        old = base / "old"
        old.mkdir()
        (old / "tests").mkdir()
        shutil.copytree(ROOT / "tests", old / "tests", dirs_exist_ok=True, ignore=shutil.ignore_patterns("__pycache__"))
        shutil.copy(ROOT / "pyproject.toml", old / "pyproject.toml")
        archive = subprocess.run(["git", "archive", "HEAD", "src"], cwd=ROOT, capture_output=True, check=True)  # noqa: S603, S607
        subprocess.run(["tar", "-x", "-C", str(old)], input=archive.stdout, check=True)  # noqa: S603, S607
        status, failed, last = run(old)
        head = subprocess.run(["git", "rev-parse", "--short", "HEAD"], cwd=ROOT, capture_output=True, text=True, check=True).stdout.strip()  # noqa: S603, S607
        print(f"the shipped tests on src as it is at HEAD ({head}), before the remedy: exit {status}; {last}")
        print(f"    not collected or failed: {failed}")

        print("\n== 1. each mutant: one change to the shipped source, and the tests that fail on it")
        caught = 0
        for name, what, file, before, after, expected in MUTANTS:
            where = base / name
            copy_of(where, ROOT)
            path = where / file
            text = path.read_text(encoding="utf-8")
            if text.count(before) != 1:
                sys.exit(f"{name}: the line it changes is found {text.count(before)} times in {file}")
            path.write_text(text.replace(before, after), encoding="utf-8")
            if name == ALERTING:
                (OUT / "alert-about-to-open.txt").write_text(time.strftime("%H:%M:%S") + "\n")
                print(f"\n{time.strftime('%H:%M:%S')}  {name} is run now: under it a real alert opens, in a child the test kills at its limit")
            status, failed, last = run(where)
            if name == ALERTING:
                (OUT / "alert-killed.txt").write_text(time.strftime("%H:%M:%S") + "\n")
                print(f"{time.strftime('%H:%M:%S')}  {name}'s tests have ended: the child with the alert is killed")
            named = [test for test in expected if test in failed]
            verdict = "caught" if status != 0 and len(named) == len(expected) else "SURVIVED" if status == 0 else "CAUGHT BY OTHER TESTS"
            caught += verdict == "caught"
            print(f"\n{name}  {what}  ({file})")
            print(f"    {verdict}: exit {status}; {last}")
            print(f"    named for it and failing: {named}")
            others = [test for test in failed if test not in expected]
            if others:
                print(f"    failing besides: {others}")
            missing = [test for test in expected if test not in failed]
            if missing:
                print(f"    NAMED FOR IT AND NOT FAILING: {missing}")
        print(f"\nmutants: {len(MUTANTS)}; caught by every test named for them: {caught}; not: {len(MUTANTS) - caught}")


if __name__ == "__main__":
    main()

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"

#!/usr/bin/env bash
# usage: transcript.sh <out file> <title> <script to embed>... -- <command...>   (from the repository root)
# Writes one transcript outside the tree: its date and branch, the machine and the tree before,
# the command's whole output, the tree and the machine after, the two trees compared, and every
# script it ran, embedded. The caller copies the file into the spec's verification/ afterwards.
set -uo pipefail
R=$(cd "$(dirname "$0")" && pwd)
out=$1; title=$2; shift 2
scripts=()
while [ "$1" != "--" ]; do scripts+=("$1"); shift; done
shift
{
  echo "$title, run $(date '+%Y-%m-%d %H:%M %Z') on branch"
  echo "$(git branch --show-current), Vital at /Library/Audio/Plug-Ins/VST3/Vital.vst3"
  echo "the machine before the run: $(bash "$R/machine.sh")"
  echo "the tree before the run (this transcript is written outside the tree and copied in afterwards):"
  bash "$R/tree.sh" | tee "$R/out/tree-before.txt"
  echo
  echo "\$ $*" | sed "s|$R|<scratchpad>/i20|g"
  "$@" 2>&1
  echo "exit=$?"
  echo
  echo "the tree after the run:"
  bash "$R/tree.sh" | tee "$R/out/tree-after.txt"
  if cmp -s "$R/out/tree-before.txt" "$R/out/tree-after.txt"; then echo "cmp: the tree after the run is the tree before it"; else echo "cmp: THE TREE CHANGED DURING THE RUN"; fi
  echo "the machine after the run: $(bash "$R/machine.sh")"
  echo
  echo "the scripts this run used, each embedded below with its blob:"
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do echo "    $(git hash-object "$s")  <scratchpad>/i20/$(basename "$s")"; done
  for s in "${scripts[@]}" "$R/tree.sh" "$R/machine.sh" "$R/transcript.sh"; do
    echo; echo "--- <scratchpad>/i20/$(basename "$s")"; echo '```'; cat "$s"; echo '```'
  done
} > "$out"
