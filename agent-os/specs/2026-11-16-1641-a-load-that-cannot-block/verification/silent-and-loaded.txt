4b, the verdict in a fresh child and, for a text that raised no alert and killed nothing, what
the state read after it showed (the history `older`). The remedy run says what the shipped
package made of each text. This reads both tables and sets them side by side for the texts that
raised no alert: how many the shipped guard refuses, how many still load, and of those that
load, what Vital made of each. It reads two kept transcripts only. No plugin is loaded.

import collections
import re
import sys
from pathlib import Path

import texts3

RUN_4 = re.compile(r"^\s+(\d+)\s+(yes|no )\s*/\s*(passes|refuses|not reached)\s+(\S+)\s+\S+\s+\S+\s+\S+\s+(.*)$")
SHIPPED = re.compile(r"^\s+(\d+)\s+(the instrument|the host|its child died)\s+(killed its child|raised Vital's alert|raised no alert)\s+(.+?)\s{2,}\S")


def table(transcript: Path, first: str, last: str) -> list[str]:
    lines = transcript.read_text(encoding="utf-8").splitlines()
    start = next(n for n, line in enumerate(lines) if first in line)
    end = next(n for n, line in enumerate(lines) if n > start and line.startswith(last))
    return lines[start + 1 : end]


def main(run_4: Path, shipped: Path) -> None:
    labels = [label for label, _, _ in texts3.every_text()]
    sections = [section for _, section, _ in texts3.every_text()]
    did: dict[int, str] = {}
    writer: dict[int, str] = {}
    for line in table(run_4, "what the load did (history older)", "the same verdict"):
        match = RUN_4.match(line)
        if match:
            number = int(match.group(1))
            rest = match.group(5).split("   [the pass")[0]
            did[number] = rest.rsplit(labels[number], 1)[0].strip()
            writer[number] = match.group(2).strip()
    outcome: dict[int, tuple[str, str, str]] = {}
    for line in table(shipped, "through the shipped package              floats", "texts whose child reported"):
        match = SHIPPED.match(line)
        if match:
            outcome[int(match.group(1))] = (match.group(2), match.group(3), match.group(4).strip())
    print(f"texts read from {run_4.name}: {len(did)}; from {shipped.name}: {len(outcome)}")
    silent = sorted(number for number, (_, saw, _) in outcome.items() if saw == "raised no alert")
    print(f"\ntexts that raised no alert and killed nothing in run 4: {len(silent)}")
    tally = collections.Counter(did[number] for number in silent)
    for what, count in sorted(tally.items()):
        print(f"  {count:>3}  {what}")
    print(f"  {sum(tally.values()):>3}  in all")

    print("\nwhat the shipped package makes of those, by what Vital had made of each:")
    cross: collections.Counter[tuple[str, str, str]] = collections.Counter()
    for number in silent:
        way, _, became = outcome[number]
        cross[(became, way, did[number])] += 1
    for became in sorted({key[0] for key in cross}):
        total = sum(count for key, count in cross.items() if key[0] == became)
        print(f"\n  {became}: {total}")
        for way in ("the instrument", "the host"):
            part = {key[2]: count for key, count in cross.items() if key[0] == became and key[1] == way}
            if part:
                print(f"    given to {way}: {sum(part.values())}")
                for what, count in sorted(part.items()):
                    print(f"      {count:>3}  {what}")
    loaded = [number for number in silent if outcome[number][2] == "loaded"]
    print(f"\nthe {len(loaded)} that still load through the shipped package, by what Vital made of each, both ways together:")
    both = collections.Counter(did[number] for number in loaded)
    for what, count in sorted(both.items()):
        print(f"  {count:>3}  {what}")
    print(f"  {sum(both.values()):>3}  in all")
    print("and by the section each changes, for those not held as the text gives them and not without a changed section:")
    astray = [number for number in loaded if "held as the text gives it" not in did[number] and "changes no section" not in did[number]]
    by_section = collections.Counter((did[number], sections[number] or "(section 5's own)") for number in astray)
    for (what, section), count in sorted(by_section.items()):
        print(f"  {count:>3}  {section:<16} {what}")
    print(f"  {len(astray):>3}  in all, of the {len(loaded)}")
    print("\neach of the texts that still load and are not held as given, by name:")
    for number in astray:
        print(f"  {number:>3}  given to {outcome[number][0]:<14}  the writer can write it: {writer[number]:<3}  {did[number]:<62} {labels[number]}")


if __name__ == "__main__":
    main(Path(sys.argv[1]), Path(sys.argv[2]))

"""The generated set of malformed texts for run 3 of issue #20's probes.

Each is the start preset with one value inside one section replaced by another JSON type, or
removed, or its array made one entry shorter or longer. Every value of every section is reached,
down to DEPTH levels below the section, through the first entry of each array (the entries of an
array in the start preset are alike). Section 5's 20 texts come first, under their own names.
Nothing here loads a text into Vital.
"""

import copy
import json
from collections.abc import Iterator

import common as c
from patchbay_vital import check_loadable, loads
from patchbay_vital.errors import VitalError

SECTIONS = ("custom_warps", "lfos", "modulations", "random_values", "sample", "wavetables")
DEPTH = 8
"""Levels below a section that are reached: a wavetable's keyframe values are at the seventh."""
REPLACEMENTS: dict[str, object] = {
    "a number": 5,
    "a string": "x",
    "an empty array": [],
    "an empty object": {},
    "null": None,
    "true": True,
}

Path = tuple[str | int, ...]


def kind(value: object) -> str:
    if isinstance(value, bool):
        return "true"
    if isinstance(value, (int, float)):
        return "a number"
    if isinstance(value, str):
        return "a string"
    if isinstance(value, list):
        return "an empty array" if not value else "an array"
    if isinstance(value, dict):
        return "an empty object" if not value else "an object"
    return "null"


def nodes(value: object, path: Path = (), depth: int = 0) -> Iterator[tuple[Path, object]]:
    """Every value at or below one, each array through its first entry, to DEPTH levels."""
    yield path, value
    if depth == DEPTH:
        return
    if isinstance(value, dict):
        for key, inner in value.items():
            yield from nodes(inner, (*path, key), depth + 1)
    elif isinstance(value, list) and value:
        yield from nodes(value[0], (*path, 0), depth + 1)


def spelled(section: str, path: Path) -> str:
    return section + "".join(f"[{step}]" if isinstance(step, int) else f".{step}" for step in path)


def changed(section: object, path: Path, change: str) -> object:
    """A copy of a section with the value at a path replaced, removed, shortened or lengthened."""
    if not path:
        return _one(section, change)
    root = copy.deepcopy(section)
    parent = root
    for step in path[:-1]:
        parent = parent[step]  # type: ignore[index]
    last = path[-1]
    if change == "removed":
        del parent[last]  # type: ignore[attr-defined]
    else:
        parent[last] = _one(parent[last], change)  # type: ignore[index]
    return root


def _one(value: object, change: str) -> object:
    if change == "one entry fewer":
        return copy.deepcopy(value[:-1])  # type: ignore[index]
    if change == "one entry more":
        return copy.deepcopy([*value, value[-1]])  # type: ignore[misc, index]
    return copy.deepcopy(REPLACEMENTS[change])


def changes_of(path: Path, value: object) -> list[str]:
    """What is done to one value: each other JSON type, and what its place allows."""
    found = [name for name in REPLACEMENTS if name != kind(value)]
    if path and isinstance(path[-1], str):
        found.append("removed")
    if isinstance(value, list) and value:
        found += ["one entry fewer", "one entry more"]
    return found


def generated() -> list[tuple[str, str, bytes]]:
    """Each generated text: its label, the section it changes, and its bytes."""
    start = json.loads(c.START)
    texts = []
    for section in SECTIONS:
        for path, value in nodes(start["settings"][section]):
            for change in changes_of(path, value):
                text = copy.deepcopy(start)
                text["settings"][section] = changed(start["settings"][section], path, change)
                label = f"{spelled(section, path)}: {change}"
                raw = json.dumps(text, sort_keys=True, separators=(",", ":")).encode("utf-8")
                texts.append((label, section, raw))
    return texts


def every_text() -> list[tuple[str, str | None, bytes]]:
    """Section 5's 20 texts, then the generated ones."""
    named: list[tuple[str, str | None, bytes]] = [
        (label, None, build()) for label, build in c.CASES.items()
    ]
    return named + list(generated())


def judged(text: bytes) -> tuple[str, str]:
    """Whether the package's writer can write a text, and what the instrument's guard does."""
    try:
        preset = loads(text.decode("utf-8"))
    except VitalError as error:
        return f"no ({type(error).__name__})", "not reached"
    except Exception as error:  # noqa: BLE001  a probe's read of a text that may be malformed
        return f"no ({type(error).__name__}, not the package's error)", "not reached"
    try:
        check_loadable(preset, c.MAP)
    except VitalError as error:
        return "yes", f"refuses ({str(error)[:60]})"
    return "yes", "passes"


if __name__ == "__main__":
    texts = every_text()
    made = generated()
    print(f"texts: {len(texts)}, of them generated {len(made)}, distinct {len({raw for _, _, raw in texts})}")
    by_section: dict[str, int] = {}
    for _, section, _ in made:
        by_section[section] = by_section.get(section, 0) + 1
    print(f"generated, by section: {by_section}")
    verdicts: dict[tuple[str, str], int] = {}
    for _, _, raw in texts:
        writer, guard = judged(raw)
        key = (writer.split(" (")[0], guard.split(" (")[0])
        verdicts[key] = verdicts.get(key, 0) + 1
    print(f"(the writer can write it, the guard): {verdicts}")
    for label, _, _ in made[:6] + made[-6:]:
        print("  ", label)

def variant(top: dict[str, object] | None = None, **settings: object) -> bytes:
    """The start preset with settings replaced (None removes one) and top-level keys set."""
    changed = json.loads(START)
    for key, value in settings.items():
        if value is None:
            del changed["settings"][key]
        else:
            changed["settings"][key] = value
    changed.update(top or {})
    return json.dumps(changed, sort_keys=True, separators=(",", ":")).encode("utf-8")

def floats_of(text: bytes) -> dict[str, float]:
    """A text's float settings by key, read as the package reads JSON; none where it has none."""
    try:
        return dict(_floats(text))
    except Exception:  # noqa: BLE001  a probe's read of a text that may be malformed
        return {}


def sections_of(text: bytes) -> dict[str, object]:
    """A text's settings that are not numbers: its sections, parsed."""
    settings = json.loads(text).get("settings", {})
    if not isinstance(settings, dict):
        return {}
    return {
        key: value
        for key, value in settings.items()
        if isinstance(value, (dict, list, str))
    }


def environment() -> bytes:
    """The process's floating-point environment, as fegetenv gives it."""
    saved = ctypes.create_string_buffer(512)
    _LIBC.fegetenv(saved)
    return saved.raw


def subnormals_kept() -> bool:
    """Whether a subnormal result is still a subnormal, so flush-to-zero is off."""
    tiny = sys.float_info.min
    return tiny / 4.0 != 0.0


def plant(host: VitalHost, text: dict[str, float]) -> dict[str, tuple[float, float]]:
    """Set each marker to the end of its range the text's value is not near.

    Returns, by marker, the normalized value planted and the one the text's value has, the
    parameter's default where the text lacks the key.
    """
    planted = {}
    for key in MARKERS:
        record = MAP.find(ParameterId(key))
        target = record.scale.to_normalized(text.get(key, record.preset_default))
        normalized = 0.0 if target > 0.5 else 1.0
        host.set(ParameterId(key), normalized)
        planted[key] = (normalized, target)
    return planted


def markers_read(host: VitalHost, planted: dict[str, tuple[float, float]]) -> tuple[int, int]:
    """How many markers moved off the planted value, and how many read the text's exactly."""
    moved = exact = 0
    for key, (normalized, target) in planted.items():
        now = host.get(ParameterId(key))
        moved += now != normalized
        exact += now == target
    return moved, exact


def floats_off(state: dict[str, float], text: dict[str, float]) -> tuple[list[str], list[str]]:
    """The map's keys whose float in the state is not the text's, and, for keys the text lacks,
    those not at the parameter's default."""
    held = [key for key in KEYS if key in text and state.get(key) != text[key]]
    lacking = [key for key in KEYS if key not in text and state.get(key) != DEFAULTS[key]]
    return held, lacking


def sections_off(state: dict[str, object], text: dict[str, object]) -> list[str]:
    """The names of the text's sections the state does not hold as the text does."""
    return sorted(name for name in text if state.get(name) != text[name])


def timed(call: Callable[[], object]) -> tuple[float, object]:
    """Run a call; return its milliseconds and what it gave."""
    start = time.perf_counter()
    result = call()
    return (time.perf_counter() - start) * 1000.0, result


def spread(values: list[float]) -> str:
    """A sample's least, median and greatest, to one decimal."""
    return (
        f"min {min(values):.1f}, median {statistics.median(values):.1f}, max {max(values):.1f}"
    )

#!/usr/bin/env bash
# usage: tree.sh   (run from the repository root)
# Prints HEAD, git's status of the work tree and the blob of every path that status names, so two
# calls can be compared with cmp: the tree a run read, and the tree it left.
set -euo pipefail
echo "  HEAD $(git rev-parse HEAD)"
git status --porcelain=v1 --untracked-files=all > "${TMPDIR:-/tmp}/tree.$$"
echo "  git status --porcelain=v1 --untracked-files=all, $(wc -l < "${TMPDIR:-/tmp}/tree.$$" | tr -d ' ') lines:"
sed 's/^/    /' "${TMPDIR:-/tmp}/tree.$$"
echo "  git hash-object of each of those paths (a deleted one marked so):"
cut -c4- "${TMPDIR:-/tmp}/tree.$$" | LC_ALL=C sort | while IFS= read -r path; do
  if [ -e "$path" ]; then echo "    $(git hash-object "$path")  $path"; else echo "    (deleted)  $path"; fi
done
rm -f "${TMPDIR:-/tmp}/tree.$$"
